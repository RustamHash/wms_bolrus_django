def shipping_grades(consignee) -> list:
    """Сорта, которые можно отгрузить этому грузополучателю.

    Список категории закрывает всех её контрагентов. Пустая категория оставляет
    галочки на самом грузополучателе. Пусто и там — можно любой сорт.
    """
    if consignee is None:
        return []
    if consignee.category_id:
        grades = list(consignee.category.allowed_grades.all())
        if grades:
            return sorted(grades, key=lambda item: (item.sort_order, item.name))
    return sorted(consignee.allowed_grades.all(), key=lambda item: (item.sort_order, item.name))


def allowed_grade_ids(consignee) -> set[int] | None:
    grades = shipping_grades(consignee)
    if not grades:
        return None
    return {item.id for item in grades}


def effective_grade_ids(line, consignee) -> set[int] | None:
    cached = getattr(line, "_effective_grades", None)
    if cached is not None:
        return cached[0]
    line_ids = {item.id for item in line.grades.all()} or None
    party_ids = allowed_grade_ids(consignee)
    if line_ids is None:
        result = party_ids
    elif party_ids is None:
        result = line_ids
    else:
        result = line_ids & party_ids
    line._effective_grades = (result,)
    return result


def effective_grade_names(line, consignee) -> str:
    ids = effective_grade_ids(line, consignee)
    if not ids:
        return ""
    found = {}
    for item in line.grades.all():
        found[item.id] = item
    if consignee is not None:
        for item in shipping_grades(consignee):
            found.setdefault(item.id, item)
    chosen = sorted((found[item_id] for item_id in ids if item_id in found), key=lambda item: (item.sort_order, item.name))
    return ", ".join(item.name for item in chosen)


def grade_names(consignee) -> str:
    if consignee is None:
        return ""
    return ", ".join(item.name for item in shipping_grades(consignee))


def grade_for_consignee(consignee):
    """Сортность по категории грузополучателя. Пусто, если категория не задана или сорт ей запрещён."""
    if consignee is None or not consignee.category_id:
        return None
    grade = consignee.category.grade
    party = allowed_grade_ids(consignee)
    if party is not None and grade.id not in party:
        return None
    return grade


def _write_category_grade(line, grade) -> None:
    from apps.inventory.models import Movement

    if Movement.objects.filter(task_line__order_line_id=line.id).exists():
        return
    current = {item.id for item in line.grades.all()}
    if current != {grade.id}:
        line.grades.set([grade])
        line._effective_grades = ({grade.id},)
    for task_line in line.task_lines.all():
        if task_line.grade_id != grade.id:
            task_line.grade_id = grade.id
            task_line.save(update_fields=["grade"])


def stamp_order_grades(order) -> None:
    grade = grade_for_consignee(order.consignee)
    if grade is None:
        return
    lines = order.lines.filter(for_task=True, grade_manual=False).prefetch_related("grades", "task_lines")
    for line in lines:
        _write_category_grade(line, grade)


def fill_consignee_orders(consignee) -> None:
    from apps.planning.models import OrderLine

    grade = grade_for_consignee(consignee)
    if grade is None:
        return
    lines = OrderLine.objects.filter(
        order__consignee=consignee,
        order__status="open",
        for_task=True,
        grade_manual=False,
    ).prefetch_related("grades", "task_lines")
    for line in lines:
        _write_category_grade(line, grade)

def roles_required(*roles, tsd=()):
    def decorator(view):
        @wraps(view)
        def wrapped(request, *args, **kwargs):
            user = request.user
            path = request.path
            on_tsd = path.startswith("/tsd")
            login_url = "/tsd/login" if on_tsd else "/login"
            if not user.is_authenticated or not user.is_active:
                return redirect(login_url)
            if not on_tsd and request.method != "POST" and not user.opens_web():
                if user.opens_tsd() and not (path.startswith("/shift") and user.can_shift()):
                    return redirect("/tsd")
            needed = set(tsd)
            if needed and needed & user.granted_access():
                return view(request, *args, **kwargs)
            if "shift" in roles and user.can_shift():
                return view(request, *args, **kwargs)
            if on_tsd and path.rstrip("/") == "/tsd" and user.can_shift():
                return view(request, *args, **kwargs)
            if user.role != "admin":
                allowed = set(roles)
                if "office" not in allowed:
                    allowed.add("office")
                if user.role not in allowed:
                    messages.error(request, "Недостаточно прав")
                    return redirect("/tsd" if on_tsd or user.opens_tsd() else "/")
            return view(request, *args, **kwargs)

        return wrapped

    return decorator

def line_caliber(line) -> str:
    if line.caliber_mode == "exact" and line.caliber_id:
        return line.caliber.name
    if line.caliber_mode == "range" and line.caliber_from_id and line.caliber_to_id:
        return f"{line.caliber_from.name}–{line.caliber_to.name}"
    return ""


def describe_line(line) -> str:
    parts = [line.product_title()]
    caliber = line_caliber(line)
    if caliber:
        parts.append(caliber)
    grades = sorted(line.grades.all(), key=lambda item: (item.sort_order, item.name))
    if grades:
        parts.append("сортность: " + ", ".join(item.name for item in grades))
    return ", ".join(parts)


def unit_lines(unit) -> list[str]:
    lot = unit.lot
    return [
        lot.product.name,
        f"{lot.caliber.name} · {unit.grade.name}",
        f"{format_kg(unit.net_kg)} кг · {unit.packaging.name}",
        f"{lot.supplier.name} · {lot.received_on.strftime('%d.%m.%Y')}",
    ]


@dataclass
class UnitGroup:
    product: str
    caliber: str
    grade: str
    count: int = 0
    kg: Decimal = field(default_factory=lambda: Decimal("0.000"))
    units: list = field(default_factory=list)


def group_units(units) -> list[UnitGroup]:
    order: list[tuple] = []
    buckets: dict[tuple, UnitGroup] = {}
    for unit in units:
        if unit.status in ("rejected", "transformed"):
            continue
        key = (unit.lot.product.name, unit.lot.caliber.name, unit.grade.name)
        if key not in buckets:
            buckets[key] = UnitGroup(*key)
            order.append(key)
        group = buckets[key]
        group.units.append(unit)
        group.count += 1
        group.kg += as_kg(unit.net_kg)
    return [buckets[key] for key in order]

        raise WarehouseError("Некорректный режим")
    if scope == InventoryDocument.Scope.PRODUCTS and not product_ids:
        raise WarehouseError("Отметьте номенклатуру")
    document = InventoryDocument.objects.create(
        doc_date=doc_date,
        scope=scope,
        notes=notes or "",
        created_by=user,
    )
    document.number = document_number("ИН", document.id)
    document.save(update_fields=["number"])
    if product_ids:
        document.products.set(product_ids)
    return document


@transaction.atomic
def start_document(document: InventoryDocument) -> int:
    document = InventoryDocument.objects.select_for_update().get(pk=document.pk)
    if document.status != InventoryDocument.Status.DRAFT:
        raise WarehouseError("Начать можно только черновик")
    product_ids = None
    if document.scope == InventoryDocument.Scope.PRODUCTS:
        product_ids = list(document.products.values_list("pk", flat=True))
        if not product_ids:
            raise WarehouseError("Отметьте номенклатуру")
    lines = []
    for movement in stock_inbounds(product_ids):
        lines.append(
            InventoryLine(
                document=document,
                pallet_id=movement.pallet_id,
                lot_id=movement.lot_id,
                supplier_id=movement.lot.supplier_id,
                location=movement.pallet.location or "",
                book_kg=as_kg(movement.qty_kg),
            )
        )
    if lines:
        InventoryLine.objects.bulk_create(lines)
    document.status = InventoryDocument.Status.COUNTING
    document.save(update_fields=["status"])
    return len(lines)

def _post_found(line: InventoryLine, user) -> None:
    fact = as_kg(line.fact_kg)
    if fact <= 0:
        raise WarehouseError("У новой паллеты вес должен быть больше нуля")
    if not line.product_id or not line.caliber_id or not line.grade_id:
        raise WarehouseError("У новой паллеты не указан товар, калибр или сорт")
    task, receipt = _opening_task(line.document, line.supplier, user)
    lot = _lot_on_receipt(receipt, line.product_id, line.caliber_id, line.grade_id)
    task_line = TaskLine.objects.create(
        task=task,
        product_id=line.product_id,
        caliber_id=line.caliber_id,
        grade_id=line.grade_id,
        qty_kg=fact,
    )
    now = timezone.now()
    pallet = Pallet.objects.create(
        kind="inbound",
        status="closed",
        cell=line.cell,
        location=line.place,
        created_by=user,
        closed_by=user,
        closed_at=now,
    )
    pallet.number = document_number("ПЛ", pallet.id)
    pallet.barcode = barcode("PL", pallet.id)
    pallet.save(update_fields=["number", "barcode"])
    post_in(task=task, task_line=task_line, pallet=pallet, lot=lot, qty_kg=fact, user=user)
    line.pallet = pallet
    line.lot = lot
    line.save(update_fields=["pallet", "lot"])


def _post_line(line: InventoryLine, user) -> None:
    if line.pallet_id is None:
        _post_found(line, user)
        return
    movement = (
        Movement.objects.select_for_update()
        .filter(pallet_id=line.pallet_id, kind=Movement.Kind.IN)
        .select_related("lot", "pallet")
        .first()
    )
    label = line.pallet.number or f"#{line.pallet_id}"
    if movement is None:
        raise WarehouseError(f"Нет прихода по паллете {label}")
    if has_left(movement):
        raise WarehouseError(f"По паллете {label} уже есть отбор, отгрузка или переработка")
    fact = as_kg(line.fact_kg)
    if fact == 0:
        remove_in(movement)
        return
    book = as_kg(line.book_kg)
    supplier_changed = line.supplier_id != line.lot.supplier_id
    if fact == book and not supplier_changed:
        return
    lot = movement.lot
    qty = fact if fact != book else as_kg(movement.qty_kg)
    if supplier_changed:
        lot = _lot_for_supplier(movement.lot, line.supplier_id)
    if lot.id == movement.lot_id and as_kg(movement.qty_kg) == qty:
        return
    update_in(movement, lot=lot, qty_kg=qty)


@transaction.atomic
def post_document(document: InventoryDocument, user) -> None:
    document = InventoryDocument.objects.select_for_update().get(pk=document.pk)
    if document.status != InventoryDocument.Status.COUNTING:
        raise WarehouseError("Провести можно только документ в подсчёте")
    lines = list(document.lines.select_related("lot", "pallet", "supplier", "product", "caliber", "grade", "cell__zone"))
    pending = sum(1 for line in lines if line.fact_kg is None)
    if pending:
        raise WarehouseError(f"Не посчитано паллет: {pending}")
    for line in lines:
        _post_line(line, user)
    document.status = InventoryDocument.Status.POSTED
    document.posted_by = user
    document.posted_at = timezone.now()
    document.save(update_fields=["status", "posted_by", "posted_at"])


@transaction.atomic
def cancel_document(document: InventoryDocument) -> None:
    document = InventoryDocument.objects.select_for_update().get(pk=document.pk)
    if document.status not in (InventoryDocument.Status.DRAFT, InventoryDocument.Status.COUNTING):
        raise WarehouseError("Проведённый документ нельзя отменить")
    document.status = InventoryDocument.Status.CANCELLED
    document.save(update_fields=["status"])

class InventoryLine(models.Model):
    document = models.ForeignKey(
        InventoryDocument, on_delete=models.CASCADE, related_name="lines", verbose_name="Документ"
    )
    pallet = models.ForeignKey(
        "storage.Pallet",
        null=True,
        blank=True,
        on_delete=models.PROTECT,
        related_name="inventory_lines",
        verbose_name="Паллета",
    )
    lot = models.ForeignKey(
        "receiving.Lot",
        null=True,
        blank=True,
        on_delete=models.PROTECT,
        related_name="inventory_lines",
        verbose_name="Партия учёта",
    )
    product = models.ForeignKey(
        "catalog.Product",
        null=True,
        blank=True,
        on_delete=models.PROTECT,
        related_name="inventory_found_lines",
        verbose_name="Товар",
    )
    caliber = models.ForeignKey(
        "catalog.Caliber",
        null=True,
        blank=True,
        on_delete=models.PROTECT,
        related_name="inventory_found_lines",
        verbose_name="Калибр",
    )
    grade = models.ForeignKey(
        "catalog.Grade",
        null=True,
        blank=True,
        on_delete=models.PROTECT,
        related_name="inventory_found_lines",
        verbose_name="Сорт",
    )
    supplier = models.ForeignKey(
        "catalog.Supplier", on_delete=models.PROTECT, related_name="inventory_lines", verbose_name="Поставщик"
    )
    location = models.CharField("Место", max_length=120, blank=True, default="")
    cell = models.ForeignKey(
        "storage.WarehouseCell",
        null=True,
        blank=True,
        on_delete=models.PROTECT,
        related_name="inventory_lines",
        verbose_name="Ячейка",
    )
    book_kg = models.DecimalField("Учёт, кг", max_digits=12, decimal_places=3)
    fact_kg = models.DecimalField("Факт, кг", max_digits=12, decimal_places=3, null=True, blank=True)
    counted_by = models.ForeignKey(
        "accounts.User",
        null=True,
        blank=True,
        on_delete=models.PROTECT,
        related_name="inventory_lines_counted",
        verbose_name="Посчитал",
    )
    counted_at = models.DateTimeField("Посчитано", null=True, blank=True)

    @property
    def place(self) -> str:
        if self.cell_id:
            return self.cell.label()
        return self.location

    class Meta:
        db_table = "inventory_lines"
        verbose_name = "Строка инвентаризации"
        verbose_name_plural = "Строки инвентаризации"
        ordering = ["id"]
        constraints = [
            models.UniqueConstraint(fields=["document", "pallet"], name="inventory_line_document_pallet"),
        ]

    def __str__(self):
        pallet = self.pallet.number if self.pallet_id else "новая"
        return f"{self.document.number} · {pallet}"

    @property
    def found(self) -> bool:
        return self.pallet_id is None

    @property
    def delta_kg(self):
        if self.fact_kg is None:
            return None
        return self.fact_kg - self.book_kg
