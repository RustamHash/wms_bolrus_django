import re

from django.contrib import messages
from django.shortcuts import redirect

from django.contrib.auth import logout

from apps.accounts.access import opens_tsd, opens_web, path_action, screen_allows, screens_for_path

_OPERATION = re.compile(r"^/(?:tasks/\d+/(?:receive|pick|ship)|movements/\d+/delete)$")


class StaffWebGate:
    def __init__(self, get_response):
        self.get_response = get_response

    def __call__(self, request):
        user = getattr(request, "user", None)
        if user is not None and user.is_authenticated and not opens_web(user):
            path = request.path
            if path.startswith("/tsd") or path.startswith("/static") or path == "/logout":
                return self.get_response(request)
            if path.startswith("/shift") and user.can_shift():
                return self.get_response(request)
            if request.method == "POST" and _OPERATION.match(path):
                return self.get_response(request)
            if opens_tsd(user):
                return redirect("/tsd")
            logout(request)
            messages.error(request, "У пользователя нет доступа в веб и на ТСД")
            return redirect("/login")
        return self.get_response(request)


class ScreenGate:
    """Меню веба и ТСД: чтение на GET, запись на POST, удаление на /delete."""

    def __init__(self, get_response):
        self.get_response = get_response

    def __call__(self, request):
        user = getattr(request, "user", None)
        if user is None or not user.is_authenticated:
            return self.get_response(request)
        if user.is_superuser or user.role == "admin":
            return self.get_response(request)
        screens = screens_for_path(request.path)
        if not screens:
            return self.get_response(request)
        action = path_action(request.path, request.method)
        if action == "delete":
            allowed = ("delete",)
        elif action == "view":
            allowed = ("view",)
        else:
            allowed = ("change", "add")
        if any(screen_allows(user, screen, name) for screen in screens for name in allowed):
            return self.get_response(request)
        messages.error(request, "Недостаточно прав")
        return redirect("/tsd" if request.path.startswith("/tsd") else "/")

def hash_secret(secret: str) -> str:
    salt = secrets.token_bytes(16)
    digest = hashlib.pbkdf2_hmac("sha256", secret.encode("utf-8"), salt, 120_000)
    return salt.hex() + "$" + digest.hex()


def verify_secret(secret: str, stored: str | None) -> bool:
    if not stored or "$" not in stored:
        return False
    salt_hex, digest_hex = stored.split("$", 1)
    try:
        salt = bytes.fromhex(salt_hex)
    except ValueError:
        return False
    digest = hashlib.pbkdf2_hmac("sha256", secret.encode("utf-8"), salt, 120_000)
    return secrets.compare_digest(digest.hex(), digest_hex)


def new_badge_token() -> str:

def _base_name(name: str) -> str:
    cut = name
    for marker in ("ГОСТ", "ТУ", "СТО", "ТН ВЭД"):
        index = cut.find(marker)
        if index != -1:
            cut = cut[:index]
    cut = re.sub(r"\([^)]*\)", " ", cut)
    cut = re.sub(r"\bсвеж\w*\b", " ", cut, flags=re.IGNORECASE)
    return " ".join(cut.split())


def plain_title(name: str) -> str:
    text = " ".join((name or "").split())
    text = re.sub(r"\s+,", ",", text)
    chars = []
    started = False
    for char in text:
        if char.isalpha():
            chars.append(char.upper() if not started else char.lower())
            started = True
        else:
            chars.append(char)
    text = "".join(chars)
    text = re.sub(r"\bрепч\b", "репчатый", text, flags=re.IGNORECASE)
    if re.search(r"капуст", text, flags=re.IGNORECASE):
        text = re.sub(r"\bб/к\b", "белокочанная", text, flags=re.IGNORECASE)
    return " ".join(text.split())[:120]


def pack_note(name: str) -> str:
    match = re.search(r"\(([^)]*сетк[^)]*)\)", name or "", flags=re.IGNORECASE)
    if not match:
        return ""
    note = re.sub(r"(\d)\s*кг", r"\1 кг", match.group(1).lower())
    return " ".join(note.split())


def with_pack(name: str, title: str) -> str:
    note = pack_note(name)
    if note and note not in title.lower():
        title = f"{title}, {note}"
    return title[:120]


def critical_family(name: str) -> str:
    if name.lower().startswith("филе сельди"):
        return "Филе сельди"
    lowered = name.lower()
    if "заморож" in lowered:
        return "Замороженная продукция"
    family = _base_name(name)
    if "очищен" in lowered:
        family = re.sub(r"\b(?:сыр\w*|поздн\w*)\b", " ", family, flags=re.IGNORECASE)
        if re.search(r"\bкартофел", family, flags=re.IGNORECASE):
            family = re.sub(r"\bстолов\w*\b", " ", family, flags=re.IGNORECASE)
        family = " ".join(family.split())
    return with_pack(name, plain_title(family))

def shipping_grades(consignee) -> list:
    """Сорта, которые можно отгрузить этому грузополучателю.

    Список категории закрывает всех её контрагентов. Пустая категория оставляет
    галочки на самом грузополучателе. Пусто и там — можно любой сорт.
    """
    if consignee is None:
        return []
    if consignee.category_id:
        grades = list(consignee.category.allowed_grades.all())
        if grades:
            return sorted(grades, key=lambda item: (item.sort_order, item.name))
    return sorted(consignee.allowed_grades.all(), key=lambda item: (item.sort_order, item.name))


def allowed_grade_ids(consignee) -> set[int] | None:
    grades = shipping_grades(consignee)
    if not grades:
        return None
    return {item.id for item in grades}


def effective_grade_ids(line, consignee) -> set[int] | None:
    cached = getattr(line, "_effective_grades", None)
    if cached is not None:
        return cached[0]
    line_ids = {item.id for item in line.grades.all()} or None
    party_ids = allowed_grade_ids(consignee)
    if line_ids is None:
        result = party_ids
    elif party_ids is None:
        result = line_ids
    else:
        result = line_ids & party_ids
    line._effective_grades = (result,)
    return result


def effective_grade_names(line, consignee) -> str:

def stock_title(name: str) -> str:
    lowered = name.lower()
    if "морож" not in lowered:
        return critical_family(name)
    family = _base_name(name)
    if "морож" not in family.lower():
        family = f"{family} замороженная"
    return with_pack(name, plain_title(family))


def _catalog(source: Path | bytes) -> dict[str, dict[str, str]]:
    rows = read_rows(source, 1)
    header = rows.get(1) or {}
    columns = {name.strip(): column for column, name in header.items()}
    found = {}
    for number, cells in rows.items():
        if number == 1:
            continue
        code = (cells.get(columns.get("Код", ""), "") or "").strip()
        if not code:
            continue
        found[code] = {
            "name": (cells.get(columns.get("Наименование", ""), "") or "").strip(),
            "group": (cells.get(columns.get("В группе", ""), "") or "").strip(),
        }
    return found


def _used(source: Path | bytes) -> list[tuple[str, str]]:
    rows = []
    for cells in read_rows(source, 2).values():
        raw = (cells.get("A") or "").strip()
        code, _, name = raw.partition(",")
        code = code.strip()
        name = name.strip()
        if not code.startswith("НФ-") or not name:
            continue
        rows.append((code, name))
    return rows


def _group(name: str, groups: dict[str, ProductGroup]) -> ProductGroup | None:
    if not name:
        return None
    found = groups.get(name)
    if found is None:
        found = ProductGroup.objects.create(name=name)
        groups[name] = found
    return found


def _product(title: str, group: ProductGroup | None) -> Product:
    found = Product.objects.filter(name=title).first()
    if found is not None and group is not None and found.group_id not in (None, group.id):
        titled = f"{title} ({group.name})"[:120]
        found = Product.objects.filter(name=titled).first()
        title = titled
    if found is None:
        found = warehouse_product(title)
    if group is not None and found.group_id is None:
        found.group = group
        found.save(update_fields=["group"])
    return found


def _fill_known_groups(catalog: dict[str, dict[str, str]], used_codes: set[str], groups: dict[str, ProductGroup]) -> None:
    named: dict[int, set[str]] = {}
    aliases = ProductAlias.objects.filter(external_product__code__in=used_codes).select_related("external_product")
    for alias in aliases:
        group_name = (catalog.get(alias.external_product.code) or {}).get("group") or ""
        if group_name:
            named.setdefault(alias.product_id, set()).add(group_name)
    for product in Product.objects.filter(pk__in=named, group__isnull=True):
        names = named[product.id]
        if len(names) != 1:
            continue
        product.group = _group(next(iter(names)), groups)
        product.save(update_fields=["group"])


@transaction.atomic
def load_used_nomenclature(source: Path | bytes) -> dict[str, int]:
    catalog = _catalog(source)
    used = _used(source)
    groups = {item.name: item for item in ProductGroup.objects.all()}
    _fill_known_groups(catalog, {code for code, _name in used}, groups)
    created_external = 0
    created_product = 0
    created_alias = 0
    grouped = 0
    missing = 0
    products_before = set(Product.objects.values_list("id", flat=True))
    for code, name in used:
        card = catalog.get(code)
        if card is None:
            missing += 1
            continue
        existed = ExternalProduct.objects.filter(code=code).exists()
        external = upsert_external(code, card["name"] or name)
        if not existed:
            created_external += 1
        group = _group(card["group"], groups)
        if group is not None and external.group_id != group.id:
            external.group = group
            external.save(update_fields=["group"])
            grouped += 1
        if ProductAlias.objects.filter(external_product=external).exists():
            continue
        product = _product(stock_title(card["name"] or name), group)
        ensure_alias(external, product)
        created_alias += 1
        if product.id not in products_before:
            created_product += 1
            products_before.add(product.id)
    return {
        "used": len(used),
        "external": created_external,
        "products": created_product,
        "aliases": created_alias,
        "groups": grouped,
        "missing": missing,
    }

def _clean(value: str, limit: int) -> str:
    return " ".join((value or "").split())[:limit]


def upsert_supplier(name: str, code: str = "") -> Supplier:
    name = _clean(name, 200)
    code = _clean(code, 32)
    if not name:
        raise WarehouseError("Укажите поставщика")
    found = Supplier.objects.filter(code=code).first() if code else None
    if found is None:
        found = Supplier.objects.filter(name=name).first()
    if found is None:
        return Supplier.objects.create(name=name, code=code or None)
    changed = []
    if found.name != name:
        found.name = name
        changed.append("name")
    if code and found.code != code:

        changed.append("code")
    if changed:

    return found

def load_consignee_addresses(source) -> dict:
    from apps.common.xlsx import read_rows

    rows = read_rows(source)
    header = rows.get(1) or {}
    columns = {value.strip(): key for key, value in header.items()}

    def cell(row: dict, title: str) -> str:
        key = columns.get(title)
        if not key:
            return ""
        return (row.get(key) or "").strip()

    def moment(row: dict) -> float:
        raw = cell(row, "Дата").replace(",", ".")
        try:
            return float(raw)
        except ValueError:
            return 0.0

    parties = {}
    buyers = {}
    for number, row in rows.items():
        if number == 1:
            continue
        buyer_name = cell(row, "Покупатель / Поставщик")
        buyer_code = cell(row, "Код Покупателя")
        if buyer_name or buyer_code:
            buyers[buyer_code or buyer_name] = (buyer_name, buyer_code)
        address = _clean(cell(row, "Адрес доставки"), 1000)
        if not address:
            continue
        code = cell(row, "Код Грузополучателя")
        name = cell(row, "Грузополучатель")
        if not code and not name:
            code = buyer_code
            name = buyer_name
        if not code and not name:
            continue
        record = (name, code, address, moment(row), number)
        key = code or name
        previous = parties.get(key)
        if previous is None or (record[3], record[4]) >= (previous[3], previous[4]):
            parties[key] = record

    created = 0
    updated = 0
    unchanged = 0
    for name, code, address, _moment, _number in parties.values():
        existed = Consignee.objects.filter(code=code).first() if code else None

def _next_product_code() -> str:
    numbers = []
    for code, _name in CRITICAL_PRODUCTS:
        if code.startswith("Т-") and code[2:].isdigit():
            numbers.append(int(code[2:]))
    for code in Product.objects.values_list("code", flat=True):
        if code.startswith("Т-") and code[2:].isdigit():
            numbers.append(int(code[2:]))
    return f"Т-{max(numbers, default=0) + 1:03d}"
