def _optional_positive(form, name: str, label: str) -> int | None:
    raw = text(form, name)
    if not raw:
        return None
    try:
        value = int(raw)
    except ValueError as exc:
        raise WarehouseError(f"Некорректное поле «{label}»") from exc
    if value <= 0:
        raise WarehouseError(f"«{label}» должно быть больше нуля")
    return value


def apply_vehicle(item: Vehicle, form) -> Vehicle:

    item.plate_number = plate
    item.brand_model = text(form, "brand_model")
    item.trailer_plate = text(form, "trailer_plate")
    item.load_capacity_kg = _optional_positive(form, "load_capacity_kg", "Тоннаж, кг")
    item.pallet_capacity = _optional_positive(form, "pallet_capacity", "Кол-во паллет")
    item.is_active = text(form, "is_active") == "1"
    item.save()

NO_ZONE_ID = 0


def active_route_order_exists():
    return Exists(
        RouteOrder.objects.filter(
            order_id=OuterRef("pk"),
            route__status__in=Route.ASSIGNED_STATUSES,
        )
    )


def unassigned_orders_queryset(ship_date: date | None = None):
    qs = (
        Order.objects.filter(status__in=("open", "partial"))
        .annotate(on_route=active_route_order_exists())
        .filter(on_route=False)
        .select_related("consignee", "consignee__zone", "seller")
        .prefetch_related(Prefetch("lines", queryset=OrderLine.objects.only("id", "order_id", "qty_kg")))
        .order_by("ship_date", "id")
    )
    if ship_date is not None:
        qs = qs.filter(ship_date=ship_date)
    return qs


def order_weights(orders) -> dict[int, Decimal]:
    ids = [order.id for order in orders]
    if not ids:
        return {}
    rows = (
        OrderLine.objects.filter(order_id__in=ids)
        .values("order_id")
        .annotate(total=Sum("qty_kg"))
    )
    return {row["order_id"]: row["total"] or Decimal("0") for row in rows}


def delivery_rank(order) -> tuple:
    """Зона, затем очередность точки. Ноль — без очереди, такие точки в конце зоны."""
    consignee = order.consignee if order.consignee_id else None
    zone_name = ""
    seq = 0
    party = ""
    if consignee is not None:
        party = consignee.name or ""
        seq = consignee.delivery_order or 0
        if consignee.zone_id and consignee.zone is not None:
            zone_name = consignee.zone.name or ""
    return (zone_name.lower(), seq <= 0, seq, party.lower(), order.id)


def _order_brief(order, weight: Decimal) -> dict:
    consignee = order.consignee if order.consignee_id else None
    if consignee is not None:
        party = consignee.name
        address = (consignee.address or "").strip()
        delivery_order = consignee.delivery_order or 0
    else:
        party = ""
        address = ""
        delivery_order = 0
    return {
        "id": order.id,
        "number": order.number or order.external_number or f"#{order.pk}",
        "party": party,
        "address": address,
        "weight": weight,
        "delivery_order": delivery_order,
    }


def group_orders_by_zone(orders, weights: dict[int, Decimal] | None = None) -> list[dict]:
    weights = weights or {}
    by_zone: dict[int, list] = defaultdict(list)
    for order in orders:
        zone_id = order.consignee.zone_id if order.consignee_id and order.consignee.zone_id else NO_ZONE_ID
        by_zone[zone_id].append(order)

    rows = []
    for zone_id, zone_orders in by_zone.items():
        if zone_id == NO_ZONE_ID:
            zone_name = "Без зоны"
        else:
            zone = zone_orders[0].consignee.zone
            zone_name = zone.name if zone else "Без зоны"
        addresses = {
            (order.consignee.address if order.consignee_id else "").strip()
            for order in zone_orders
        }
        addresses.discard("")
        zone_orders.sort(key=delivery_rank)
        order_rows = [_order_brief(order, weights.get(order.id) or Decimal("0")) for order in zone_orders]
        rows.append(
            {
                "zone_id": zone_id,
                "zone_name": zone_name,
                "order_count": len(zone_orders),
                "tt_count": len(addresses) or len(zone_orders),
                "weight": sum(item["weight"] for item in order_rows),
                "order_ids": [item["id"] for item in order_rows],
                "orders": order_rows,
            }
        )
    rows.sort(key=lambda row: (row["zone_name"] == "Без зоны", row["zone_name"].lower(), row["zone_id"]))
    return rows

def create_route_with_orders(
    *,
    route_date: date,
    driver: Driver | None,
    order_ids: list[int],
    name: str = "",
    status: str = Route.Status.DRAFT,
) -> Route:
    wanted = list(dict.fromkeys(order_ids))
    found = {order.id: order for order in Order.objects.filter(pk__in=wanted, status__in=("open", "partial"))}
    missing = [oid for oid in wanted if oid not in found]
    if missing:
        raise WarehouseError("Некоторые заказы не найдены или закрыты")
    taken = set(
        RouteOrder.objects.filter(
            order_id__in=wanted,
            route__status__in=Route.ASSIGNED_STATUSES,
        ).values_list("order_id", flat=True)
    )
    free = [found[oid] for oid in wanted if oid not in taken]
    if not free:
        raise WarehouseError("Выбранные заказы уже в маршрутах")

    with transaction.atomic():
        route = Route(
            route_date=route_date,
            name=name or (driver.full_name if driver else ""),
            driver=driver,
            status=status,
        )
        route.save()
        RouteOrder.objects.bulk_create(
            [
                RouteOrder(route=route, order=order, position=index)
                for index, order in enumerate(free, start=1)
            ]
        )
    return route


def add_orders_to_route(route: Route, order_ids: list[int]) -> int:
    if route.status == Route.Status.CANCELLED:
        raise WarehouseError("Нельзя менять отменённый маршрут")
    if route.status not in Route.LIVE_STATUSES:
        raise WarehouseError("Нельзя менять маршрут, который уже поставлен к отгрузке или отгружен")
    wanted = list(dict.fromkeys(order_ids))
    found = list(
        Order.objects.filter(pk__in=wanted, status__in=("open", "partial")).select_related(
            "consignee", "consignee__zone"
        )
    )
    taken = set(
        RouteOrder.objects.filter(
            order_id__in=wanted,
            route__status__in=Route.ASSIGNED_STATUSES,
        ).values_list("order_id", flat=True)
    )
    free = [order for order in found if order.id not in taken]
    if not free:
        return 0
    free.sort(key=delivery_rank)
    with transaction.atomic():
        rows = list(
            RouteOrder.objects.filter(route=route)
            .select_related("order__consignee", "order__consignee", "order__consignee__zone")
            .order_by("position", "id")
        )
        sequence = [row.order for row in rows]
        for order in free:
            sequence.insert(_delivery_slot(sequence, order), order)
        by_order = {row.order_id: row for row in rows}
        fresh = []
        for position, order in enumerate(sequence, start=1):
            row = by_order.get(order.id)
            if row is None:
                fresh.append(RouteOrder(route=route, order=order, position=position))
            elif row.position != position:
                row.position = position
                row.save(update_fields=["position"])
        if fresh:
            RouteOrder.objects.bulk_create(fresh)
    return len(free)


def _delivery_slot(sequence: list, order) -> int:
    key = delivery_rank(order)
    last = None
    for index, current in enumerate(sequence):
        if delivery_rank(current) <= key:
            last = index
    return 0 if last is None else last + 1

def delete_routes(ids: list[int]) -> tuple[list[str], list[str]]:
    """Удаляет маршруты; заказы снова свободны для маршрутизации."""
    from django.db.models.deletion import ProtectedError

    deleted = []
    failed = []
    routes = {route.id: route for route in Route.objects.filter(pk__in=ids)}
    for route_id in ids:
        route = routes.get(route_id)
        if route is None:
            failed.append(f"{route_id}: маршрут не найден")
            continue
        label = route.name or f"#{route.id}"
        try:
            with transaction.atomic():
                route.delete()
            deleted.append(label)
        except ProtectedError:
            failed.append(f"{label}: маршрут используется")
    return deleted, failed


def reorder_route_orders(route: Route, order_ids: list[int]) -> None:
    if route.status not in Route.LIVE_STATUSES:
        raise WarehouseError("Можно менять только черновик или запланированный маршрут")
    current = list(RouteOrder.objects.filter(route=route).order_by("position", "id"))
    by_order = {row.order_id: row for row in current}
    if set(order_ids) != set(by_order):
        raise WarehouseError("Список заказов не совпадает с маршрутом")
    with transaction.atomic():
        for index, order_id in enumerate(order_ids, start=1):
            row = by_order[order_id]
            if row.position != index:
                row.position = index
                row.save(update_fields=["position"])


def move_route_order(route: Route, order_id: int, direction: str) -> None:
    if route.status not in Route.LIVE_STATUSES:
        raise WarehouseError("Можно менять только черновик или запланированный маршрут")
    rows = list(RouteOrder.objects.filter(route=route).order_by("position", "id"))
    index = next((i for i, row in enumerate(rows) if row.order_id == order_id), None)
    if index is None:
        raise WarehouseError("Заказ не найден в маршруте")
    if direction == "up" and index > 0:
        rows[index], rows[index - 1] = rows[index - 1], rows[index]
    elif direction == "down" and index < len(rows) - 1:
        rows[index], rows[index + 1] = rows[index + 1], rows[index]
    else:
        return
    with transaction.atomic():
        for pos, row in enumerate(rows, start=1):
            if row.position != pos:
                row.position = pos
                row.save(update_fields=["position"])

def load_local(value: datetime) -> datetime:
    return timezone.localtime(value) if timezone.is_aware(value) else value


def set_planned_load_at(route: Route, value: datetime | None, *, warn_conflict: bool = True) -> str | None:
    warning = None
    if value is not None and route.driver_id:
        local = load_local(value)
        bucket = slot_bucket(value)
        conflict = (
            Route.objects.filter(
                driver_id=route.driver_id,
                planned_load_at__isnull=False,
            )
            .exclude(pk=route.pk)
            .exclude(status=Route.Status.CANCELLED)
        )
        for other in conflict:
            if not other.planned_load_at:
                continue
            other_local = load_local(other.planned_load_at)
            if other_local.date() == local.date() and slot_bucket(other.planned_load_at) == bucket:
                warning = f"У водителя уже есть маршрут в слоте {local.strftime('%d.%m')} {bucket}"
                break
    route.planned_load_at = value
    route.save(update_fields=["planned_load_at", "updated_at"])
    return warning if warn_conflict else None


def slot_bucket(value: datetime) -> str:
    local = load_local(value)
    minute = 0 if local.minute < 30 else 30
    return f"{local.hour:02d}:{minute:02d}"


def parse_load_datetime(load_date: date | None, raw_time: str) -> datetime | None:
    """Дата и время погрузки. Дата может отличаться от даты рейса (погрузка накануне)."""
    text = (raw_time or "").strip()
    if not text:
        return None
    if load_date is None:
        raise WarehouseError("Укажите дату погрузки")
    try:
        parts = text.split(":")
        hour = int(parts[0])
        minute = int(parts[1]) if len(parts) > 1 else 0
        naive = datetime.combine(load_date, time(hour=hour, minute=minute))
    except (TypeError, ValueError, IndexError) as exc:
        raise WarehouseError("Некорректное время погрузки") from exc
    if timezone.is_naive(naive):
        return timezone.make_aware(naive, timezone.get_current_timezone())
    return naive


def parse_load_date(raw: str) -> date | None:
    value = (raw or "").strip()
    if not value:
        return None
    try:
        return date.fromisoformat(value)
    except ValueError as exc:
        raise WarehouseError("Некорректная дата погрузки") from exc

NO_DRIVER_ID = 0


def assignment_date_key(ship_date: date | None) -> str:
    return ship_date.isoformat() if ship_date else "default"


def _read_id_map(session, store_key: str, ship_date: date | None) -> dict[int, int]:
    store = session.get(store_key) or {}
    raw = store.get(assignment_date_key(ship_date)) or {}
    result = {}
    for raw_id, driver_id in raw.items():
        try:
            item_id = int(raw_id)
            did = int(driver_id)
        except (TypeError, ValueError):
            continue
        if did >= 0:
            result[item_id] = did
    return result


def _write_id(session, store_key: str, ship_date: date | None, item_id: int, driver_id: int | None) -> None:
    store = session.get(store_key) or {}
    date_key = assignment_date_key(ship_date)
    day_map = dict(store.get(date_key) or {})
    key = str(int(item_id))
    if driver_id is None or driver_id < 0:
        day_map.pop(key, None)
    else:
        day_map[key] = int(driver_id)
    store[date_key] = day_map
    session[store_key] = store
    session.modified = True


def _drop_ids(session, store_key: str, ship_date: date | None, item_ids: Iterable[int]) -> None:
    store = session.get(store_key) or {}
    date_key = assignment_date_key(ship_date)
    day_map = dict(store.get(date_key) or {})
    for item_id in item_ids:
        day_map.pop(str(int(item_id)), None)
    store[date_key] = day_map
    session[store_key] = store
    session.modified = True


def get_session_assignments(session, ship_date: date | None) -> dict[int, int]:
    """zone_id -> driver_id; 0 = создать маршрут без водителя."""
    return _read_id_map(session, SESSION_ASSIGNMENTS_KEY, ship_date)


def set_session_assignment(session, ship_date: date | None, zone_id: int, driver_id: int | None) -> None:
    _write_id(session, SESSION_ASSIGNMENTS_KEY, ship_date, zone_id, driver_id)


def clear_session_assignments(session, ship_date: date | None, zone_ids: Iterable[int]) -> None:
    _drop_ids(session, SESSION_ASSIGNMENTS_KEY, ship_date, zone_ids)


def get_session_order_assignments(session, ship_date: date | None) -> dict[int, int]:
    """order_id -> driver_id; 0 = маршрут без водителя. Нет ключа — заказ следует зоне."""
    return _read_id_map(session, SESSION_ORDER_ASSIGNMENTS_KEY, ship_date)


def set_session_order_assignment(session, ship_date: date | None, order_id: int, driver_id: int | None) -> None:
    _write_id(session, SESSION_ORDER_ASSIGNMENTS_KEY, ship_date, order_id, driver_id)


def clear_session_order_assignments(session, ship_date: date | None, order_ids: Iterable[int]) -> None:
    _drop_ids(session, SESSION_ORDER_ASSIGNMENTS_KEY, ship_date, order_ids)


def routing_rows(ship_date: date, session) -> tuple[list[dict], dict[int, int], Decimal]:
    orders = list(unassigned_orders_queryset(ship_date))
    weights = order_weights(orders)
    rows = group_orders_by_zone(orders, weights)
    assignments = get_session_assignments(session, ship_date)
    order_assignments = get_session_order_assignments(session, ship_date)
    total_weight = Decimal("0")
    for row in rows:
        row["driver_id"] = assignments.get(row["zone_id"])
        overrides = 0
        for order in row["orders"]:
            driver_id = order_assignments.get(order["id"])
            order["driver_id"] = driver_id
            order["assigned"] = driver_id is not None
            if driver_id is not None:
                overrides += 1
        row["override_count"] = overrides
        total_weight += row["weight"] or Decimal("0")
    return rows, assignments, total_weight


def driver_load_summary(rows: list[dict], drivers) -> list[dict]:
    """Вес заказов по эффективному водителю. Перегруз — если вес больше тоннажа авто."""
    weights: dict[int, Decimal] = defaultdict(lambda: Decimal("0"))
    counts: dict[int, int] = defaultdict(int)
    for row in rows:
        zone_driver = row.get("driver_id")
        for order in row.get("orders") or []:
            driver_id = order["driver_id"] if order.get("assigned") else zone_driver
            if driver_id is None:
                continue
            weights[driver_id] += order.get("weight") or Decimal("0")
            counts[driver_id] += 1
    by_id = {driver.id: driver for driver in drivers}
    result = []
    for driver_id, weight in weights.items():
        if driver_id == NO_DRIVER_ID:
            result.append(
                {
                    "driver_id": NO_DRIVER_ID,
                    "name": "Без водителя",
                    "weight": weight,
                    "count": counts[driver_id],
                    "capacity": None,
                    "over": False,
                }
            )
            continue
        driver = by_id.get(driver_id)
        capacity = None
        if driver is not None and driver.vehicle_id and driver.vehicle and driver.vehicle.load_capacity_kg:
            capacity = driver.vehicle.load_capacity_kg
        result.append(
            {
                "driver_id": driver_id,
                "name": driver.full_name if driver is not None else f"#{driver_id}",
                "weight": weight,
                "count": counts[driver_id],
                "capacity": capacity,
                "over": capacity is not None and weight > capacity,
            }
        )
    result.sort(key=lambda item: (item["name"] == "Без водителя", item["name"].lower()))
    return result


def _row_orders(row: dict) -> list[dict]:
    orders = row.get("orders")
    if orders is not None:
        return orders
    return [{"id": order_id} for order_id in row.get("order_ids") or []]


def _order_override(order: dict) -> int | None:
    """Явный водитель заказа. None — заказ следует зоне."""
    if order.get("assigned"):
        return order.get("driver_id")
    if "assigned" in order:
        return None
    return order.get("driver_id")


def partition_route_orders(zone_rows: list[dict]) -> tuple[dict[int, list[int]], list[dict], list[int]]:
    """Заказы с явным водителем идут ему, остальные — водителю зоны.

    Возвращает заказы по водителям, маршруты «без водителя» по зонам
    и зоны, чьё назначение реально использовано.
    """
    by_driver: dict[int, list[int]] = defaultdict(list)
    no_driver_zones: list[dict] = []
    inherited_zone_ids: list[int] = []
    for row in zone_rows:
        zone_driver = row.get("driver_id")
        no_driver_ids: list[int] = []
        used_inheritance = False
        for order in _row_orders(row):
            explicit = _order_override(order)
            driver_id = explicit if explicit is not None else zone_driver
            if driver_id is None:
                continue
            if explicit is None:
                used_inheritance = True
            if driver_id == NO_DRIVER_ID:
                no_driver_ids.append(order["id"])
            else:
                by_driver[driver_id].append(order["id"])
        if no_driver_ids:
            no_driver_zones.append(
                {
                    "zone_id": row["zone_id"],
                    "zone_name": row["zone_name"],
                    "order_ids": no_driver_ids,
                }
            )
        if used_inheritance and zone_driver is not None:
            inherited_zone_ids.append(row["zone_id"])
    return by_driver, no_driver_zones, inherited_zone_ids


def consumed_for_create(zone_rows: list[dict]) -> tuple[list[int], list[int]]:
    by_driver, no_driver_zones, inherited_zone_ids = partition_route_orders(zone_rows)
    order_ids = [order_id for ids in by_driver.values() for order_id in ids]
    for zone in no_driver_zones:
        order_ids.extend(zone["order_ids"])
    return order_ids, inherited_zone_ids


def consumed_for_route(route: Route, zone_rows: list[dict]) -> tuple[list[int], list[int]]:
    target = route.driver_id if route.driver_id else NO_DRIVER_ID
    by_driver, no_driver_zones, inherited_zone_ids = partition_route_orders(zone_rows)
    if target == NO_DRIVER_ID:
        order_ids = [order_id for zone in no_driver_zones for order_id in zone["order_ids"]]
    else:
        order_ids = list(by_driver.get(target, []))
    inherited = set(inherited_zone_ids)
    zone_ids = [
        row["zone_id"]
        for row in zone_rows
        if row["zone_id"] in inherited and row.get("driver_id") == target
    ]
    return order_ids, zone_ids


def _zone_label(names: list[str]) -> str:
    unique = []
    for name in names:
        cleaned = (name or "").strip() or "Без зоны"
        if cleaned not in unique:
            unique.append(cleaned)
    return ", ".join(unique)[:255]


def _driver_zone_names(zone_rows: list[dict]) -> dict[int, list[str]]:
    names: dict[int, list[str]] = defaultdict(list)
    for row in zone_rows:
        zone_driver = row.get("driver_id")
        label = (row.get("zone_name") or "").strip() or "Без зоны"
        seen = set()
        for order in _row_orders(row):
            explicit = _order_override(order)
            driver_id = explicit if explicit is not None else zone_driver
            if driver_id is None or driver_id == NO_DRIVER_ID or driver_id in seen:
                continue
            seen.add(driver_id)
            names[driver_id].append(label)
    return names


def create_routes_from_zone_assignments(
    *,
    ship_date: date,
    zone_rows: list[dict],
) -> list[Route]:
    """Один маршрут на водителя; «Без водителя» — по одному маршруту на зону."""
    by_driver, no_driver_zones, _inherited = partition_route_orders(zone_rows)
    if not by_driver and not no_driver_zones:
        raise WarehouseError("Назначьте водителя или «Без водителя» на зоны или заказы")

    drivers = {
        driver.id: driver
        for driver in Driver.objects.filter(pk__in=by_driver.keys(), is_active=True).select_related("vehicle")
    }
    zone_names = _driver_zone_names(zone_rows)
    created = []
    for driver_id, order_ids in by_driver.items():
        driver = drivers.get(driver_id)
        if driver is None:
            raise WarehouseError("Водитель не найден или неактивен")
        unique_ids = list(dict.fromkeys(order_ids))
        if not unique_ids:
            continue
        route = create_route_with_orders(
            route_date=ship_date,
            driver=driver,
            order_ids=unique_ids,
            name=_zone_label(zone_names.get(driver_id, [])) or driver.full_name,
            status=Route.Status.DRAFT,
        )
        created.append(route)

    for row in no_driver_zones:
        unique_ids = list(dict.fromkeys(row["order_ids"]))
        if not unique_ids:
            continue
        route = create_route_with_orders(
            route_date=ship_date,
            driver=None,
            order_ids=unique_ids,
            name=row["zone_name"],
            status=Route.Status.DRAFT,
        )
        created.append(route)

    if not created:
        raise WarehouseError("Не удалось создать маршруты")
    return created


def add_assigned_orders_to_route(
    *,
    route: Route,
    zone_rows: list[dict],
) -> int:
    if route.status not in Route.LIVE_STATUSES:
        raise WarehouseError("Можно добавлять только в черновик или запланированный маршрут")
    order_ids, _zone_ids = consumed_for_route(route, zone_rows)
    if route.driver_id:
        if not order_ids:
            raise WarehouseError("Нет заказов для этого водителя")
    elif not order_ids:
        raise WarehouseError("Отметьте зоны или заказы как «Без водителя»")
    return add_orders_to_route(route, list(dict.fromkeys(order_ids)))

def _routes_for_delivery_day(day: date):
    """Все маршруты с датой доставки (рейса) = day."""
    return (
        Route.objects.filter(route_date=day)
        .exclude(status=Route.Status.CANCELLED)
        .select_related("driver")
    )


def schedule_context(day: date) -> dict:
    """График по дате доставки: список, кто во сколько грузится."""
    routes = list(
        _routes_for_delivery_day(day).annotate(
            order_count=Count("route_orders", distinct=True),
            weight=Sum("route_orders__order__lines__qty_kg"),
        )
    )
    items = []
    for route in routes:
        items.append(
            {
                "id": route.id,
                "name": route.name or f"Маршрут {route.id}",
                "driver": route.driver.full_name if route.driver_id else "Без водителя",
                "when": _load_stamp(route.planned_load_at),
                "timed": bool(route.planned_load_at),
                "order_count": route.order_count or 0,
                "weight": route.weight or Decimal("0"),
                "sort": (_sort_key_plan(route.planned_load_at), route.id),
            }
        )
    items.sort(key=lambda item: item["sort"])
    for item in items:
        del item["sort"]
    return {"day": day, "items": items}


def _tt_key(order) -> str:
    if not order.consignee_id:
        return f"order:{order.id}"
    address = (order.consignee.address or "").strip()
    if address:
        return address.casefold()
    return f"consignee:{order.consignee_id}"


def _clock(value) -> str:
    if not value:
        return ""
    local = load_local(value)
    return f"{local.hour:02d}:{local.minute:02d}"


def _load_stamp(value) -> str:
    """Дата и время плана погрузки, всегда вместе."""
    if not value:
        return ""
    local = load_local(value)
    return f"{local:%d.%m.%Y} {local.hour:02d}:{local.minute:02d}"


def _fact_label(value, delivery_day: date) -> str:
    if not value:
        return ""
    local = load_local(value)
    clock = f"{local.hour:02d}:{local.minute:02d}"
    if local.date() != delivery_day:
        return f"{local:%d.%m} {clock}"
    return clock


def _sort_key_plan(value) -> tuple:
    """Полная дата+время плана, чтобы вечерняя погрузка шла раньше утренней."""
    if not value:
        return (1, date.max, 0, 0)
    local = load_local(value)
    return (0, local.date(), local.hour, local.minute)


def schedule_print_tree(day: date) -> dict:
    """
    Все маршруты с датой доставки = day, по строке на маршрут.
    Сортировка по плану погрузки. План — дата и время, как в графике на экране.
    Строки: план | факт | водитель | маршрут | комментарий | вес | ТТ.
    """
    routes = list(
        _routes_for_delivery_day(day)
        .prefetch_related(
            Prefetch(
                "route_orders",
                queryset=RouteOrder.objects.select_related(
                    "order__consignee",
                    "order__consignee",
                ).order_by("position", "id"),
            )
        )
        .order_by("planned_load_at", "id")
    )
    all_orders = [stop.order for route in routes for stop in route.route_orders.all()]
    weights = order_weights(all_orders)

    prepared = []
    for route in routes:
        driver_name = route.driver.full_name if route.driver_id else "Без водителя"
        route_name = route.name or f"Маршрут {route.id}"
        route_weight = Decimal("0")
        tt: set[str] = set()
        for stop in route.route_orders.all():
            order = stop.order
            route_weight += weights.get(order.id) or Decimal("0")
            tt.add(_tt_key(order))
        prepared.append(
            {
                "sort": (
                    _sort_key_plan(route.planned_load_at),
                    driver_name.casefold(),
                    route_name.casefold(),
                    route.id,
                ),
                "plan": _load_stamp(route.planned_load_at),
                "fact": _fact_label(route.actual_load_at, day),
                "label": driver_name,
                "route": route_name,
                "notes": (route.notes or "").strip(),
                "weight": route_weight,
                "tt": tt,
            }
        )
    prepared.sort(key=lambda item: item["sort"])

    table_rows = []
    grand_weight = Decimal("0")
    grand_tt: set[str] = set()
    for item in prepared:
        table_rows.append(
            {
                "kind": "driver",
                "plan": item["plan"],
                "fact": item["fact"],
                "label": item["label"],
                "route": item["route"],
                "notes": item["notes"],
                "weight": item["weight"],
                "tt_count": len(item["tt"]),
            }
        )
        grand_weight += item["weight"]
        grand_tt |= item["tt"]

    return {
        "day": day,
        "rows": table_rows,
        "total_weight": grand_weight,
        "total_tt": len(grand_tt),
        "route_count": len(routes),
    }

def _stamp(obj, field: str, label: str) -> None:
    if getattr(obj, field) is not None:
        raise WarehouseError(f"{label} уже зафиксировано")
    setattr(obj, field, timezone.now())

def parse_shift_date(raw: str) -> date:
    value = (raw or "").strip()
    if not value:
        return timezone.localdate()
    try:
        return date.fromisoformat(value)
    except ValueError as exc:
        raise WarehouseError("Некорректная дата") from exc


def parse_shift_period(raw_from: str, raw_to: str, raw_date: str = "") -> tuple[date, date, str]:
    today = timezone.localdate()
    start_raw = (raw_from or "").strip()
    end_raw = (raw_to or "").strip()
    single = (raw_date or "").strip()
    if not start_raw and not end_raw and single:
        start_raw = end_raw = single
    if not start_raw and not end_raw:
        return today, today, "today"
    start = parse_shift_date(start_raw or end_raw)
    end = parse_shift_date(end_raw or start_raw)
    if end < start:
        raise WarehouseError("Дата окончания раньше начала")
    preset = ""
    if start == end:
        if start == today:
            preset = "today"
        elif start == today - timedelta(days=1):
            preset = "yesterday"
        elif start == today + timedelta(days=1):
            preset = "tomorrow"
    return start, end, preset


def task_status_label(tasks, work: str) -> str:
    if not tasks:
        return "Нет задания"
    statuses = {task.status for task in tasks}
    if work in statuses:
        return TASK_STATUS[work]
    if "open" in statuses:
        return TASK_STATUS["open"]
    if statuses <= {"closed", "picked"}:
        if statuses == {"picked"}:
            return TASK_STATUS["picked"]
        return TASK_STATUS["closed"]
    status = next(iter(statuses))
    return TASK_STATUS.get(status, status)

def revert_route(route: Route) -> Route:
    if route.status == Route.Status.DOCKED:
        route.status = Route.Status.PLANNED
        route.docked_at = None
        route.save()
        return route
    if route.status == Route.Status.COMPLETED:
        route.status = Route.Status.DOCKED
        route.actual_load_at = None
        route.save()
        return route
    raise WarehouseError("Этот маршрут нельзя вернуть")

def _posted_load_times(request, ship_date):
    """Время загрузки по водителю. Пустое поле — маршрут без плана."""
    load_day = parse_load_date(text(request.POST, "planned_load_date")) or ship_date
    times = {}
    for key, raw in request.POST.items():
        if not key.startswith("load_time_"):
            continue
        value = (raw or "").strip()
        if not value:
            continue
        try:
            driver_id = int(key.removeprefix("load_time_"))
        except ValueError as exc:
            raise WarehouseError("Некорректное время загрузки") from exc
        times[driver_id] = parse_load_datetime(load_day, value)
    return times

def _print_task_ids(order_ids, *, field: str, kind: str) -> list[int]:
    from apps.planning.models import Task

    if not order_ids:
        return []
    attr = f"{field}_id"
    grouped: dict[int, list[int]] = {}
    query = Task.objects.filter(**{f"{field}_id__in": order_ids, "kind": kind}).order_by("id")
    for task in query:
        grouped.setdefault(getattr(task, attr), []).append(task.id)
    return [task_id for order_id in order_ids for task_id in grouped.get(order_id, [])]


def _shift_tasks_pdf(request, *, source: str, kind: str, empty: str):
    from apps.printing.views import load_print_tasks, task_sheet_response

    start, end, _preset = _shift_bounds(request)
    if source == "orders":
        order_ids = [row["id"] for row in shift_orders(start, end)]
        field = "order"
    else:
        order_ids = [row["id"] for row in shift_incoming(start, end)]
        field = "purchase_order"
    tasks = load_print_tasks(_print_task_ids(order_ids, field=field, kind=kind))
    if not tasks:
        return HttpResponse(empty)
    return task_sheet_response(tasks)

    def action():
        route = Route.objects.filter(pk=route_id).first()

        fn(route)
        return route

    result, response = guarded(request, _shift_back(request), action)
    if response:
        return response
    return done(request, _shift_back(request), message)
