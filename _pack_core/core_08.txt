_font_ready = False


def ensure_font() -> None:
    global _font_ready
    if not _font_ready:
        pdfmetrics.registerFont(TTFont("WmsFont", find_font()))
        _font_ready = True


class Sheet:
    def __init__(self, pagesize):
        self.buffer = BytesIO()
        self.pagesize = pagesize
        self.canvas = canvas.Canvas(self.buffer, pagesize=pagesize)
        ensure_font()
        self.width, self.height = pagesize
        self.y = self.height - 15 * mm

    def page_break(self) -> None:
        if self.y < 25 * mm:
            self.canvas.showPage()
            ensure_font()
            self.y = self.height - 15 * mm

    def text(self, value: str, size: int = 11, gap: float | None = None) -> None:
        self.page_break()
        self.canvas.setFont("WmsFont", size)
        self.canvas.drawString(15 * mm, self.y, value[:180])
        self.y -= gap or (size + 4)

    def barcode(self, value: str) -> None:
        self.page_break()
        barcode = code128.Code128(value, barHeight=12 * mm, barWidth=0.35 * mm, humanReadable=True)
        barcode.drawOn(self.canvas, 15 * mm, self.y - 16 * mm)
        self.y -= 22 * mm

    def finish(self) -> bytes:
        self.canvas.save()
        return self.buffer.getvalue()

def _wrap(value: str, size: int, width: float) -> list[str]:
    words = (value or "").split()
    if not words:
        return [""]
    lines: list[str] = []
    current = ""
    for word in words:
        trial = word if not current else f"{current} {word}"
        if pdfmetrics.stringWidth(trial, "WmsFont", size) <= width:
            current = trial
            continue
        if current:
            lines.append(current)
        current = word
    if current:
        lines.append(current)
    return lines or [""]


def _line_caliber(line) -> str:
    if line.caliber_id:
        return line.caliber.name
    order_line = line.order_line
    if order_line is None:
        return ""
    if order_line.caliber_mode == "range" and order_line.caliber_from_id and order_line.caliber_to_id:
        return f"{order_line.caliber_from.name}–{order_line.caliber_to.name}"
    return ""


def _line_grade(line) -> str:
    if line.grade_id:
        return line.grade.name
    return ""


def _task_title(task) -> str:
    titles = {"pick": "Задание на отбор", "ship": "Задание на отгрузку", "receive": "Задание на приёмку"}
    title = titles.get(task.kind, "Задание")
    return f"{title} {task.number or task.id}"


def _product_group_name(line) -> str:
    product = line.product
    if product is not None and product.group_id:
        return product.group.name
    return "Без группы"


def _lines_by_product_group(lines) -> list[tuple[str, list]]:
    buckets: dict[str, list] = {}
    order: list[str] = []
    for line in lines:
        title = _product_group_name(line)
        if title not in buckets:
            buckets[title] = []
            order.append(title)
        buckets[title].append(line)
    order.sort(key=lambda name: (name == "Без группы", name.casefold()))
    return [(title, buckets[title]) for title in order]

def _text(pdf, x, y, value: str, size: int = 11) -> float:
    pdf.setFont("WmsFont", size)
    pdf.drawString(x, y, value)
    return y - (size + 4)


def _task_sheet(pdf, task, left, right, height, columns, statuses) -> float:
    y = height - 14 * mm
    pdf.setFont("WmsFont", 16)
    pdf.drawString(left, y, _task_title(task)[:90])
    y -= 8 * mm
    status = statuses.get(task.status, task.status)
    y = _text(pdf, left, y, f"Статус: {status}", 10)
    order = task.order
    purchase = task.purchase_order
    if order is not None:
        number = order.number or ""
        y = _text(pdf, left, y, f"Заказ {number}".strip())
        y = _text(pdf, left, y, f"Дата отгрузки: {order.ship_date.strftime('%d.%m.%Y')}")
        if order.consignee_id:
            y = _text(pdf, left, y, f"Грузополучатель: {order.consignee.name}"[:140])
        y = _text(pdf, left, y, f"Продавец: {order.seller.name}"[:140])
        terms = (order.shipping_terms or "").strip()
        if terms:
            y = _text(pdf, left, y, "Условия отгрузки:", 11)
            for part in terms.splitlines():
                for line in _wrap(part, 10, right - left):
                    y = _text(pdf, left, y, line, 10)
    elif purchase is not None:
        supplier = task.supplier.name if task.supplier_id else purchase.supplier.name
        y = _text(pdf, left, y, f"Заказ поставщику {purchase.number or ''}".strip())
        y = _text(pdf, left, y, f"Поставщик: {supplier}"[:140])
        if task.ttn_number:
            y = _text(pdf, left, y, f"ТТН {task.ttn_number}")
        if task.doc_date:
            y = _text(pdf, left, y, f"Дата прихода: {task.doc_date.strftime('%d.%m.%Y')}")
    y -= 2 * mm
    return y


def _row(pdf, y, left, cells, widths, size: int, numeric: set[int]) -> float:
    pad = 1.6 * mm
    line_h = size + 2
    wrapped = []
    for index, (value, width) in enumerate(zip(cells, widths)):
        wrapped.append(_wrap(str(value), size, width - 2 * pad))
    height = max(len(item) for item in wrapped) * line_h + 2 * pad
    x = left
    pdf.setFont("WmsFont", size)
    for index, (lines, width) in enumerate(zip(wrapped, widths)):
        pdf.rect(x, y - height, width, height, stroke=1, fill=0)
        cursor = y - pad - size
        for line in lines:
            if index in numeric:
                pdf.drawRightString(x + width - pad, cursor, line)
            else:
                pdf.drawString(x + pad, cursor, line)
            cursor -= line_h
        x += width
    return y - height


def _task_lines(pdf, task, y, left, right, height, columns, *, by_groups: bool = False) -> None:
    widths = [item[1] for item in columns]
    titles = [item[0] for item in columns]
    bottom = 18 * mm
    table_width = sum(widths)

    def header(top: float) -> float:
        return _row(pdf, top, left, titles, widths, 9, {4, 5})

    def ensure(top: float, need: float) -> float:
        if top - need >= bottom:
            return top
        pdf.showPage()
        top = height - 14 * mm
        top = _text(pdf, left, top, f"{task.number or task.id}, продолжение", 12)
        top -= 2 * mm
        return header(top)

    if y < bottom + 20 * mm:
        pdf.showPage()
        y = height - 14 * mm
        y = _text(pdf, left, y, f"{task.number or task.id}, продолжение", 12)
        y -= 2 * mm
    y = header(y)
    lines = list(task.lines.all())
    sections = _lines_by_product_group(lines) if by_groups else [("", lines)]
    total = 0
    number = 0
    for group_title, group_lines in sections:
        if by_groups and group_title:
            y = ensure(y, 10 * mm)
            y -= 2 * mm
            pdf.setFont("WmsFont", 10)
            pdf.drawString(left + 1.6 * mm, y - 8, group_title[:90])
            pdf.rect(left, y - 10, table_width, 10, stroke=1, fill=0)
            y -= 10
        for line in group_lines:
            number += 1
            cells = [
                str(number),
                line.product.name,
                _line_caliber(line),
                _line_grade(line),
                format_kg(line.qty_kg),
                "",
            ]
            y = ensure(y, 8 * mm)
            y = _row(pdf, y, left, cells, widths, 9, {0, 4, 5})
            total += line.qty_kg
    if not lines:
        y = _text(pdf, left, y - 4 * mm, "Строк нет")
    else:
        y -= 8 * mm
        y = _text(pdf, left, y, f"Итого план {format_kg(total)} кг", 11)
    if y < bottom + 16 * mm:
        pdf.showPage()
        y = height - 14 * mm
    y -= 8 * mm
    _text(pdf, left, y, "Отобрал ____________________", 11)
    _text(pdf, left + 90 * mm, y, "Дата ______________", 11)

def routes_sheets_pdf(sheets: list) -> bytes:
    """Несколько маршрутных листов в одном PDF (каждый маршрут с новой страницы)."""
    from django.utils import timezone

    if not sheets:
        raise ValueError("Нет маршрутов для печати")

    ensure_font()
    buffer = BytesIO()
    page = landscape(A4)
    pdf = canvas.Canvas(buffer, pagesize=page)
    width, height = page
    left = 12 * mm
    columns = (
        ("№", 12 * mm),
        ("Заказ", 36 * mm),
        ("Грузополучатель", 70 * mm),
        ("Адрес", 90 * mm),
        ("Зона", 36 * mm),
        ("Кг", 22 * mm),
    )
    widths = [item[1] for item in columns]
    bottom = 18 * mm

    for index, (route, stops, weights) in enumerate(sheets):
        if index:
            pdf.showPage()
        y = height - 14 * mm
        title = route.name or f"Маршрут {route.id}"
        pdf.setFont("WmsFont", 16)
        pdf.drawString(left, y, f"Маршрутный лист · {title}"[:90])
        y -= 8 * mm
        y = _text(pdf, left, y, f"Дата: {route.route_date.strftime('%d.%m.%Y')}")
        driver_name = route.driver.full_name if route.driver_id else "—"
        y = _text(pdf, left, y, f"Водитель: {driver_name}"[:140])
        y = _text(pdf, left, y, f"Автомобиль: {route.vehicle_text or '—'}"[:140])
        if route.planned_load_at:
            load_at = timezone.localtime(route.planned_load_at)
            if load_at.date() != route.route_date:
                y = _text(pdf, left, y, f"Погрузка: {load_at.strftime('%d.%m.%Y %H:%M')}")
            else:
                y = _text(pdf, left, y, f"Погрузка: {load_at.strftime('%H:%M')}")
        else:
            y = _text(pdf, left, y, "Погрузка: не назначена")
        y = _text(pdf, left, y, f"Статус: {route.get_status_display()}")
        notes = (route.notes or "").strip()
        if notes:
            y = _text(pdf, left, y, "Комментарий:", 11)
            for part in notes.splitlines() or [notes]:
                for line in _wrap(part, 10, width - 24 * mm):
                    y = _text(pdf, left, y, line, 10)
        y -= 2 * mm

        def header(top: float) -> float:
            return _row(pdf, top, left, [item[0] for item in columns], widths, 9, {0, 5})

        def ensure(top: float, need: float) -> float:
            if top - need >= bottom:
                return top
            pdf.showPage()
            top = height - 14 * mm
            top = _text(pdf, left, top, f"{title}, продолжение", 12)
            top -= 2 * mm
            return header(top)

        y = header(y)
        total = 0
        for stop in stops:
            order = stop.order
            consignee = order.consignee
            zone = ""
            address = ""
            consignee_name = ""
            if consignee is not None:
                consignee_name = consignee.name
                address = consignee.address or ""
                zone = consignee.zone.name if consignee.zone_id else "Без зоны"
            else:
                zone = "Без зоны"
            weight = weights.get(order.id) or 0
            total += weight
            cells = [
                str(stop.position),
                order.number or str(order.id),
                consignee_name,
                address,
                zone,
                format_kg(weight),
            ]
            y = ensure(y, 8 * mm)
            y = _row(pdf, y, left, cells, widths, 8, {0, 5})
        y -= 8 * mm
        if not stops:
            y = _text(pdf, left, y, "Остановок нет")
        else:
            y = _text(pdf, left, y, f"Итого {format_kg(total)} кг · остановок {len(stops)}", 11)
        y -= 6 * mm
        _text(pdf, left, y, "Водитель ____________________", 11)
        _text(pdf, left + 90 * mm, y, "Склад ______________", 11)

    pdf.save()
    return buffer.getvalue()


def schedule_sheet_pdf(tree: dict) -> bytes:

        f"Итого {format_kg(tree.get('total_weight', 0))} кг · ТТ {tree.get('total_tt', 0)}",
        10,

    y -= 2 * mm

        y = _text(pdf, left, y, "Нет маршрутов с назначенным планом погрузки")

        y = header(y)
        for row in rows:
            cells = [
                row.get("plan") or "",
                row.get("fact") or "",
                row.get("label") or "",
                row.get("route") or "",
                row.get("notes") or "",

            y = _row(pdf, y, left, cells, widths, 9, {5, 6})
        y -= 4 * mm
        y = _text(
            pdf,
            left,
            y,
            f"Итого {format_kg(tree.get('total_weight', 0))} кг · ТТ {tree.get('total_tt', 0)}",
            11,
        )

    pdf.save()
    return buffer.getvalue()

def apply_printer(item: Printer, form) -> None:
    from apps.common.forms import text
    from apps.printing.installed import list_installed

    name = text(form, "name")
    if not name:
        raise PrintError("Укажите название")
    queue = text(form, "queue_name")
    installed = {row["name"] for row in list_installed()}
    if queue not in installed:
        raise PrintError("Выберите принтер, установленный на сервере")
    item.name = name
    item.printer_type = "tsc" if "tsc" in queue.lower() else "network"
    item.port = 9100
    item.queue_name = queue
    item.host = ""
    item.is_default = form.get("is_default") == "1"
    item.save()
    if item.is_default:
        Printer.objects.filter(printer_type=item.printer_type).exclude(pk=item.pk).update(is_default=False)


def default_printer(printer_type: str) -> Printer:
    printer = Printer.objects.filter(printer_type=printer_type, is_default=True).first()
    if printer is None:
        printer = Printer.objects.filter(printer_type=printer_type).first()
    if printer is None:
        if printer_type == "tsc":
            raise PrintError("Не задан термопринтер TSC")
        raise PrintError("Не задан сетевой принтер")
    return printer


def send_raw(host: str, port: int, payload: bytes) -> None:
    if not host:
        raise PrintError("У принтера не задан адрес")
    try:
        with socket.create_connection((host, int(port or 9100)), timeout=5) as sock:
            sock.sendall(payload)
    except OSError as exc:
        raise PrintError(f"Принтер недоступен: {exc}") from exc


def send_tspl(printer: Printer, payload: bytes) -> str:
    if printer.queue_name:
        from apps.printing.installed import print_target

        target = print_target(printer.queue_name)
        _print_raw_queue(target, payload, f"tsc-{printer.id}")
        return f"Этикетка отправлена в очередь «{printer.queue_name}»"
    send_raw(printer.host, printer.port or 9100, payload)
    return f"Этикетка отправлена на TSC «{printer.name}»"


class _DocInfo(ctypes.Structure):
    _fields_ = [
        ("pDocName", wintypes.LPWSTR),
        ("pOutputFile", wintypes.LPWSTR),
        ("pDatatype", wintypes.LPWSTR),
    ]


def _print_raw_queue(queue: str, payload: bytes, job_name: str) -> None:
    if os.name == "nt":
        _print_raw_windows(queue, payload, job_name)
        return
    _print_raw_cups(queue, payload, job_name)


def _print_raw_windows(queue: str, payload: bytes, job_name: str) -> None:
    winspool = ctypes.WinDLL("winspool.drv", use_last_error=True)
    winspool.OpenPrinterW.argtypes = [wintypes.LPWSTR, ctypes.POINTER(wintypes.HANDLE), wintypes.LPVOID]
    winspool.OpenPrinterW.restype = wintypes.BOOL
    winspool.ClosePrinter.argtypes = [wintypes.HANDLE]
    winspool.ClosePrinter.restype = wintypes.BOOL
    winspool.StartDocPrinterW.argtypes = [wintypes.HANDLE, wintypes.DWORD, ctypes.POINTER(_DocInfo)]
    winspool.StartDocPrinterW.restype = wintypes.DWORD
    winspool.EndDocPrinter.argtypes = [wintypes.HANDLE]
    winspool.EndDocPrinter.restype = wintypes.BOOL
    winspool.StartPagePrinter.argtypes = [wintypes.HANDLE]
    winspool.StartPagePrinter.restype = wintypes.BOOL
    winspool.EndPagePrinter.argtypes = [wintypes.HANDLE]
    winspool.EndPagePrinter.restype = wintypes.BOOL
    winspool.WritePrinter.argtypes = [wintypes.HANDLE, wintypes.LPVOID, wintypes.DWORD, ctypes.POINTER(wintypes.DWORD)]
    winspool.WritePrinter.restype = wintypes.BOOL

    handle = wintypes.HANDLE()
    if not winspool.OpenPrinterW(queue, ctypes.byref(handle), None):
        raise PrintError(f"Принтер «{queue}» не найден среди принтеров Windows")
    try:
        info = _DocInfo(job_name, None, "RAW")
        if not winspool.StartDocPrinterW(handle, 1, ctypes.byref(info)):
            raise PrintError(_windows_print_error(queue))
        try:
            if not winspool.StartPagePrinter(handle):
                raise PrintError(_windows_print_error(queue))
            written = wintypes.DWORD()
            buffer = ctypes.create_string_buffer(payload)
            if not winspool.WritePrinter(handle, buffer, len(payload), ctypes.byref(written)):
                raise PrintError(_windows_print_error(queue))
            if written.value != len(payload):
                raise PrintError(f"В очередь «{queue}» ушло {written.value} байт из {len(payload)}")
            if not winspool.EndPagePrinter(handle):
                raise PrintError(_windows_print_error(queue))
        finally:
            winspool.EndDocPrinter(handle)
    finally:
        winspool.ClosePrinter(handle)


def _windows_print_error(queue: str) -> str:
    error = ctypes.WinError(ctypes.get_last_error())
    detail = str(error).strip()
    if detail:
        return f"Очередь «{queue}» не приняла этикетку: {detail}"
    return f"Очередь «{queue}» не приняла этикетку"

def _print_windows_queue(queue: str, payload: bytes, job_name: str) -> str:
    folder = Path(tempfile.gettempdir()) / "wms-print"
    folder.mkdir(exist_ok=True)
    path = folder / f"{job_name}.pdf"
    path.write_bytes(payload)
    script = Path(__file__).with_name("print_queue.ps1")
    env = os.environ.copy()
    env["WMS_PDF"] = str(path)
    env["WMS_PRINTER"] = queue
    env["WMS_JOB"] = job_name
    try:
        result = subprocess.run(
            ["powershell", "-NoProfile", "-ExecutionPolicy", "Bypass", "-File", str(script)],
            env=env,
            capture_output=True,
            timeout=90,
        )
    except subprocess.TimeoutExpired as exc:
        raise PrintError("Печать Windows не завершилась. Проверьте, что принтер не ждёт подтверждения") from exc
    except OSError as exc:
        raise PrintError(f"Не удалось вызвать печать Windows: {exc}") from exc
    if result.returncode != 0:
        detail = _ps_text(result.stderr) or _ps_text(result.stdout)
        raise PrintError(_print_failure(queue, detail))
    return f"PDF отправлен в очередь «{queue}»"

def _ps_text(data: bytes | None) -> str:
    if not data:
        return ""
    for encoding in ("utf-8-sig", "cp866", "cp1251"):
        try:
            return data.decode(encoding).strip()
        except UnicodeDecodeError:
            continue
    return data.decode("utf-8", errors="replace").strip()

    if detail == "PDF_EMPTY":
        return "PDF не содержит страниц"
    if detail:

def _parts(total: Decimal, count: int) -> list[Decimal]:
    total = as_kg(total)
    each = (total / Decimal(count)).quantize(Decimal("0.001"))
    parts = [each] * (count - 1)
    last = as_kg(total - each * (count - 1))
    parts.append(last)
    if any(part <= 0 for part in parts):
        raise WarehouseError("Слишком много упаковок для этого веса")
    return parts

    order = customer_line.order

def expected_net(gross: Decimal, places: int, pallet_type: str, package_tare) -> Decimal:
    pallet = as_kg(PALLET_TARE[pallet_type])
    packages = as_kg(package_tare) * places
    return as_kg(gross - pallet - packages)


def _resolve_weight(places: int, pallet_type: str, gross_kg, net_kg, package_tare, manual: bool):
    if places < 1 or places > 500:
        raise WarehouseError("Укажите количество упаковок")
    if pallet_type not in PALLET_TARE:
        raise WarehouseError("Укажите вид паллета")
    gross = as_kg(gross_kg)
    if gross <= 0:
        raise WarehouseError("Укажите грязный вес")
    net = as_kg(net_kg) if manual else expected_net(gross, places, pallet_type, package_tare)
    if net <= 0 or net > gross:
        raise WarehouseError("Чистый вес должен быть больше нуля и не больше грязного")
    return gross, net

        return True
    return bool(consignee and consignee.category_id and consignee.category_id in allowed)

        verbose_name = "Зона склада"
        verbose_name_plural = "Зоны склада"
        ordering = ["code"]

    def __str__(self) -> str:
        return f"{self.code} · {self.name}"


def cell_address(zone_code: str, code: str) -> str:
    zone = (zone_code or "").strip()
    place = (code or "").strip()
    if not zone:
        return place[:120]
    prefix = f"{zone}-"
    if place.casefold().startswith(prefix.casefold()):
        return place[:120]
    return f"{zone}-{place}"[:120]

        db_table = "warehouse_cells"

        verbose_name_plural = "Ячейки"
        ordering = ["zone__code", "code"]
        constraints = [
            models.UniqueConstraint(fields=["zone", "code"], name="warehouse_cell_zone_code"),
        ]

    def label(self) -> str:
        return cell_address(self.zone.code, self.code)

        return self.label()
