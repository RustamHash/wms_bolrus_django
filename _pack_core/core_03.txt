from decimal import Decimal

from apps.common.exceptions import WarehouseError

def _lock_balance(lot_id: int) -> Balance:
    row = Balance.objects.select_for_update().filter(lot_id=lot_id).first()
    if row is None:
        row = Balance.objects.create(
            lot_id=lot_id,
            free_kg=Decimal("0.000"),
            reserved_kg=Decimal("0.000"),
            picked_kg=Decimal("0.000"),
        )
    return row


def inbound_is_reserved(movement) -> bool:
    pallet = movement.pallet
    return bool(
        pallet.reserved_line_id
        or pallet.reserved_consignee_id
        or (pallet.kind == "inbound" and pallet.order_id)
    )


def reserve_free_inbound(movement) -> None:
    """Переносит вес принятой паллеты из свободного остатка в резерв. Паллета уже отмечена резервом."""
    if movement.kind != "in":
        raise WarehouseError("Резерв ставится на принятую паллету")
    if has_left(movement):
        raise WarehouseError("Паллета уже отобрана, отгружена или списана в переработку")
    if not inbound_is_reserved(movement):
        raise WarehouseError("Паллета не отмечена резервом")
    qty = as_kg(movement.qty_kg)
    _apply(_lock_balance(movement.lot_id), -qty, Decimal("0"), qty)


def _apply(row: Balance, free_delta: Decimal, picked_delta: Decimal, reserved_delta: Decimal = Decimal("0")) -> None:
    row.free_kg = as_kg(row.free_kg) + as_kg(free_delta)
    row.reserved_kg = as_kg(row.reserved_kg) + as_kg(reserved_delta)
    row.picked_kg = as_kg(row.picked_kg) + as_kg(picked_delta)
    if row.free_kg < 0 or row.reserved_kg < 0 or row.picked_kg < 0:
        raise WarehouseError("Недостаточно остатка")
    row.save(update_fields=["free_kg", "reserved_kg", "picked_kg"])


LEFT_KINDS = (
    Movement.Kind.PICK,
    Movement.Kind.OUT,
    Movement.Kind.TRANSFORM_OUT,
    Movement.Kind.WASTE,
)
TAKEN_KINDS = (
    Movement.Kind.PICK,
    Movement.Kind.TRANSFORM_OUT,
    Movement.Kind.WASTE,
)


def _kg_by_pallet(kinds, pallet_ids=None) -> dict[int, Decimal]:
    query = Movement.objects.filter(kind__in=kinds)
    if pallet_ids is not None:
        query = query.filter(pallet_id__in=list(pallet_ids))
    rows = query.values("pallet_id").annotate(total=Sum("qty_kg"))
    return {row["pallet_id"]: as_kg(row["total"] or 0) for row in rows}


def inbound_kg_by_pallet(pallet_ids=None) -> dict[int, Decimal]:
    return _kg_by_pallet((Movement.Kind.IN,), pallet_ids)


def taken_kg_by_pallet(pallet_ids=None) -> dict[int, Decimal]:
    """Вес, который уже нельзя отобрать: отбор, переработка и отход."""
    return _kg_by_pallet(TAKEN_KINDS, pallet_ids)


def shipped_kg_by_pallet(pallet_ids=None) -> dict[int, Decimal]:
    return _kg_by_pallet((Movement.Kind.OUT,), pallet_ids)


def open_plan_kg_by_pallet(pallet_ids=None, exclude_task_id: int | None = None) -> dict[int, Decimal]:
    """Неотобранный план назначений. Отбор этого задания при exclude_task_id не вычитается."""
    from apps.picking.models import PickAllocation

    query = PickAllocation.objects.all()
    if pallet_ids is not None:
        query = query.filter(pallet_id__in=list(pallet_ids))
    if exclude_task_id is not None:
        query = query.exclude(task_line__task_id=exclude_task_id)
    allocs = list(query.values("pallet_id", "task_line_id", "qty_kg"))
    if not allocs:
        return {}
    line_ids = {row["task_line_id"] for row in allocs}
    pallets = {row["pallet_id"] for row in allocs}
    picks = (
        Movement.objects.filter(kind=Movement.Kind.PICK, task_line_id__in=line_ids, pallet_id__in=pallets)
        .values("task_line_id", "pallet_id")
        .annotate(total=Sum("qty_kg"))
    )
    picked = {(row["task_line_id"], row["pallet_id"]): as_kg(row["total"] or 0) for row in picks}
    found: dict[int, Decimal] = {}
    for row in allocs:
        left = as_kg(row["qty_kg"]) - picked.get((row["task_line_id"], row["pallet_id"]), Decimal("0.000"))
        if left <= 0:
            continue
        found[row["pallet_id"]] = found.get(row["pallet_id"], Decimal("0.000")) + left
    return found


def available_kg_by_pallet(pallet_ids=None) -> dict[int, Decimal]:
    """Приход минус отбор, переработка и ещё не отобранный план."""
    inbound = inbound_kg_by_pallet(pallet_ids)
    taken = taken_kg_by_pallet(list(inbound) or pallet_ids)
    planned = open_plan_kg_by_pallet(list(inbound) or pallet_ids)
    found = {}
    for pallet_id, qty in inbound.items():
        left = as_kg(qty - taken.get(pallet_id, 0) - planned.get(pallet_id, 0))
        found[pallet_id] = left if left > 0 else Decimal("0.000")
    return found


def pallet_room(pallet_id: int, *, exclude_task_id: int | None = None) -> Decimal:
    """Сколько ещё можно отобрать, не занимая план других заданий."""
    inbound = inbound_kg_by_pallet([pallet_id]).get(pallet_id, Decimal("0.000"))
    taken = taken_kg_by_pallet([pallet_id]).get(pallet_id, Decimal("0.000"))
    others = open_plan_kg_by_pallet([pallet_id], exclude_task_id=exclude_task_id).get(pallet_id, Decimal("0.000"))
    left = as_kg(inbound - taken - others)
    return left if left > 0 else Decimal("0.000")


def unpicked_kg(pallet_id: int) -> Decimal:
    inbound = inbound_kg_by_pallet([pallet_id]).get(pallet_id, Decimal("0.000"))
    taken = taken_kg_by_pallet([pallet_id]).get(pallet_id, Decimal("0.000"))
    left = as_kg(inbound - taken)
    return left if left > 0 else Decimal("0.000")


def physical_left(pallet_id: int) -> Decimal:
    """Вес, который ещё физически на паллете: приход минус отгрузка и переработка."""
    inbound = inbound_kg_by_pallet([pallet_id]).get(pallet_id, Decimal("0.000"))
    gone = _kg_by_pallet((Movement.Kind.OUT, Movement.Kind.TRANSFORM_OUT, Movement.Kind.WASTE), [pallet_id])
    left = as_kg(inbound - gone.get(pallet_id, 0))
    return left if left > 0 else Decimal("0.000")


def consumed_pallet_ids():
    """Паллеты без доступного веса и паллеты, ушедшие в переработку."""
    available = available_kg_by_pallet()
    gone = {pallet_id for pallet_id, qty in available.items() if qty <= 0}
    gone.update(reworked_pallet_ids())
    return list(gone)


def reworked_pallet_ids():
    return Movement.objects.filter(kind__in=(Movement.Kind.TRANSFORM_OUT, Movement.Kind.WASTE)).values_list(
        "pallet_id", flat=True
    )


def has_left(movement) -> bool:
    """По паллете уже есть отбор, отгрузка или переработка."""
    return Movement.objects.filter(pallet_id=movement.pallet_id, kind__in=LEFT_KINDS).exists()


def has_downstream(movement) -> bool:
    """Паллета целиком ушла из доступного остатка."""
    if unpicked_kg(movement.pallet_id) > 0:
        return False
    return Movement.objects.filter(pallet_id=movement.pallet_id, kind__in=LEFT_KINDS).exists()


def downstream_pallet_ids(pallet_ids) -> set[int]:
    ids = [item for item in pallet_ids if item]
    if not ids:
        return set()
    return set(Movement.objects.filter(pallet_id__in=ids, kind__in=LEFT_KINDS).values_list("pallet_id", flat=True))

    rows = (
        Movement.objects.filter(kind=kind)
        .exclude(**{f"{lookup}__isnull": True})

    return {row[lookup]: as_kg(row["total"]) for row in rows}


@transaction.atomic
def post_in(*, task, task_line, pallet, lot, qty_kg: Decimal, user, reserved: bool = False) -> Movement:
    qty = as_kg(qty_kg)
    if qty <= 0:
        raise WarehouseError("Вес паллеты должен быть больше нуля")
    movement = Movement.objects.create(
        task=task,
        task_line=task_line,
        pallet=pallet,
        kind="in",
        lot=lot,
        qty_kg=qty,
        created_by=user,
    )
    if reserved:
        _apply(_lock_balance(lot.id), Decimal("0"), Decimal("0"), qty)
    else:
        _apply(_lock_balance(lot.id), qty, Decimal("0"))
    return movement


@transaction.atomic
def update_in(movement, *, lot, qty_kg: Decimal) -> Movement:
    if movement.kind != "in":
        raise WarehouseError("Правится только приход")
    if has_left(movement):
        raise WarehouseError("По паллете уже есть отбор, отгрузка или переработка")
    qty = as_kg(qty_kg)
    if qty <= 0:
        raise WarehouseError("Вес паллеты должен быть больше нуля")
    reserved = inbound_is_reserved(movement)
    old = as_kg(movement.qty_kg)
    if reserved:
        _apply(_lock_balance(movement.lot_id), Decimal("0"), Decimal("0"), -old)
    else:
        _apply(_lock_balance(movement.lot_id), -old, Decimal("0"))
    movement.lot = lot
    movement.qty_kg = qty
    movement.save(update_fields=["lot", "qty_kg"])
    if reserved:
        _apply(_lock_balance(lot.id), Decimal("0"), Decimal("0"), qty)
    else:
        _apply(_lock_balance(lot.id), qty, Decimal("0"))
    return movement


@transaction.atomic
def remove_in(movement) -> None:
    if movement.kind != "in":
        raise WarehouseError("Удаляется только приход")
    if has_left(movement):
        raise WarehouseError("По паллете уже есть отбор, отгрузка или переработка")
    qty = as_kg(movement.qty_kg)
    if inbound_is_reserved(movement):
        _apply(_lock_balance(movement.lot_id), Decimal("0"), Decimal("0"), -qty)
    else:
        _apply(_lock_balance(movement.lot_id), -qty, Decimal("0"))
    movement.delete()


@transaction.atomic
def post_pick(*, task, task_line, pallet, source: Movement, user, qty_kg: Decimal) -> Movement:
    if source.kind != "in" or source.pallet_id != pallet.id:
        raise WarehouseError("Отбирать можно только принятую паллету")
    qty = as_kg(qty_kg)
    if qty <= 0:
        raise WarehouseError("Вес отбора должен быть больше нуля")
    room = pallet_room(pallet.id, exclude_task_id=task.id)
    if qty > room:
        raise WarehouseError(f"На паллете недостаточно веса. Доступно {room} кг")
    movement = Movement.objects.create(
        task=task,
        task_line=task_line,
        pallet=pallet,
        kind="pick",
        lot_id=source.lot_id,
        qty_kg=qty,
        source=source,
        created_by=user,
    )
    # Отобранный вес уходит из остатка: из резерва, если паллета была под заказ, иначе из свободного.
    if inbound_is_reserved(source):
        _apply(_lock_balance(source.lot_id), Decimal("0"), Decimal("0"), -qty)
    else:
        _apply(_lock_balance(source.lot_id), -qty, Decimal("0"))
    if unpicked_kg(pallet.id) <= 0 and (pallet.reserved_line_id or pallet.reserved_consignee_id or pallet.order_id):
        pallet.reserved_line = None
        pallet.reserved_consignee = None
        pallet.order = None
        pallet.save(update_fields=["reserved_line", "reserved_consignee", "order"])
    return movement


@transaction.atomic
def post_out(*, task, task_line, pallet, source: Movement, user) -> Movement:
    if source.kind != "pick" or source.pallet_id != pallet.id:
        raise WarehouseError("Отгружается только отобранный вес")
    if Movement.objects.filter(source=source, kind="out").exists():
        raise WarehouseError("Этот отбор уже отгружен")
    qty = as_kg(source.qty_kg)
    movement = Movement.objects.create(
        task=task,
        task_line=task_line,
        pallet=pallet,
        kind="out",
        lot_id=source.lot_id,
        qty_kg=qty,
        source=source,
        created_by=user,
    )
    return movement


def lock_balances(lot_ids) -> None:
    for lot_id in sorted(set(lot_ids)):
        _lock_balance(lot_id)


def _release_kg(lot_id: int, qty: Decimal, reserved: bool) -> None:
    qty = as_kg(qty)
    if qty <= 0:
        raise WarehouseError("Вес списания должен быть больше нуля")
    if reserved:
        _apply(_lock_balance(lot_id), Decimal("0"), Decimal("0"), -qty)
    else:
        _apply(_lock_balance(lot_id), -qty, Decimal("0"))

def lot_reserve_rows(lot_id: int) -> list[dict]:
    """Строки заказов, под которые ещё стоит резерв этой партии."""
    from apps.common.present import describe_line

    movements = (
        Movement.objects.filter(kind="in", lot_id=lot_id)
        .select_related(
            "pallet__reserved_line__order__consignee",
            "pallet__reserved_line__product",
            "pallet__reserved_line__caliber",
            "pallet__reserved_line__caliber_from",
            "pallet__reserved_line__caliber_to",
            "pallet__order__consignee",
            "pallet__reserved_consignee",
        )
        .prefetch_related("pallet__reserved_line__grades")
        .order_by("id")
    )
    pallet_ids = [item.pallet_id for item in movements]
    inbound = inbound_kg_by_pallet(pallet_ids)
    taken = taken_kg_by_pallet(pallet_ids)
    groups: dict[tuple, dict] = {}
    seen: set[int] = set()
    for movement in movements:
        if movement.pallet_id in seen:
            continue
        seen.add(movement.pallet_id)
        pallet = movement.pallet
        if not (pallet.reserved_line_id or pallet.reserved_consignee_id or pallet.order_id):
            continue
        left = as_kg(inbound.get(pallet.id, 0) - taken.get(pallet.id, 0))
        if left <= 0:
            continue
        if pallet.reserved_line_id:
            key = ("line", pallet.reserved_line_id)
        elif pallet.order_id:
            key = ("order", pallet.order_id)
        else:
            key = ("consignee", pallet.reserved_consignee_id)
        bucket = groups.get(key)
        if bucket is None:
            bucket = _reserve_bucket(pallet, describe_line)
            groups[key] = bucket
        bucket["kg"] = as_kg(bucket["kg"] + left)
        bucket["pallets"].append({"id": pallet.id, "number": pallet.number or f"#{pallet.id}"})
    return sorted(groups.values(), key=lambda item: (item["order_number"], item["line_id"] or 0))


def _reserve_bucket(pallet, describe_line) -> dict:
    line = pallet.reserved_line
    order = line.order if line is not None and line.order_id else pallet.order
    consignee = ""
    if line is not None and line.order_id and line.order.consignee_id:
        consignee = line.order.consignee.name
    elif order is not None and order.consignee_id:
        consignee = order.consignee.name
    elif pallet.reserved_consignee_id:
        consignee = pallet.reserved_consignee.name
    return {
        "line_id": line.id if line is not None else None,
        "order_id": order.id if order is not None else None,
        "order_number": (order.number if order is not None else "") or "",
        "ship_date": order.ship_date if order is not None else None,
        "consignee": consignee,
        "title": describe_line(line) if line is not None else "Без строки заказа",
        "qty_kg": as_kg(line.qty_kg) if line is not None else None,
        "kg": Decimal("0.000"),
        "pallets": [],
    }


def lot_journal(lot_id: int) -> list[dict]:
    rows = (
        Movement.objects.filter(lot_id=lot_id)
        .select_related("pallet", "task", "transform_document", "created_by")
        .order_by("-id")
    )
    found = []
    for item in rows:
        document = ""
        href = ""
        if item.task_id:
            document = item.task.number or f"#{item.task_id}"
            href = f"/tasks/{item.task_id}"
        elif item.transform_document_id:
            document = item.transform_document.number or f"#{item.transform_document_id}"
            href = f"/transform/{item.transform_document_id}"
        found.append(
            {
                "created_at": item.created_at,
                "kind": item.get_kind_display(),
                "pallet_id": item.pallet_id,
                "pallet_number": item.pallet.number or f"#{item.pallet_id}",
                "qty_kg": as_kg(item.qty_kg),
                "document": document,
                "href": href,
                "user": item.created_by.name if item.created_by_id else "",
            }
        )
    return found


def stock_rows() -> list[dict]:
    rows = Balance.objects.select_related(
        "lot__product", "lot__caliber", "lot__grade", "lot__supplier", "lot__receipt"
    ).order_by("lot__received_on", "id")
    result = []
    for row in rows:
        if as_kg(row.free_kg) == 0 and as_kg(row.reserved_kg) == 0 and as_kg(row.picked_kg) == 0:
            continue
        lot = row.lot
        source = lot.get_source_display()
        if lot.is_opening:
            source = f"{source} · начальный"
        result.append(
            {
                "lot_id": lot.id,
                "number": lot.number or "",
                "product": lot.product.name,
                "caliber": lot.caliber.name,
                "grade": lot.grade.name,
                "supplier": lot.supplier.name,
                "source": source,
                "receipt": lot.receipt.number or "",
                "free_kg": as_kg(row.free_kg),
                "reserved_kg": as_kg(row.reserved_kg),
                "picked_kg": as_kg(row.picked_kg),
            }
        )
    return result

def parse_tolerance(raw: str) -> Decimal:
    value_text = (raw or "").strip().replace(",", ".")
    if not value_text:
        return Decimal("0.000")
    try:
        value = Decimal(value_text)
    except InvalidOperation as exc:
        raise WarehouseError("Некорректный допуск") from exc
    value = value.quantize(Decimal("0.001"))
    if value < 0:
        raise WarehouseError("Допуск не может быть отрицательным")
    return value


def _totals(document: TransformDocument) -> tuple[Decimal, Decimal, Decimal]:
    incoming = document.inputs.aggregate(total=Sum("qty_kg"))["total"]
    outgoing = document.outputs.aggregate(total=Sum("qty_kg"))["total"]
    waste = document.wastes.aggregate(total=Sum("qty_kg"))["total"]
    return as_kg(incoming or 0), as_kg(outgoing or 0), as_kg(waste or 0)


def mass_state(document: TransformDocument) -> dict:
    incoming, outgoing, waste = _totals(document)
    gap = as_kg(incoming - outgoing - waste)
    tolerance = as_kg(document.tolerance_kg)
    return {
        "input_kg": incoming,
        "output_kg": outgoing,
        "waste_kg": waste,
        "gap_kg": gap,
        "tolerance_kg": tolerance,
        "fits": abs(gap) <= tolerance and incoming > 0,
    }


def _draft(document: TransformDocument) -> TransformDocument:
    document = TransformDocument.objects.select_for_update().get(pk=document.pk)
    if document.status != TransformDocument.Status.DRAFT:
        raise WarehouseError("Менять можно только черновик")
    return document


def _stock_inbound(pallet_id: int):
    inbound = (
        Movement.objects.select_for_update()
        .select_related("lot", "pallet")

def suggest_categories(document: TransformDocument) -> set[int] | None:
    sets = []
    inputs = document.inputs.select_related("pallet").prefetch_related("pallet__allowed_categories")
    for line in inputs:
        found = {item.id for item in line.pallet.allowed_categories.all()}
        if found:
            sets.append(found)
    if not sets:
        return set()
    shared = set.intersection(*sets)
    if not shared:
        return None
    return shared


def output_category_ids(document: TransformDocument) -> list[int]:
    if document.categories_set:
        return list(document.allowed_categories.values_list("id", flat=True))
    suggested = suggest_categories(document)
    if suggested is None:
        raise WarehouseError("У входных паллет нет общей категории. Отметьте категории выхода")
    return sorted(suggested)

def stock_choices(document: TransformDocument) -> list[dict]:
    used = set(document.inputs.values_list("pallet_id", flat=True))

        .select_related("pallet", "lot__product", "lot__caliber", "lot__grade")
        .order_by("lot__product__name", "pallet__number", "id")
    )
    choices = []
    for movement in rows:
        if movement.pallet_id in used:
            continue
        pallet = movement.pallet

        )
        if pallet.order_id or pallet.reserved_consignee_id or pallet.reserved_line_id:

def _parent_lot(inbounds: list) -> Lot:
    lots = [inbound.lot for _line, inbound in inbounds]
    return min(lots, key=lambda lot: (lot.received_on, lot.id))


def _split(inbounds: list, output_kg: Decimal) -> list[tuple]:
    remaining = as_kg(output_kg)
    rows = []
    for line, inbound in inbounds:
        qty = as_kg(inbound.qty_kg)
        out_part = as_kg(min(qty, remaining))
        remaining = as_kg(remaining - out_part)
        rows.append((line, inbound, out_part, as_kg(qty - out_part)))
    return rows


def _clear_pallet(pallet: Pallet) -> None:
    unit_ids = list(pallet.units.values_list("id", flat=True))
    if unit_ids:
        Allocation.objects.filter(unit_id__in=unit_ids).delete()
    pallet.units.exclude(status="rejected").update(status="transformed")
    if pallet.order_id or pallet.reserved_consignee_id or pallet.reserved_line_id:
        pallet.order = None
        pallet.reserved_consignee = None
        pallet.reserved_line = None

@transaction.atomic
def post_document(document: TransformDocument, user) -> str:
    document = (
        TransformDocument.objects.select_for_update(of=("self",))
        .select_related("output_line__order__consignee", "output_consignee")
        .get(pk=document.pk)
    )
    if document.status != TransformDocument.Status.DRAFT:
        raise WarehouseError("Провести можно только черновик")
    if document.kind not in (Lot.Source.REWORK, Lot.Source.WORKSHOP):
        raise WarehouseError("Укажите переборку или цех")
    inputs = list(document.inputs.select_related("pallet", "lot").order_by("id"))
    outputs = list(document.outputs.select_related("product", "caliber", "grade").order_by("id"))
    if not inputs:
        raise WarehouseError("Добавьте паллету на вход")
    locked = []
    for line in inputs:
        inbound = _stock_inbound(line.pallet_id)
        label = line.pallet.number or f"#{line.pallet_id}"
        if as_kg(inbound.qty_kg) != as_kg(line.qty_kg) or inbound.lot_id != line.lot_id:
            raise WarehouseError(f"Паллета {label} изменилась. Уберите её и добавьте снова")
        if InventoryLine.objects.filter(
            pallet_id=line.pallet_id, document__status=InventoryDocument.Status.COUNTING
        ).exists():
            raise WarehouseError(f"Паллета {label} в незакрытой инвентаризации")
        locked.append((line, inbound))
    incoming, outgoing, declared = _totals(document)
    gap = as_kg(incoming - outgoing - declared)
    tolerance = as_kg(document.tolerance_kg)
    if abs(gap) > tolerance:
        raise WarehouseError(
            "Баланс масс не сходится: "
            f"вход {format_kg(incoming)}, выход {format_kg(outgoing)}, отход {format_kg(declared)}, "
            f"расхождение {format_kg(gap)} при допуске {format_kg(tolerance)}"
        )
    lock_balances(inbound.lot_id for _line, inbound in locked)
    category_ids = output_category_ids(document)
    _output_reserve_fits(document)
    output_reserved = bool(document.output_line_id or document.output_consignee_id)
    parent = _parent_lot(locked)
    for line, inbound, out_part, waste_part in _split(locked, outgoing):
        reserved = inbound_is_reserved(inbound)
        if out_part > 0:
            post_transform_out(
                pallet=inbound.pallet,
                lot=inbound.lot,
                qty_kg=out_part,
                user=user,
                document=document,
                reserved=reserved,
                source=inbound,
            )
        if waste_part > 0:
            post_waste(
                pallet=inbound.pallet,
                lot=inbound.lot,
                qty_kg=waste_part,
                user=user,
                document=document,
                reserved=reserved,
                source=inbound,
            )
        _clear_pallet(inbound.pallet)
    for line in outputs:
        lot = Lot.objects.create(
            receipt_id=parent.receipt_id,
            parent=parent,
            source=document.kind,
            supplier_id=parent.supplier_id,
            product_id=line.product_id,
            caliber_id=line.caliber_id,
            grade_id=line.grade_id,
            received_on=document.doc_date,
            expires_on=line.expires_on or parent.expires_on,
            is_opening=False,
        )
        pallet = _output_pallet(user)
        if category_ids:
            pallet.allowed_categories.set(category_ids)
        if document.output_line_id:
            pallet.reserved_line_id = document.output_line_id
            pallet.order_id = document.output_line.order_id
        elif document.output_consignee_id:
            pallet.reserved_consignee_id = document.output_consignee_id
        if output_reserved:
            pallet.save(update_fields=["reserved_line", "order", "reserved_consignee"])
        post_transform_in(
            pallet=pallet,
            lot=lot,
            qty_kg=line.qty_kg,
            user=user,
            document=document,
            reserved=output_reserved,
        )
        line.lot = lot
        line.pallet = pallet
        line.save(update_fields=["lot", "pallet"])
    book = as_kg(incoming - outgoing) if incoming >= outgoing else Decimal("0.000")
    document.status = TransformDocument.Status.POSTED
    document.ledger_waste_kg = book
    document.posted_by = user
    document.posted_at = timezone.now()
    document.save(update_fields=["status", "ledger_waste_kg", "posted_by", "posted_at"])
    message = "Документ проведён"
    if outgoing > incoming:
        message += f". Выход больше входа на {format_kg(outgoing - incoming)} кг, в допуске"
    elif declared != book:
        message += f". Отход в движениях {format_kg(book)} кг, в строках {format_kg(declared)} кг"
    return message

def _contains(value: str, needle: str) -> bool:
    return needle.casefold() in (value or "").casefold()


def _demand_query(request):
    if "date_from" not in request.GET and "date_to" not in request.GET:
        today = timezone.localdate()
        date_from_raw = today.isoformat()
        date_to_raw = (today + timedelta(days=7)).isoformat()
    else:
        date_from_raw = request.GET.get("date_from") or ""
        date_to_raw = request.GET.get("date_to") or ""
    product = (request.GET.get("product") or "").strip()
    party = (request.GET.get("party") or "").strip()
    date_from = query_date(date_from_raw)
    date_to = query_date(date_to_raw)

    def line_visible(row) -> bool:
        if date_from and row["ship_date"] < date_from:
            return False
        if date_to and row["ship_date"] > date_to:
            return False
        if product and not _contains(row["product"], product) and not _contains(row["spec"], product):
            return False
        if party and not _contains(row["consignee"], party):
            return False
        return True

    def inbound_visible(row) -> bool:
        receipt = row["receipt_date"]
        if date_from or date_to:
            if receipt is None:
                return False
            if date_from and receipt < date_from:
                return False
            if date_to and receipt > date_to:
                return False
        if product and not _contains(row["product"], product):
            return False
        return True

    return {
        "date_from": date_from_raw,
        "date_to": date_to_raw,
        "product": product,
        "party": party,
        "filtered": bool(date_from or date_to or product or party),
        "line_visible": line_visible,
        "inbound_visible": inbound_visible,
    }


def _id_set(values) -> set[int]:
    chosen = set()
    for raw in values:
        try:
            chosen.add(int(raw))
        except (TypeError, ValueError):
            continue
    return chosen


def _flag(values, index) -> bool:
    return index < len(values) and values[index] == "1"


def _demand_grouping(request):
    known = {key for key, _label in GROUP_FIELDS}
    labels = dict(GROUP_FIELDS)
    if request.GET.get("group_set") == "1":
        keys = request.GET.getlist("group_by")
        ons = request.GET.getlist("group_on")
        rows = []
        seen = set()
        for index, key in enumerate(keys):
            if key not in known or key in seen:
                continue
            seen.add(key)
            rows.append({"key": key, "label": labels[key], "on": True if not ons else _flag(ons, index)})
    else:
        legacy = request.GET.get("group") or "product"
        if legacy == "client":
            legacy = "consignee"
        if legacy not in known:
            legacy = "product"
        rows = [{"key": legacy, "label": labels[legacy], "on": True}]
    return rows, [row["key"] for row in rows if row["on"]]


def _demand_columns(request):
    labels = {key: label for key, label, _num in DEMAND_COLUMNS}
    nums = {key for key, _label, num in DEMAND_COLUMNS if num}
    if request.GET.get("col_set") == "1":
        keys = request.GET.getlist("col")
        ons = request.GET.getlist("col_on")
        rows = []
        seen = set()
        for index, key in enumerate(keys):
            if key not in labels or key in seen:
                continue
            seen.add(key)
            rows.append({"key": key, "label": labels[key], "num": key in nums, "on": _flag(ons, index)})
    else:
        rows = [
            {"key": key, "label": label, "num": num, "on": key in DEFAULT_COLUMNS}
            for key, label, num in DEMAND_COLUMNS
        ]
    visible = []
    labeled = False
    for row in rows:
        if not row["on"]:
            continue
        item = dict(row)
        item["total_label"] = not item["num"] and not labeled
        if item["total_label"]:
            labeled = True
        visible.append(item)
    return rows, visible


def _demand_picks(request):
    fields = {key for key, _label in PICK_FIELDS}
    ops = {key for key, _label in PICK_OPS}
    labels = dict(PICK_FIELDS)
    op_labels = dict(PICK_OPS)
    if request.GET.get("pick_set") != "1":
        return []
    rows = []
    keys = request.GET.getlist("pick_field")
    operators = request.GET.getlist("pick_op")
    values = request.GET.getlist("pick_value")
    ons = request.GET.getlist("pick_on")
    for index, key in enumerate(keys):
        if key not in fields:
            continue
        raw_op = operators[index] if index < len(operators) else "equals"
        raw_op = OP_ALIAS.get(raw_op, raw_op)
        op = raw_op if raw_op in ops else "equals"
        rows.append(
            {
                "field": key,
                "label": labels[key],
                "op": op,
                "op_label": op_labels[op],
                "value": values[index] if index < len(values) else "",
                "on": _flag(ons, index),
            }
        )
    return rows


class _Down:
    def __init__(self, value):
        self.value = value

    def __lt__(self, other):
        return self.value > other.value


def _raw(row, field):
    key = VALUE_KEYS.get(field, field)
    if key not in row and field == "ship_date":
        return row.get("receipt_date")
    return row.get(key) if key in row else None


def _blank(value) -> bool:
    return value is None or (isinstance(value, str) and not value.strip())


def _texts(value) -> tuple[str, ...]:
    if _blank(value):
        return ("",)
    if isinstance(value, datetime):
        value = value.date()
    if hasattr(value, "strftime"):
        return (value.strftime("%d.%m.%Y"), value.isoformat())
    if isinstance(value, Decimal):
        plain = format(value, "f")
        shown = f"{value:.3f}"
        return (plain, plain.replace(".", ","), shown, shown.replace(".", ","))
    return (str(value),)


def _number(text):
    try:
        return Decimal(str(text).replace(" ", "").replace(",", "."))
    except (InvalidOperation, ValueError):
        return None


def _day(text):
    text = str(text).strip()
    for fmt in ("%Y-%m-%d", "%d.%m.%Y"):
        try:
            return datetime.strptime(text, fmt).date()
        except ValueError:
            continue
    return None


def _equals(value, needle: str) -> bool:
    if isinstance(value, Decimal):
        number = _number(needle)
        return number is not None and value == number
    if hasattr(value, "strftime"):
        day = _day(needle)
        current = value.date() if isinstance(value, datetime) else value
        return day is not None and current == day
    return any(item.casefold() == needle.casefold() for item in _texts(value))


def _contains_text(value, needle: str) -> bool:
    folded = needle.casefold()
    return any(folded in item.casefold() for item in _texts(value))


def _ordered(value, needle: str):
    if isinstance(value, Decimal):
        number = _number(needle)
        return None if number is None else (value, number)
    if hasattr(value, "strftime"):
        day = _day(needle)
        current = value.date() if isinstance(value, datetime) else value
        return None if day is None else (current, day)
    return (str(value or "").casefold(), needle.casefold())


def _compare(value, op, needle: str) -> bool:
    op = OP_ALIAS.get(op, op)
    if op == "is_empty":
        return _blank(value)
    if op == "is_not_empty":
        return not _blank(value)
    if op == "equals":
        return _equals(value, needle)
    if op == "not_equals":
        return not _equals(value, needle)
    if op == "not_contains":
        return not _contains_text(value, needle)
    if op == "starts_with":
        return any(item.casefold().startswith(needle.casefold()) for item in _texts(value))
    if op == "ends_with":
        return any(item.casefold().endswith(needle.casefold()) for item in _texts(value))
    if op in ("in_list", "not_in_list"):
        items = [line.strip() for line in needle.splitlines() if line.strip()]
        found = any(_equals(value, item) for item in items) if items else False
        return found if op == "in_list" else not found
    pair = _ordered(value, needle)
    if pair is None:
        return False
    left, right = pair
    if op == "greater":
        return left > right
    if op == "less":
        return left < right
    if op == "greater_or_equal":
        return left >= right
    if op == "less_or_equal":
        return left <= right
    return _contains_text(value, needle)


def _pick_live(pick) -> bool:
    if not pick["on"]:
        return False
    if pick["op"] in ("is_empty", "is_not_empty"):
        return True
    if pick["op"] in ("equals", "not_equals"):
        return True
    return bool(str(pick["value"]).strip())


def _picks_ok(row, picks, allowed=None) -> bool:
    for pick in picks:
        if not _pick_live(pick):
            continue
        if allowed is not None and pick["field"] not in allowed:
            continue
        key = VALUE_KEYS.get(pick["field"], pick["field"])
        if key not in row and not (pick["field"] == "ship_date" and "receipt_date" in row):
            continue
        if not _compare(_raw(row, pick["field"]), pick["op"], str(pick["value"])):
            return False
    return True


def _norm(value):
    if _blank(value):
        return (0, "")
    if isinstance(value, datetime):
        value = value.date()
    if hasattr(value, "toordinal"):
        return (1, value.toordinal())
    if isinstance(value, Decimal):
        return (2, value)
    return (3, str(value).casefold())


def _ordering(sorts):
    active = [item for item in sorts if item["on"]]
    if not active:
        return None

    def key(row):
        parts = []
        for item in active:
            rank, raw = _norm(_raw(row, item["field"]))
            if item["desc"] and isinstance(raw, (int, Decimal)):
                parts.append((rank, -raw))
            elif item["desc"]:
                parts.append((rank, _Down(raw)))
            else:
                parts.append((rank, raw))
        return tuple(parts)

    return key


def _demand_sorts(request):
    labels = {key: label for key, label, _num in DEMAND_COLUMNS}
    if request.GET.get("sort_set") != "1":
        return []
    rows = []
    keys = request.GET.getlist("sort_by")
    directions = request.GET.getlist("sort_dir")
    ons = request.GET.getlist("sort_on")
    seen = set()
    for index, key in enumerate(keys):
        if key not in labels or key in seen:
            continue
        seen.add(key)
        direction = directions[index] if index < len(directions) and directions[index] in ("asc", "desc") else "asc"
        rows.append(
            {
                "field": key,
                "label": labels[key],
                "dir": direction,
                "on": _flag(ons, index),
                "desc": direction == "desc",
            }
        )
    return rows

def _demand_line_cell(row, column):

        return row["ship_date"]
    if key in _LINE_KG:

def _demand_sum_cell(source, column, text_style, num_style):
    if column.get("total_label"):
        return source["title"], text_style
    field = _LINE_KG.get(column["key"])

        return source[field], num_style

def _demand_order_rows(context) -> list[list]:
    columns = context["columns"]
    header = [(column["label"], "header") for column in columns] or [("Поля", "header")]
    rows = [header]

    def walk(nodes):
        for item in nodes:
            if item["title"]:
                titled = {**item, "title": ("  " * item["level"]) + item["title"]}
                rows.append(
                    [_demand_sum_cell(titled, column, "group", "group_num") for column in columns]
                    or [(item["title"], "group")]
                )
            walk(item["children"])
            for line in item["rows"]:
                rows.append([_demand_line_cell(line, column) for column in columns])

    groups = context["groups"]
    if groups:
        walk(groups)
        totals = {
            "title": "Итого",
            "ordered_kg": context["ordered_total"],
            "picked_kg": context["picked_total"],
            "reserve_kg": context["reserve_total"],
            "free_kg": context["free_total"],
            "blocked_kg": context["blocked_total"],
            "stock_kg": context["stock_total"],
            "incoming_kg": context["incoming_total"],
            "short_kg": context["short_total"],
        }
        rows.append([_demand_sum_cell(totals, column, "total", "total_num") for column in columns] or [("Итого", "total")])
    else:
        if context["filtered"]:
            message = "Ничего не найдено."
        elif context["has_orders"]:
            message = "Нехватки нет."
        else:
            message = "Открытых заказов нет."
        rows.append([(message, "text")])
    return rows

@roles_required("receiver", "warehouse")
def demand(request):
    chosen = request.GET.getlist("all")
    show_all = "1" in chosen if chosen else True
    grouping, modes = _demand_grouping(request)
    column_rows, columns = _demand_columns(request)
    picks = _demand_picks(request)
    sorts = _demand_sorts(request)
    categories, category_ids = _demand_categories(request)
    export = request.GET.get("export") == "xlsx"
    built = request.GET.get("run") == "1" or export
    filters = _demand_query(request)
    picks_on = any(_pick_live(pick) for pick in picks)
    ordering = _ordering(sorts)
    context = {
        "groups": [],
        "show_all": show_all,
        "grouping": grouping,
        "column_rows": column_rows,
        "columns": columns,
        "picks": picks,
        "sorts": sorts,
        "pick_fields": [{"key": key, "label": label} for key, label in PICK_FIELDS],
        "pick_ops": [{"key": key, "label": label} for key, label in PICK_OPS],
        "group_fields": [{"key": key, "label": label} for key, label in GROUP_FIELDS],
        "column_fields": [{"key": key, "label": label} for key, label, _num in DEMAND_COLUMNS],
        "categories": categories,
        "built": built,
        "has_orders": False,
        "ordered_total": Decimal("0"),
        "picked_total": Decimal("0"),
        "reserve_total": Decimal("0"),
        "free_total": Decimal("0"),
        "blocked_total": Decimal("0"),
        "stock_total": Decimal("0"),
        "incoming_total": Decimal("0"),
        "short_total": Decimal("0"),
        "inbound": [],
        "inbound_total": Decimal("0"),
        "date_from": filters["date_from"],
        "date_to": filters["date_to"],
        "product": filters["product"],
        "party": filters["party"],
        "filtered": filters["filtered"] or picks_on or (category_ids is not None and category_ids != {item["id"] for item in categories}),
    }
    if not built:
        return render(request, "demand.html", context)
    report = demand_report()
    lines = report["lines"] if show_all else [row for row in report["lines"] if row["short_kg"] > 0]
    lines = [row for row in lines if filters["line_visible"](row) and _picks_ok(row, picks)]
    inbound = [row for row in report["inbound"] if filters["inbound_visible"](row) and _picks_ok(row, picks, INBOUND_PICKS)]
    if ordering:
        inbound = sorted(inbound, key=ordering)
    if category_ids is not None:
        lines = [row for row in lines if row["category_id"] in category_ids]
        inbound = [row for row in inbound if row["category_id"] in category_ids]
    context.update(
        {
            "groups": group_demand(lines, modes, ordering),
            "has_orders": report["has_orders"],
            "ordered_total": as_kg(sum((row["ordered_kg"] for row in lines), Decimal("0"))),
            "picked_total": as_kg(sum((row["picked_kg"] for row in lines), Decimal("0"))),
            "reserve_total": as_kg(sum((row["reserve_kg"] for row in lines), Decimal("0"))),
            "free_total": as_kg(sum((row["free_kg"] for row in lines), Decimal("0"))),
            "blocked_total": as_kg(sum((row["blocked_kg"] for row in lines), Decimal("0"))),
            "stock_total": as_kg(sum((row["stock_kg"] for row in lines), Decimal("0"))),
            "incoming_total": as_kg(sum((row["incoming_kg"] for row in lines), Decimal("0"))),
            "short_total": as_kg(sum((row["short_kg"] for row in lines), Decimal("0"))),
            "inbound": inbound,
            "inbound_total": as_kg(sum((row["left_kg"] for row in inbound), Decimal("0"))),
        }
    )
    if export:
        return _demand_xlsx(context)
    return render(request, "demand.html", context)
