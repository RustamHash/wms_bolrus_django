def _picked_grades(form) -> list[Grade]:
    raw_ids = []
    for raw in form.getlist("allowed_grade"):
        try:
            raw_ids.append(int(raw))
        except (TypeError, ValueError) as exc:
            raise WarehouseError("Выберите сортность") from exc
    grades = list(Grade.objects.filter(pk__in=set(raw_ids)).order_by("sort_order", "name"))
    if len(grades) != len(set(raw_ids)):
        raise WarehouseError("Выберите сортность")
    return grades


def assign_consignee_grades(consignee, grade_ids: list[int]) -> str:
    wanted = list(dict.fromkeys(grade_ids))
    grades = list(Grade.objects.filter(pk__in=wanted).order_by("sort_order", "name"))
    if len(grades) != len(wanted):
        raise WarehouseError("Выберите сортность")
    consignee.allowed_grades.set(grades)
    if not grades:
        return "любой сорт"
    return ", ".join(item.name for item in grades)


def assign_consignee_zone(consignee, name: str) -> str:
    name = " ".join((name or "").split())
    if len(name) > 120:
        raise WarehouseError("Название зоны длиннее 120 символов")
    if not name:
        if consignee.zone_id:
            consignee.zone = None
            consignee.save(update_fields=["zone"])
        return ""
    zone = DeliveryZone.objects.filter(name__iexact=name).first()
    if zone is None:
        try:
            zone = DeliveryZone.objects.create(name=name)
        except IntegrityError:
            zone = DeliveryZone.objects.get(name=name)
    if consignee.zone_id != zone.id:
        consignee.zone = zone
        consignee.save(update_fields=["zone"])
    return zone.name


def ensure_category_grade(name: str) -> Grade:
    grade = Grade.objects.filter(name=name).first()
    if grade is not None:
        return grade
    last = Grade.objects.order_by("-sort_order").values_list("sort_order", flat=True).first() or 0
    return Grade.objects.create(name=name, sort_order=last + 10)

def _code(form, label: str, limit: int = 32) -> str:
    value = " ".join(text(form, "code").split())
    if not value:
        raise WarehouseError(f"Укажите {label}")
    if len(value) > limit:
        raise WarehouseError(f"{label} длиннее {limit} символов")
    return value


def _save_warehouse_zone(item, form) -> None:
    name = " ".join(text(form, "name").split())
    code = _code(form, "код зоны")
    if not name:
        raise WarehouseError("Укажите название зоны")
    if len(name) > 120:
        raise WarehouseError("Название зоны длиннее 120 символов")
    barcode_value = " ".join(text(form, "barcode").split()) or code
    if len(barcode_value) > 64:
        raise WarehouseError("QR зоны длиннее 64 символов")
    others = WarehouseZone.objects.exclude(pk=item.pk or 0)
    if others.filter(code__iexact=code).exists():
        raise WarehouseError("Такой код зоны уже есть")
    if others.filter(name__iexact=name).exists():
        raise WarehouseError("Такая зона уже есть")
    if others.filter(barcode__iexact=barcode_value).exists():
        raise WarehouseError("Такой QR зоны уже есть")
    item.code = code
    item.name = name
    item.barcode = barcode_value
    item.save()


def _save_warehouse_cell(item, form) -> None:
    zone = WarehouseZone.objects.filter(pk=optional_int(form, "zone") or 0).first()
    if zone is None:
        raise WarehouseError("Выберите зону склада")
    code = _code(form, "место")
    name = " ".join(text(form, "name").split())
    if len(name) > 120:
        raise WarehouseError("Название ячейки длиннее 120 символов")
    barcode_value = " ".join(text(form, "barcode").split()) or cell_address(zone.code, code)
    if len(barcode_value) > 64:
        raise WarehouseError("QR ячейки длиннее 64 символов")
    same_zone = WarehouseCell.objects.filter(zone=zone).exclude(pk=item.pk or 0)
    if same_zone.filter(code__iexact=code).exists():
        raise WarehouseError("Такая ячейка уже есть в этой зоне")
    if WarehouseCell.objects.exclude(pk=item.pk or 0).filter(barcode__iexact=barcode_value).exists():
        raise WarehouseError("Такой QR ячейки уже есть")
    item.zone = zone
    item.code = code
    item.name = name
    item.barcode = barcode_value
    item.save()


def apply_master(kind: str, item, form) -> None:
    if kind == "warehouse-zones":
        _save_warehouse_zone(item, form)
        return
    if kind == "warehouse-cells":
        _save_warehouse_cell(item, form)
        return
    if kind == "printers":
        from apps.printing.services import apply_printer

        apply_printer(item, form)
        return
    if kind == "aliases":
        external = ExternalProduct.objects.filter(pk=optional_int(form, "external_product_id") or 0).first()
        product = Product.objects.filter(pk=optional_int(form, "product_id") or 0).first()
        if external is None or product is None:
            raise WarehouseError("Выберите номенклатуру 1С и номенклатуру склада")
        taken = ProductAlias.objects.filter(external_product=external).exclude(pk=item.pk or 0).first()
        if taken is not None:
            raise WarehouseError("Соответствие для этого кода уже есть")
        item.external_product = external
        item.product = product
        item.save()
        return
    name = text(form, "name")
    if not name:
        raise WarehouseError("Укажите название")
    item.name = name
    if kind in ("calibers", "grades", "categories"):
        item.sort_order = optional_int(form, "sort_order") or 0
    if kind == "categories":
        taken = ConsigneeCategory.objects.filter(name=item.name).exclude(pk=item.pk or 0).exists()
        if taken:
            raise WarehouseError("Такая категория уже есть")
        item.grade = ensure_category_grade(item.name)
    if kind == "products":
        item.code = text(form, "code")
        item.barcode = text(form, "barcode")
        if not item.code:
            raise WarehouseError("Укажите код")
        if not item.barcode:
            raise WarehouseError("Укажите штрихкод")
        if Product.objects.filter(code=item.code).exclude(pk=item.pk or 0).exists():
            raise WarehouseError("Такой код уже есть")
        if Product.objects.filter(barcode=item.barcode).exclude(pk=item.pk or 0).exists():
            raise WarehouseError("Такой штрихкод уже есть")
        group_id = optional_int(form, "group_id")
        if group_id is not None and not ProductGroup.objects.filter(pk=group_id).exists():
            raise WarehouseError("Выберите группу")
        item.group_id = group_id
    if kind == "externals":
        item.code = text(form, "code")
        if not item.code:
            raise WarehouseError("Укажите код")
        if ExternalProduct.objects.filter(code=item.code).exclude(pk=item.pk or 0).exists():
            raise WarehouseError("Такой код уже есть")
        group_id = optional_int(form, "group_id")
        if group_id is not None and not ProductGroup.objects.filter(pk=group_id).exists():
            raise WarehouseError("Выберите группу")
        item.group_id = group_id
    if kind == "suppliers":
        item.code = text(form, "code") or None
        if item.code and Supplier.objects.filter(code=item.code).exclude(pk=item.pk or 0).exists():
            raise WarehouseError("Такой код уже есть")
    if kind == "consignees":
        item.code = text(form, "code") or None
        if item.code and Consignee.objects.filter(code=item.code).exclude(pk=item.pk or 0).exists():
            raise WarehouseError("Такой код уже есть")
        item.address = text(form, "address")
        item.shipping_terms = text(form, "shipping_terms")
        zone_id = optional_int(form, "zone_id")
        if zone_id and not DeliveryZone.objects.filter(pk=zone_id).exists():
            raise WarehouseError("Выберите зону доставки")
        item.zone_id = zone_id
        delivery_order = optional_int(form, "delivery_order")
        if delivery_order is None:
            item.delivery_order = 0
        elif delivery_order < 0:
            raise WarehouseError("Очередность не может быть отрицательной")
        else:
            item.delivery_order = delivery_order
        category_id = optional_int(form, "category_id")
        if category_id and not ConsigneeCategory.objects.filter(pk=category_id).exists():
            raise WarehouseError("Выберите категорию")
        item.category_id = category_id
    if kind == "sellers":
        item.code = text(form, "code")
        if not item.code:
            raise WarehouseError("Укажите код продавца")
        item.requisites = text(form, "requisites")
    if kind == "packagings":
        raw = text(form, "tare_kg").replace(",", ".")
        if not raw:
            item.tare_kg = Decimal("0.000")
        else:
            try:
                tare = Decimal(raw).quantize(Decimal("0.001"))
            except InvalidOperation as exc:
                raise WarehouseError("Некорректный вес тары") from exc
            if tare < 0:
                raise WarehouseError("Вес тары не может быть отрицательным")
            item.tare_kg = tare
    item.save()
    if kind == "categories":
        from apps.catalog.grade_policy import fill_consignee_orders

        item.allowed_grades.set(_picked_grades(form))
        for consignee in Consignee.objects.filter(category=item).select_related("category").prefetch_related(
            "allowed_grades", "category__allowed_grades"
        ):
            fill_consignee_orders(consignee)
    if kind == "consignees":
        item.allowed_grades.set(_picked_grades(form))
        from apps.catalog.grade_policy import fill_consignee_orders

        fill_consignee_orders(item)

def wms(request):
    storage = messages.get_messages(request)
    flashes = []
    for message in storage:
        level = "error" if message.level >= messages.ERROR else "ok"
        text = str(message)
        if flashes and flashes[-1]["m"] == text and flashes[-1]["l"] == level:
            continue
        flashes.append({"m": text, "l": level})
    notices, unread = notice_state(request.user)
    return {
        "flashes": flashes,
        "notices": notices,
        "notice_unread": unread,
        "ROLE_LABELS": ROLE_LABELS,

def accept(get=None, post=None):
    def view(request, *args, **kwargs):
        handler = post if request.method == "POST" else get
        if handler is None:
            return HttpResponseNotAllowed(["POST"] if post else ["GET"])
        return handler(request, *args, **kwargs)

    return view


def roles_required(*roles, tsd=()):
    def decorator(view):
        @wraps(view)
        def wrapped(request, *args, **kwargs):
            user = request.user
            path = request.path
            on_tsd = path.startswith("/tsd")
            login_url = "/tsd/login" if on_tsd else "/login"
            if not user.is_authenticated or not user.is_active:
                return redirect(login_url)
            if not on_tsd and request.method != "POST" and not user.opens_web():
                if user.opens_tsd() and not (path.startswith("/shift") and user.can_shift()):
                    return redirect("/tsd")
            needed = set(tsd)
            if needed and needed & user.granted_access():
                return view(request, *args, **kwargs)
            if "shift" in roles and user.can_shift():
                return view(request, *args, **kwargs)
            if on_tsd and path.rstrip("/") == "/tsd" and user.can_shift():
                return view(request, *args, **kwargs)
            if user.role != "admin":
                allowed = set(roles)
                if "office" not in allowed:
                    allowed.add("office")
                if user.role not in allowed:
                    messages.error(request, "Недостаточно прав")
                    return redirect("/tsd" if on_tsd or user.opens_tsd() else "/")
            return view(request, *args, **kwargs)

        return wrapped

    return decorator

    try:
        with transaction.atomic():
            return fn(), None
    except WarehouseError as exc:

        return None, redirect(back)
    except (IntegrityError, ProtectedError):
        messages.error(request, "Запись уже есть или используется")
        return None, redirect(back)


def done(request, url, message):
    messages.success(request, message)
    return redirect(url)

from datetime import date, timedelta

from django.utils import timezone

from apps.common.forms import query_date


def period_date(period_type: str, today: date | None = None) -> date | None:
    today = today or timezone.localdate()
    if period_type == "today":
        return today
    if period_type == "yesterday":
        return today - timedelta(days=1)
    if period_type == "tomorrow":
        return today + timedelta(days=1)
    if period_type == "start_of_week":
        return today - timedelta(days=today.weekday())
    if period_type == "start_of_last_week":
        return today - timedelta(days=today.weekday() + 7)
    if period_type == "end_of_week":
        return today + timedelta(days=6 - today.weekday())
    if period_type == "end_of_last_week":
        return today + timedelta(days=6 - today.weekday() - 7)
    if period_type == "start_of_month":
        return today.replace(day=1)
    if period_type == "start_of_last_month":
        first = today.replace(day=1)
        return (first - timedelta(days=1)).replace(day=1)
    if period_type == "end_of_month":
        if today.month == 12:
            return date(today.year, 12, 31)
        return date(today.year, today.month + 1, 1) - timedelta(days=1)
    if period_type == "end_of_last_month":
        return today.replace(day=1) - timedelta(days=1)
    return None


def _bound(kind: str, raw: str) -> date | None:
    if kind == "custom":
        return query_date(raw)
    if kind:
        return period_date(kind)
    return None


def range_from_query(params, prefix: str, *, default_from: str = "") -> dict:
    """Границы «от/до» по пресету. Пустой «от» при заданном «до» считается сегодняшним днём."""
    from_type = (params.get(f"{prefix}_from_type") or "").strip()
    to_type = (params.get(f"{prefix}_to_type") or "").strip()
    from_value = (params.get(f"{prefix}_from") or "").strip()
    to_value = (params.get(f"{prefix}_to") or "").strip()
    if not any((from_type, to_type, from_value, to_value)):
        if not default_from:
            return {
                "from_type": "",
                "from_value": "",
                "to_type": "",
                "to_value": "",
                "from_date": None,
                "to_date": None,
            }
        from_type = default_from
    from_date = _bound(from_type, from_value)
    to_date = _bound(to_type, to_value)
    if from_date is None and (to_type or to_value):
        from_date = period_date("today")
    return {
        "from_type": from_type,
        "from_value": from_value if from_type == "custom" else "",
        "to_type": to_type,
        "to_value": to_value if to_type == "custom" else "",
        "from_date": from_date,
        "to_date": to_date,
    }

def _split_cell(cell):
    style = None
    if isinstance(cell, tuple):
        cell, style = cell
    if isinstance(cell, datetime):
        cell = cell.date()

        if isinstance(cell, date):
            style = "date"

def _cell_xml(ref: str, cell) -> str:
    value, style = _split_cell(cell)
    if value is None or value == "":
        if not style:
            return ""
        return f'<c r="{ref}" s="{style}"/>'
    if isinstance(value, date):
        return f'<c r="{ref}" s="{style}"><v>{_excel_day(value)}</v></c>'
    if isinstance(value, Decimal):
        return f'<c r="{ref}" s="{style}"><v>{format(value, "f")}</v></c>'
    if isinstance(value, bool):
        value = "1" if value else "0"
    if isinstance(value, (int, float)):
        return f'<c r="{ref}" s="{style}"><v>{value}</v></c>'
    text = escape(str(value))
    return f'<c r="{ref}" t="inlineStr" s="{style}"><is><t xml:space="preserve">{text}</t></is></c>'

def _opening_task(document: InventoryDocument, supplier: Supplier, user):
    task = (
        Task.objects.filter(kind="receive", supplier=supplier, ttn_number=document.number)
        .select_related("receipt")
        .first()
    )
    if task is not None and task.receipt_id:
        return task, task.receipt
    receipt = Receipt.objects.create(
        supplier=supplier,
        doc_date=document.doc_date,
        ttn_number=document.number,
        status="closed",
        created_by=user,
    )
    receipt.number = document_number("ПР", receipt.id)
    receipt.save(update_fields=["number"])
    task = Task.objects.create(
        kind="receive",
        status="closed",
        supplier=supplier,
        doc_date=document.doc_date,
        ttn_number=document.number,
        receipt=receipt,
        created_by=user,
    )
    task.number = document_number("ЗД", task.id)
    task.save(update_fields=["number"])
    return task, receipt

def _post_found(line: InventoryLine, user) -> None:
    fact = as_kg(line.fact_kg)
    if fact <= 0:
        raise WarehouseError("У новой паллеты вес должен быть больше нуля")
    if not line.product_id or not line.caliber_id or not line.grade_id:
        raise WarehouseError("У новой паллеты не указан товар, калибр или сорт")
    task, receipt = _opening_task(line.document, line.supplier, user)
    lot = _lot_on_receipt(receipt, line.product_id, line.caliber_id, line.grade_id)
    task_line = TaskLine.objects.create(
        task=task,
        product_id=line.product_id,
        caliber_id=line.caliber_id,
        grade_id=line.grade_id,
        qty_kg=fact,
    )
    now = timezone.now()
    pallet = Pallet.objects.create(
        kind="inbound",
        status="closed",
        cell=line.cell,
        location=line.place,
        created_by=user,
        closed_by=user,
        closed_at=now,
    )
    pallet.number = document_number("ПЛ", pallet.id)
    pallet.barcode = barcode("PL", pallet.id)
    pallet.save(update_fields=["number", "barcode"])
    post_in(task=task, task_line=task_line, pallet=pallet, lot=lot, qty_kg=fact, user=user)
    line.pallet = pallet
    line.lot = lot
    line.save(update_fields=["pallet", "lot"])


def _post_line(line: InventoryLine, user) -> None:
    if line.pallet_id is None:
        _post_found(line, user)
        return
    movement = (
        Movement.objects.select_for_update()
        .filter(pallet_id=line.pallet_id, kind=Movement.Kind.IN)
        .select_related("lot", "pallet")
        .first()
    )
    label = line.pallet.number or f"#{line.pallet_id}"
    if movement is None:
        raise WarehouseError(f"Нет прихода по паллете {label}")
    if has_left(movement):
        raise WarehouseError(f"По паллете {label} уже есть отбор, отгрузка или переработка")
    fact = as_kg(line.fact_kg)
    if fact == 0:
        remove_in(movement)
        return
    book = as_kg(line.book_kg)
    supplier_changed = line.supplier_id != line.lot.supplier_id
    if fact == book and not supplier_changed:
        return
    lot = movement.lot
    qty = fact if fact != book else as_kg(movement.qty_kg)
    if supplier_changed:
        lot = _lot_for_supplier(movement.lot, line.supplier_id)
    if lot.id == movement.lot_id and as_kg(movement.qty_kg) == qty:
        return
    update_in(movement, lot=lot, qty_kg=qty)


@transaction.atomic
def post_document(document: InventoryDocument, user) -> None:
    document = InventoryDocument.objects.select_for_update().get(pk=document.pk)
    if document.status != InventoryDocument.Status.COUNTING:
        raise WarehouseError("Провести можно только документ в подсчёте")
    lines = list(document.lines.select_related("lot", "pallet", "supplier", "product", "caliber", "grade", "cell__zone"))
    pending = sum(1 for line in lines if line.fact_kg is None)
    if pending:
        raise WarehouseError(f"Не посчитано паллет: {pending}")
    for line in lines:
        _post_line(line, user)
    document.status = InventoryDocument.Status.POSTED
    document.posted_by = user
    document.posted_at = timezone.now()
    document.save(update_fields=["status", "posted_by", "posted_at"])


@transaction.atomic
def cancel_document(document: InventoryDocument) -> None:
    document = InventoryDocument.objects.select_for_update().get(pk=document.pk)
    if document.status not in (InventoryDocument.Status.DRAFT, InventoryDocument.Status.COUNTING):
        raise WarehouseError("Проведённый документ нельзя отменить")
    document.status = InventoryDocument.Status.CANCELLED
    document.save(update_fields=["status"])

@roles_required("office")
def inventory_create(request):
    def action():
        scope = text(request.POST, "scope") or InventoryDocument.Scope.ALL
        raw_ids = request.POST.getlist("product_id")
        product_ids = []
        for raw in raw_ids:
            if not raw:
                continue
            try:
                product_ids.append(int(raw))
            except ValueError as exc:
                raise WarehouseError("Некорректная номенклатура") from exc
        if scope == InventoryDocument.Scope.PRODUCTS:
            known = set(Product.objects.filter(pk__in=product_ids).values_list("pk", flat=True))
            if len(known) != len(set(product_ids)):
                raise WarehouseError("Номенклатура не найдена")
        else:
            product_ids = []
        return create_document(
            doc_date=parse_date(text(request.POST, "doc_date"), "Дата"),
            scope=scope,
            product_ids=product_ids,
            notes=text(request.POST, "notes"),
            user=request.user,
        )

    result, response = guarded(request, "/inventory/new", action)
    if response:
        return response
    return redirect(f"/inventory/{result.id}")
