from apps.common.exceptions import WarehouseError
from apps.common.forms import text
from apps.storage.models import WarehouseCell, WarehouseZone


def place_form_context(cell: WarehouseCell | None = None) -> dict:
    return {
        "warehouse_zones": list(WarehouseZone.objects.order_by("code")),
        "warehouse_cells": list(WarehouseCell.objects.select_related("zone").order_by("zone__code", "code")),
        "selected_zone_id": cell.zone_id if cell is not None else None,
        "selected_cell_id": cell.id if cell is not None else None,
    }


def resolve_place(form, *, required: bool) -> WarehouseCell | None:
    zone_scan = text(form, "zone_scan")
    cell_scan = text(form, "cell_scan")
    zone_raw = text(form, "zone_id")
    cell_raw = text(form, "cell_id")
    if not any((zone_scan, cell_scan, zone_raw, cell_raw)):
        if required:
            raise WarehouseError("Укажите ячейку")
        return None
    zone = _zone(zone_raw, zone_scan)
    cell = _cell(cell_raw, cell_scan, zone)
    if zone is None and cell is not None:
        zone = cell.zone
    if cell is None:
        raise WarehouseError("Укажите ячейку")
    if zone is not None and cell.zone_id != zone.id:
        raise WarehouseError("Ячейка не относится к этой зоне")
    return cell


def _int(raw: str, label: str) -> int:
    try:
        return int(raw)
    except ValueError as exc:
        raise WarehouseError(f"Выберите {label}") from exc


def _zone(raw_id: str, scan: str) -> WarehouseZone | None:
    zone = None
    if raw_id:
        zone = WarehouseZone.objects.filter(pk=_int(raw_id, "зону")).first()
        if zone is None:
            raise WarehouseError("Такой зоны нет")
    if scan:
        found = (
            WarehouseZone.objects.filter(code__iexact=scan).first()
            or WarehouseZone.objects.filter(barcode__iexact=scan).first()
            or WarehouseZone.objects.filter(name__iexact=scan).first()
        )
        if found is None:
            raise WarehouseError("Такой зоны нет")
        if zone is not None and found.id != zone.id:
            raise WarehouseError("Код зоны не совпадает с выбранной зоной")
        zone = found
    return zone


def _cell(raw_id: str, scan: str, zone: WarehouseZone | None) -> WarehouseCell | None:
    cell = None
    if raw_id:
        cell = WarehouseCell.objects.select_related("zone").filter(pk=_int(raw_id, "ячейку")).first()
        if cell is None:
            raise WarehouseError("Такой ячейки нет")
    if scan:
        found = _match_cell(scan, zone)
        if cell is not None and found.id != cell.id:
            raise WarehouseError("Код ячейки не совпадает с выбранной ячейкой")
        cell = found
    return cell


def _match_cell(scan: str, zone: WarehouseZone | None) -> WarehouseCell:
    query = WarehouseCell.objects.select_related("zone")
    if zone is not None:
        query = query.filter(zone=zone)
    found = list(query.filter(barcode__iexact=scan)[:2])
    if len(found) == 1:
        return found[0]
    if len(found) > 1:
        raise WarehouseError("Такой QR есть у нескольких ячеек")
    needle = "".join(scan.split()).casefold()
    labeled = [cell for cell in query if "".join(cell.label().split()).casefold() == needle]
    if len(labeled) == 1:
        return labeled[0]
    if len(labeled) > 1:
        raise WarehouseError("Такое имя есть у нескольких ячеек")
    coded = list(query.filter(code__iexact=scan)[:2])
    if len(coded) == 1:
        return coded[0]
    if len(coded) > 1:
        raise WarehouseError("Такое место есть в нескольких зонах. Укажите ячейку целиком, например А-01")
    raise WarehouseError("Такой ячейки нет")

def active_units(pallet: Pallet) -> list[HandlingUnit]:
    return [unit for unit in pallet.units.all() if unit.status != "rejected"]


def unit_key(unit: HandlingUnit) -> tuple:
    lot = unit.lot
    return (lot.product_id, lot.caliber_id, unit.grade_id)


def allocation_of(unit: HandlingUnit):
    return Allocation.objects.filter(unit_id=unit.id).select_related("order_line").first()


def _status_after_detach(pallet: Pallet, unit: HandlingUnit) -> str:
    if unit.status == "allocated":
        return "allocated"
    if pallet.status == "closed":
        return "in_stock"
    return "received"


def _locked_by_movement(unit: HandlingUnit) -> None:
    if unit.pallet_id and inbound_for_pallet(unit.pallet_id) is not None:
        raise WarehouseError("Эту паллету меняют в задании: правка идёт через движение прихода")


@transaction.atomic
def create_inbound_pallet(*, user, location: str = "", cell=None) -> Pallet:
    if cell is not None:
        location = cell.label()
    pallet = Pallet.objects.create(
        kind="inbound",
        status="open",
        location=location.strip(),
        cell=cell,
        created_by=user,
    )
    pallet.number = document_number("ПЛ", pallet.id)
    pallet.barcode = barcode("PL", pallet.id)
    pallet.save(update_fields=["number", "barcode"])
    return pallet


@transaction.atomic
def create_outbound_pallet(*, order_id: int, user, location: str = "") -> Pallet:
    order = get_row(Order, order_id, "заказ")
    pallet = Pallet.objects.create(
        kind="outbound",
        status="open",
        location=location.strip(),
        order=order,
        created_by=user,
    )
    pallet.number = document_number("ПЛ", pallet.id)
    pallet.barcode = barcode("PL", pallet.id)
    pallet.save(update_fields=["number", "barcode"])
    return pallet


def add_to_inbound(pallet: Pallet, unit: HandlingUnit) -> None:
    if inbound_for_pallet(pallet.id) is not None:
        raise WarehouseError("Принятую паллету меняют через движение задания")
    if pallet.kind != "inbound":
        raise WarehouseError("На входную паллету нельзя положить место с выхода")
    if pallet.status != "open":
        raise WarehouseError("Входная паллета закрыта. Откройте следующую")
    if unit.status in ("shipped", "rejected"):
        raise WarehouseError("Это место нельзя положить на паллету")
    if unit.pallet_id and unit.pallet_id != pallet.id:
        raise WarehouseError("Место уже лежит на другой паллете")
    keys = {unit_key(item) for item in active_units(pallet) if item.id != unit.id}
    if len(keys) > 1 or (keys and unit_key(unit) not in keys):
        raise WarehouseError(
            "На входе паллета только однородная: товар, калибр и сортность должны совпасть. Закройте паллету и откройте следующую"
        )
    unit.pallet = pallet
    if unit.status != "allocated":
        unit.status = "on_pallet"
    unit.save(update_fields=["pallet", "status"])


def add_to_outbound(pallet: Pallet, unit: HandlingUnit) -> None:
    if pallet.kind != "outbound":
        raise WarehouseError("Это не отгрузочная паллета")
    if pallet.status != "open":
        raise WarehouseError("Отгрузочная паллета закрыта")
    allocation = allocation_of(unit)
    if unit.status != "allocated" or allocation is None:
        raise WarehouseError("На выход кладут только места, уже назначенные на заказ")
    if allocation.order_line.order_id != pallet.order_id:
        raise WarehouseError("Место назначено другому заказу")
    if unit.pallet_id and unit.pallet and unit.pallet.kind == "outbound" and unit.pallet_id != pallet.id:
        raise WarehouseError("Место уже на другой отгрузочной паллете")
    unit.pallet = pallet
    unit.save(update_fields=["pallet"])


def detach_unit(unit: HandlingUnit) -> None:
    _locked_by_movement(unit)
    if unit.pallet_id is None:
        return
    pallet = unit.pallet
    if unit.status == "shipped":
        raise WarehouseError("Отгруженное место нельзя снять с паллеты")
    unit.status = _status_after_detach(pallet, unit)
    unit.pallet = None
    unit.save(update_fields=["status", "pallet"])


def close_pallet(pallet: Pallet, user) -> None:
    if pallet.status == "closed":
        return
    units = active_units(pallet)
    if pallet.kind == "inbound" and not units:
        raise WarehouseError("Нельзя закрыть пустую паллету")
    for unit in units:
        if pallet.kind == "inbound" and unit.status == "on_pallet":
            unit.status = "in_stock"
            unit.save(update_fields=["status"])
    pallet.status = "closed"
    pallet.closed_at = timezone.now()
    pallet.closed_by = user
    pallet.save(update_fields=["status", "closed_at", "closed_by"])


def change_grade(unit: HandlingUnit, grade_id: int) -> None:
    _locked_by_movement(unit)
    if unit.status in ("shipped", "rejected"):
        raise WarehouseError("У этого места нельзя менять сортность")
    if unit.pallet_id and unit.pallet.kind == "inbound":
        raise WarehouseError("Снимите место с входной паллеты, затем меняйте сортность")
    if allocation_of(unit) is not None:
        raise WarehouseError("Снимите назначение на заказ, затем меняйте сортность")
    grade = get_row(Grade, grade_id, "сортность")
    unit.grade = grade
    unit.save(update_fields=["grade"])


def reject_unit(unit: HandlingUnit, reason: str) -> None:
    _locked_by_movement(unit)
    if unit.status == "shipped":
        raise WarehouseError("Отгруженное место нельзя отправить в брак")
    if unit.status == "rejected":
        return
    if unit.pallet_id and unit.pallet.kind == "inbound":
        raise WarehouseError("Снимите место с входной паллеты, затем отметьте брак")
    if allocation_of(unit) is not None:
        raise WarehouseError("Снимите назначение на заказ, затем отметьте брак")
    unit.status = "rejected"
    unit.reject_reason = reason.strip()
    unit.pallet = None
    unit.save(update_fields=["status", "reject_reason", "pallet"])


def stock_place_rows() -> list[dict]:
    units = HandlingUnit.objects.filter(status__in=("received", "on_pallet", "in_stock", "allocated")).select_related(
        "lot__product", "lot__caliber", "grade"
    )
    buckets: dict[tuple, dict] = {}
    for unit in units:
        lot = unit.lot
        key = (lot.product.name, lot.caliber.name, unit.grade.name)
        row = buckets.setdefault(
            key,
            {
                "product": key[0],
                "caliber": key[1],
                "grade": key[2],
                "free_kg": Decimal("0.000"),
                "allocated_kg": Decimal("0.000"),
                "places": 0,
            },
        )
        kg = as_kg(unit.net_kg)
        row["places"] += 1
        if unit.status == "allocated":
            row["allocated_kg"] += kg
        else:
            row["free_kg"] += kg
    return [buckets[key] for key in sorted(buckets)]


def find_by_barcode(code: str):
    raw = (code or "").strip()
    pieces = [raw]
    first = raw.splitlines()[0].strip() if raw else ""
    if first and first != raw:
        pieces.append(first)
    for piece in pieces:
        normalized = piece.upper()
        if not normalized:
            continue
        unit = HandlingUnit.objects.filter(barcode=normalized).first()
        if unit:
            return ("unit", unit)
        pallet = Pallet.objects.filter(barcode=normalized).first() or Pallet.objects.filter(number=piece).first()
        if pallet:
            return ("pallet", pallet)
    return (None, None)

class OpenAdmin(ModelAdmin):
    def _permit(self, request, action):
        user = request.user
        if not user.is_authenticated or not user.is_active:
            return False
        if user.is_superuser or getattr(user, "role", "") == "admin":
            return True
        if not getattr(user, "access_managed", False):
            return True
        opts = self.opts
        return user.has_perm(f"{opts.app_label}.{action}_{opts.model_name}")

    def has_module_permission(self, request):
        return any(self._permit(request, action) for action, _label in ACTIONS)

    def has_view_permission(self, request, obj=None):
        return self._permit(request, "view") or self._permit(request, "change")

    def has_add_permission(self, request):
        return self._permit(request, "add")

    def has_change_permission(self, request, obj=None):
        return self._permit(request, "change")

    def has_delete_permission(self, request, obj=None):
        return self._permit(request, "delete")

"""Одна и та же команда TSC без SIZE, GAP и DENSITY.

На этом компьютере принтер сидит на USB, очередь Windows «TSC TE200».
Скрипт кладёт байты в очередь как RAW, драйвер их не перерисовывает.

Запуск на USB:
    python scripts/tsc_paper_test.py

Другая очередь:
    python scripts/tsc_paper_test.py --queue "TSC TE200"

Сеть, порт 9100:
    python scripts/tsc_paper_test.py 192.168.1.50
"""

import ctypes
import socket
import sys
from ctypes import wintypes

PAYLOAD = (
    "CLS\r\n"
    'TEXT 20,20,"3",0,1,1,"NO SIZE"\r\n'
    'TEXT 20,70,"3",0,1,1,"SAME JOB"\r\n'
    "BAR 20,130,280,8\r\n"
    "PRINT 1\r\n"
)

DEFAULT_QUEUE = "TSC TE200"


class DOCINFO(ctypes.Structure):
    _fields_ = [
        ("pDocName", wintypes.LPWSTR),
        ("pOutputFile", wintypes.LPWSTR),
        ("pDatatype", wintypes.LPWSTR),
    ]


def send_queue(queue: str, payload: bytes) -> None:
    winspool = ctypes.WinDLL("winspool.drv", use_last_error=True)
    winspool.OpenPrinterW.argtypes = [wintypes.LPWSTR, ctypes.POINTER(wintypes.HANDLE), wintypes.LPVOID]
    winspool.OpenPrinterW.restype = wintypes.BOOL
    winspool.ClosePrinter.argtypes = [wintypes.HANDLE]
    winspool.ClosePrinter.restype = wintypes.BOOL
    winspool.StartDocPrinterW.argtypes = [wintypes.HANDLE, wintypes.DWORD, ctypes.POINTER(DOCINFO)]
    winspool.StartDocPrinterW.restype = wintypes.DWORD
    winspool.EndDocPrinter.argtypes = [wintypes.HANDLE]
    winspool.EndDocPrinter.restype = wintypes.BOOL
    winspool.StartPagePrinter.argtypes = [wintypes.HANDLE]
    winspool.StartPagePrinter.restype = wintypes.BOOL
    winspool.EndPagePrinter.argtypes = [wintypes.HANDLE]
    winspool.EndPagePrinter.restype = wintypes.BOOL
    winspool.WritePrinter.argtypes = [wintypes.HANDLE, wintypes.LPVOID, wintypes.DWORD, ctypes.POINTER(wintypes.DWORD)]
    winspool.WritePrinter.restype = wintypes.BOOL

    handle = wintypes.HANDLE()
    if not winspool.OpenPrinterW(queue, ctypes.byref(handle), None):
        raise OSError(ctypes.get_last_error(), f"очередь «{queue}» не открылась")
    try:
        info = DOCINFO("tsc-paper-test", None, "RAW")
        if not winspool.StartDocPrinterW(handle, 1, ctypes.byref(info)):
            raise OSError(ctypes.get_last_error(), "Windows не приняла задание")
        try:
            if not winspool.StartPagePrinter(handle):
                raise OSError(ctypes.get_last_error(), "Windows не открыла страницу")
            written = wintypes.DWORD()
            buffer = ctypes.create_string_buffer(payload)
            if not winspool.WritePrinter(handle, buffer, len(payload), ctypes.byref(written)):
                raise OSError(ctypes.get_last_error(), "Windows не записала данные")
            if written.value != len(payload):
                raise OSError(0, f"ушло {written.value} байт из {len(payload)}")
            winspool.EndPagePrinter(handle)
        finally:
            winspool.EndDocPrinter(handle)
    finally:
        winspool.ClosePrinter(handle)


def send_tcp(host: str, port: int, payload: bytes) -> None:
    with socket.create_connection((host, port), timeout=5) as sock:
        sock.sendall(payload)


def main() -> None:
    args = sys.argv[1:]
    data = PAYLOAD.encode("ascii")
    print(PAYLOAD, end="")
    if not args:
        send_queue(DEFAULT_QUEUE, data)
        print(f"Отправлено {len(data)} байт в очередь «{DEFAULT_QUEUE}»")
        return
    if args[0] == "--queue":
        if len(args) < 2:
            print("Нужно имя очереди: python scripts/tsc_paper_test.py --queue \"TSC TE200\"")
            sys.exit(1)
        send_queue(args[1], data)
        print(f"Отправлено {len(data)} байт в очередь «{args[1]}»")
        return
    host = args[0]
    port = int(args[1]) if len(args) > 1 else 9100
    send_tcp(host, port, data)
    print(f"Отправлено {len(data)} байт на {host}:{port}")


if __name__ == "__main__":
    main()

"""Печать текста на TSC по сети, порт 9100.

Меняйте переменные ниже и запускайте:
    python scripts/tsc_print.py

Обязательны: HOST, WIDTH_MM, HEIGHT_MM, LINES.
Без SIZE принтер эту команду молча не печатает.
GAP_MM = 0 протягивает ровно HEIGHT_MM и не ищет зазор между наклейками.
"""

import socket

# Куда слать. Порт 9100 — сырой вход TSC, его менять не нужно.
HOST = "192.168.1.133"
PORT = 9100

# Размер этикетки в миллиметрах. Эти числа попадают в команду SIZE.
WIDTH_MM = 100
HEIGHT_MM = 70

# Зазор между наклейками, мм. 0 — не искать зазор, протянуть HEIGHT_MM.
GAP_MM = 2

# Что печатать. Каждая строка списка — отдельная строка на этикетке.
LINES = [
    "WMS OK",
    "Проверка",
]

# Кодовая страница кириллицы у TSC. Латиница печатается и без неё.
CODEPAGE = "1251"

# Встроенный шрифт «3», увеличение 2. Координаты в точках, не в миллиметрах.
FONT = "3"
SCALE = 2
X = 50
FIRST_Y = 50
LINE_STEP = 50


def build() -> bytes:
    if not LINES:
        raise SystemExit("LINES пустой: нечего печатать")
    parts = [
        f"SIZE {int(WIDTH_MM)} mm,{int(HEIGHT_MM)} mm\r\n",
        f"GAP {int(GAP_MM)} mm,0 mm\r\n",
        "DIRECTION 1\r\n",
        "CLS\r\n",
        f"CODEPAGE {CODEPAGE}\r\n",
    ]
    y = FIRST_Y
    for line in LINES:
        text = str(line).replace('"', "'")
        parts.append(f'TEXT {X},{y},"{FONT}",0,{SCALE},{SCALE},"{text}"\r\n')
        y += LINE_STEP
    parts.append("PRINT 1,1\r\n")
    return "".join(parts).encode("cp1251", errors="replace")


def send(payload: bytes) -> None:
    with socket.create_connection((HOST, PORT), timeout=5) as sock:
        sock.sendall(payload)


def main() -> None:
    payload = build()
    print(payload.decode("cp1251"))
    send(payload)
    print(f"Отправлено {len(payload)} байт на {HOST}:{PORT}")


if __name__ == "__main__":
    main()

### 2. Ключевые бизнес-правила и логика закрытия переборки

#### А. Принудительное закрытие наряда (force_close_order)
Обычное закрытие наряда через ТСД требует условия равенства баланса нулю. Если веса не бьются (из-за усушки овощей или скрытых потерь), бэкенд предоставляет специальный метод для Начальника смены (apps/inventory/services.py):

def force_close_sorting_order(work_order_id, manager_user, reason_text):
    """ Принудительное схлопывание баланса конвейера переборки """
    from django.db import transaction
    from django.core.exceptions import ValidationError
    
    with transaction.atomic():
        order = SortingWorkOrder.objects.select_for_update().get(id=work_order_id)
        if order.status == 'COMPLETED':
            raise ValidationError("Наряд уже закрыт.")
            
        delta = order.calculate_balance_delta() # Находим разницу веса
        
        if delta != 0:
            order.unaccounted_loss = delta
            # Генерируем балансировочную транзакцию списания усушки/потерь
            last_pallet = order.input_pallets.first() or order.output_pallets.first()
            
            InventoryTransaction.objects.create(
                transaction_type='WASTE',
                pallet=last_pallet,
                quantity=delta,
                reference_document_id=order.id,
                user=manager_user
            )
            
        order.status = 'COMPLETED'
        order.is_force_closed = True
        order.force_close_reason = reason_text
        order.force_closed_by = manager_user
        order.save()

## ЧАСТЬ 2: Технологический регламент для персонала (Инструкция склада)

### ГЛАВА 1: Приемка автотранспорта на рампе

1. Запуск процесса: Откройте интерфейс WMS на ТСД (или ПК). Найдите плановое Поступление, назначенное вам. Переведите его в статус «В работе».
2. Входной фото/инспекционный контроль: Пройдите шаги контроля на ТСД. Если физическая возможность загрузить фото отсутствует (внутренний рейс, форс-мажор), вы обязаны заполнить текстовое поле причины!
   * Шаг 1 (Госномер): Сделайте фото номера авто. Если ТС без номеров — оставьте коммент причины.
   * Шаг 2 (Кузов): Откройте ворота кузова, сделайте фото первого товарного ряда.
   * Шаг 3 (Температура фуры): Внесите в систему показатели температуры воздуха внутри рефрижератора.
   * Шаг 4 (Пакет документов): Заберите у водителя накладные и сертификаты.
     * Если документы есть: Отсканируйте всю пачку документов с помощью складского сканера или приложения ТСД в один многостраничный файл формата PDF и загрузите его в систему.
     * Если документов нет (внутренний рейс/дошлют в офис): Нажмите «Пропустить загрузку PDF» и обязательно заполните поле комментария (например: «Документы отсутствуют, рейс внутренний, сканы пришлют напрямую в офис»). Без комментария система заблокирует приемку.

### ГЛАВА 2: Выгрузка и Оценка качества
1. Выгрузка: Направьте рабочих на разгрузку поддонов в зону приемки (STAGE-RECEIPT).
2. Температурный тест плода: Возьмите игольчатый термометр-щуп. Измерьте температуру внутри плода из глубины поддона, зафиксируйте значение в ТСД.
3. Органолептическая проверка: Осмотрите ящики/сетки. 
   * Если товар чистый и калиброванный — принимайте стандартно.
   * Если товар идет навалом, вперемешку по калибрам, содержит землю, сор или следы порчи — сделайте фото дефекта, внесите примерный процент брака на глаз (например, 5%) и примите решение о приеме через переборку.

### ГЛАВА 3: Попаллетное взвешивание валового товара под переборку
1. Закатите грязный паллет на весы.
2. На экране ТСД выберите плановую строку «Товар несортированный (Вал)».
3. Учет тары: Укажите тип поддона (деревянный евро = 20 кг) и тип упаковки (сетка/биг-бэг). Введите количество упаковок на поддоне. Система автоматически вычислит чистый вес «грязного» нетто.
4. Печать этикетки LPN: Подтвердите взвешивание. Система присвоит паллету статус «Требуется переборка» (NEED_SORT) и заблокирует его для коммерческих отгрузок. 
5. Наклейте распечатанную этикетку со штрихкодом LPN на торец паллета, отсканируйте её для подтверждения.
6. Направьте паллет силами погрузчика в буферную зону ожидания сортировки (ZONE-BUFFER-SORTING).
7. Повторите процедуру для всей машины. По окончании разгрузки подпишите водителю документы и переведите Поступление на ПК в статус «Завершено».

### ГЛАВА 4: Техпроцесс конвейерной переборки (Зона сортировки)
Операция выполняется в плановом режиме силами сортировочной бригады (возможно, на следующий день).

1. Старт наряда: Старший смены на ТСД открывает «Наряд на переборку», привязанный к исходной партии заехавшего товара.
2. Подача сырья (Вход): Погрузчик привозит грязные паллеты из буфера. Рабочий у ленты конвейера обязан отсканировать штрихкод LPN грязного паллета перед тем, как высыпать его на ленту. Вес списывается в «котел» наряда.
3. Переборка: Сортировщики очищают овощи/фрукты от грязи, калибруют и укладывают в чистую складскую тару.
4. Выпуск готовой продукции (Выход): Как только на выходе из конвейера формируется строго однородный, полный паллет, весовщик закатывает его на весы.
   * Выберите на ТСД его реальный получившийся сорт (например, 1 сорт или 2 сорт).
   * Укажите чистый вес тары. Нажмите «Выпустить паллет».
   * Система распечатает этикетку со статусом «Кондиция» (READY). Наклейте её. Такой паллет погрузчик может сразу увозить на стеллажи постоянного хранения для отгрузки клиентам. Паллет сохраняет сквозную связь с исходным поставщиком.
5. Учет гнили и отходов: Гниль, землю и сор рабочие сбрасывают в технический бак. По мере заполнения бака завезите его на весы, нажмите на ТСД кнопку «Зафиксировать брак». WMS автоматически вычтет вес бака и мгновенно спишет чистый вес гнили в утиль для учета в претензиях.
6. Закрытие наряда (Валидация баланса): Когда партия полностью переработана, взвесьте финальный остаток (если поддон остался неполным) и нажмите «Закрыть наряд».
   * Сценарий А (Баланс сошелся): Входной вес равен сумме выходов и отходов. Наряд успешно закрывается.
   * Сценарий Б (Ошибка веса): ТСД выдает ошибку расхождения веса (из-за усушки, испарения влаги или погрешности). Перепроверьте, все ли баки с отходами были взвешены. Если ошибок персонала нет — вызовите Начальника смены.
7. Принудительное закрытие (Только для Начальника смены):
   * Начальник смены оценивает величину дельты. Так как процент отбраковки и усушки уникален для каждой партии и поставщика овощей, начальник имеет право списать разницу.
   * Начальник заходит под своей учетной записью в WMS, нажимает кнопку «Закрыть принудительно» и вносит обязательный комментарий причины (например: «Естественная усушка влаги и осыпание песка/земли при калибровке партии»).
   * Система автоматически проводит балансировочное списание остатков под ноль и успешно переводит наряд в статус «Завершен».
