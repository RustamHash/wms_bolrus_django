from django.db import transaction

from apps.common.exceptions import WarehouseError

from apps.common.numbers import document_number
from apps.inventory.models import Movement
from apps.inventory.services import physical_left, post_out
from apps.planning.models import TaskLine
from apps.planning.spec import same_spec, specificity
from apps.shipping.models import Shipment, ShipmentUnit
from apps.storage.models import Pallet


def _same_spec(movement, line: TaskLine) -> bool:
    return same_spec(movement.lot, line)


def _shipment(task, user) -> Shipment:
    shipment = Shipment.objects.filter(task_id=task.id).first()
    if shipment is not None:
        return shipment
    order = task.order
    shipment = Shipment.objects.create(
        order=order,
        task=task,
        seller_name=order.seller.name,
        seller_requisites=order.seller.requisites,
        client_name=order.consignee.name,
        shipping_terms=order.shipping_terms,
        created_by=user,
    )
    shipment.number = document_number("ОТ", shipment.id)
    shipment.save(update_fields=["number"])
    return shipment


def _unshipped_pick(pallet_id: int, order_id: int, movement_id: int | None):
    shipped = Movement.objects.filter(kind="out").exclude(source_id=None).values("source_id")
    query = (
        Movement.objects.filter(kind="pick", pallet_id=pallet_id, task__order_id=order_id)
        .exclude(id__in=shipped)
        .select_related("task", "lot")
    )
    if movement_id:
        return query.filter(pk=movement_id).first()
    return query.order_by("id").first()


@transaction.atomic
def ship_pallet(*, task_line_id: int, pallet_id: int, user, confirmed: bool, movement_id: int | None = None):
    if not confirmed:
        raise WarehouseError("Подтвердите условия отгрузки")
    line = get_row(TaskLine, task_line_id, "строка задания")
    task = line.task
    if task.kind != "ship" or not task.is_working():
        raise WarehouseError("Нет открытого задания на отгрузку")
    pallet = get_row(Pallet, pallet_id, "паллета")
    source = _unshipped_pick(pallet.id, task.order_id, movement_id)
    if source is None:
        raise WarehouseError("Паллета ещё не отобрана")
    if not _same_spec(source, line):
        raise WarehouseError("Паллета не совпадает со строкой задания")
    movement = post_out(task=task, task_line=line, pallet=pallet, source=source, user=user)
    shipment = _shipment(task, user)
    if physical_left(pallet.id) <= 0:
        for unit in pallet.units.all():
            unit.status = "shipped"
            unit.save(update_fields=["status"])
            ShipmentUnit.objects.get_or_create(shipment=shipment, unit=unit)
    return movement, shipment


def waiting_pallets(task) -> list[dict]:
    shipped = Movement.objects.filter(kind="out").exclude(source_id=None).values("source_id")
    rows = (
        Movement.objects.filter(kind="pick", task__order_id=task.order_id)
        .exclude(id__in=shipped)
        .select_related("task", "pallet", "lot__product", "lot__caliber", "lot__grade")
    )
    found = []
    lines = sorted(
        task.lines.select_related(
            "order_line__order__consignee",
            "order_line__caliber_from",
            "order_line__caliber_to",
            "caliber",
            "grade",
        ).prefetch_related(
            "order_line__grades",
            "order_line__order__consignee__allowed_grades",
            "order_line__order__consignee__category__allowed_grades",
        ),
        key=specificity,
    )
    for movement in rows:
        for line in lines:
            if _same_spec(movement, line):
                found.append({"line": line, "movement": movement})
                break
    return found

def crossdock_redirect(request):
    return redirect("/tasks?kind=pick")


@roles_required("warehouse")
def shipment_list(request):
    date_from = query_date(request.GET.get("date_from"))
    date_to = query_date(request.GET.get("date_to"))
    party = (request.GET.get("party") or "").strip()
    number = (request.GET.get("number") or "").strip()
    query = Shipment.objects.select_related("order").order_by("-id")
    if date_from:
        query = query.filter(created_at__date__gte=date_from)
    if date_to:
        query = query.filter(created_at__date__lte=date_to)
    if party:
        query = query.filter(Q(client_name__icontains=party) | Q(seller_name__icontains=party))
    if number:
        query = query.filter(Q(number__icontains=number) | Q(order__number__icontains=number))
    return render(
        request,
        "shipments/list.html",
        {
            "shipments": query,
            "date_from": request.GET.get("date_from") or "",
            "date_to": request.GET.get("date_to") or "",
            "party": party,
            "number": number,
        },
    )


@roles_required("warehouse")
def shipment_detail(request, shipment_id):
    shipment = get_object_or_404(
        Shipment.objects.select_related("order").prefetch_related("units__lot__product", "units__lot__caliber", "units__grade", "units__packaging"),
        pk=shipment_id,
    )
    return render(request, "shipments/detail.html", {"shipment": shipment})


@roles_required("warehouse", tsd=("ship",))
def task_ship(request, task_id):
    form = request.POST
    back = text(form, "next") or f"/tasks/{task_id}"

    def action():
        task = Task.objects.filter(pk=task_id).first()
        if task is None:
            raise WarehouseError("Задание не найдено")
        require_assigned(task, request.user)
        return ship_pallet(
            task_line_id=require_int(form, "task_line_id", "Строка задания"),
            pallet_id=require_int(form, "pallet_id", "Паллета"),
            user=request.user,
            confirmed=form.get("confirm") == "1",
            movement_id=optional_int(form, "movement_id"),
        )

    result, response = guarded(request, back, action)
    if response:
        return response
    _movement, shipment = result
    if text(form, "next"):
        return done(request, text(form, "next"), "Паллета отгружена")
    return done(request, f"/shipments/{shipment.id}", "Паллета проведена в накладную")

def category_ids(pallet) -> set[int] | None:
    found = {item.id for item in pallet.allowed_categories.all()}
    if not found:
        return None
    return found


def allows_consignee(pallet, consignee) -> bool:
    allowed = category_ids(pallet)
    if allowed is None:
        return True
    return bool(consignee and consignee.category_id and consignee.category_id in allowed)

def cell_address(zone_code: str, code: str) -> str:
    zone = (zone_code or "").strip()
    place = (code or "").strip()
    if not zone:
        return place[:120]
    prefix = f"{zone}-"
    if place.casefold().startswith(prefix.casefold()):
        return place[:120]
    return f"{zone}-{place}"[:120]

def resolve_place(form, *, required: bool) -> WarehouseCell | None:
    zone_scan = text(form, "zone_scan")
    cell_scan = text(form, "cell_scan")
    zone_raw = text(form, "zone_id")
    cell_raw = text(form, "cell_id")
    if not any((zone_scan, cell_scan, zone_raw, cell_raw)):
        if required:
            raise WarehouseError("Укажите ячейку")
        return None
    zone = _zone(zone_raw, zone_scan)
    cell = _cell(cell_raw, cell_scan, zone)
    if zone is None and cell is not None:
        zone = cell.zone
    if cell is None:
        raise WarehouseError("Укажите ячейку")
    if zone is not None and cell.zone_id != zone.id:
        raise WarehouseError("Ячейка не относится к этой зоне")
    return cell


def _int(raw: str, label: str) -> int:
    try:
        return int(raw)
    except ValueError as exc:
        raise WarehouseError(f"Выберите {label}") from exc


def _zone(raw_id: str, scan: str) -> WarehouseZone | None:
    zone = None
    if raw_id:
        zone = WarehouseZone.objects.filter(pk=_int(raw_id, "зону")).first()
        if zone is None:
            raise WarehouseError("Такой зоны нет")
    if scan:
        found = (
            WarehouseZone.objects.filter(code__iexact=scan).first()
            or WarehouseZone.objects.filter(barcode__iexact=scan).first()
            or WarehouseZone.objects.filter(name__iexact=scan).first()
        )
        if found is None:
            raise WarehouseError("Такой зоны нет")
        if zone is not None and found.id != zone.id:
            raise WarehouseError("Код зоны не совпадает с выбранной зоной")
        zone = found
    return zone


def _cell(raw_id: str, scan: str, zone: WarehouseZone | None) -> WarehouseCell | None:
    cell = None
    if raw_id:
        cell = WarehouseCell.objects.select_related("zone").filter(pk=_int(raw_id, "ячейку")).first()
        if cell is None:
            raise WarehouseError("Такой ячейки нет")
    if scan:
        found = _match_cell(scan, zone)
        if cell is not None and found.id != cell.id:
            raise WarehouseError("Код ячейки не совпадает с выбранной ячейкой")
        cell = found
    return cell


def _match_cell(scan: str, zone: WarehouseZone | None) -> WarehouseCell:
    query = WarehouseCell.objects.select_related("zone")
    if zone is not None:
        query = query.filter(zone=zone)
    found = list(query.filter(barcode__iexact=scan)[:2])
    if len(found) == 1:
        return found[0]
    if len(found) > 1:
        raise WarehouseError("Такой QR есть у нескольких ячеек")
    needle = "".join(scan.split()).casefold()
    labeled = [cell for cell in query if "".join(cell.label().split()).casefold() == needle]
    if len(labeled) == 1:
        return labeled[0]
    if len(labeled) > 1:
        raise WarehouseError("Такое имя есть у нескольких ячеек")
    coded = list(query.filter(code__iexact=scan)[:2])
    if len(coded) == 1:
        return coded[0]
    if len(coded) > 1:
        raise WarehouseError("Такое место есть в нескольких зонах. Укажите ячейку целиком, например А-01")
    raise WarehouseError("Такой ячейки нет")

def add_to_inbound(pallet: Pallet, unit: HandlingUnit) -> None:
    if inbound_for_pallet(pallet.id) is not None:
        raise WarehouseError("Принятую паллету меняют через движение задания")
    if pallet.kind != "inbound":
        raise WarehouseError("На входную паллету нельзя положить место с выхода")
    if pallet.status != "open":
        raise WarehouseError("Входная паллета закрыта. Откройте следующую")
    if unit.status in ("shipped", "rejected"):
        raise WarehouseError("Это место нельзя положить на паллету")
    if unit.pallet_id and unit.pallet_id != pallet.id:
        raise WarehouseError("Место уже лежит на другой паллете")
    keys = {unit_key(item) for item in active_units(pallet) if item.id != unit.id}
    if len(keys) > 1 or (keys and unit_key(unit) not in keys):
        raise WarehouseError(
            "На входе паллета только однородная: товар, калибр и сортность должны совпасть. Закройте паллету и откройте следующую"
        )
    unit.pallet = pallet
    if unit.status != "allocated":
        unit.status = "on_pallet"
    unit.save(update_fields=["pallet", "status"])


def add_to_outbound(pallet: Pallet, unit: HandlingUnit) -> None:
    if pallet.kind != "outbound":
        raise WarehouseError("Это не отгрузочная паллета")
    if pallet.status != "open":
        raise WarehouseError("Отгрузочная паллета закрыта")
    allocation = allocation_of(unit)
    if unit.status != "allocated" or allocation is None:
        raise WarehouseError("На выход кладут только места, уже назначенные на заказ")
    if allocation.order_line.order_id != pallet.order_id:
        raise WarehouseError("Место назначено другому заказу")
    if unit.pallet_id and unit.pallet and unit.pallet.kind == "outbound" and unit.pallet_id != pallet.id:
        raise WarehouseError("Место уже на другой отгрузочной паллете")
    unit.pallet = pallet
    unit.save(update_fields=["pallet"])


def detach_unit(unit: HandlingUnit) -> None:
    _locked_by_movement(unit)
    if unit.pallet_id is None:
        return
    pallet = unit.pallet
    if unit.status == "shipped":
        raise WarehouseError("Отгруженное место нельзя снять с паллеты")
    unit.status = _status_after_detach(pallet, unit)
    unit.pallet = None
    unit.save(update_fields=["status", "pallet"])


def close_pallet(pallet: Pallet, user) -> None:
    if pallet.status == "closed":
        return
    units = active_units(pallet)
    if pallet.kind == "inbound" and not units:
        raise WarehouseError("Нельзя закрыть пустую паллету")
    for unit in units:
        if pallet.kind == "inbound" and unit.status == "on_pallet":
            unit.status = "in_stock"
            unit.save(update_fields=["status"])
    pallet.status = "closed"
    pallet.closed_at = timezone.now()
    pallet.closed_by = user
    pallet.save(update_fields=["status", "closed_at", "closed_by"])


def change_grade(unit: HandlingUnit, grade_id: int) -> None:
    _locked_by_movement(unit)
    if unit.status in ("shipped", "rejected"):
        raise WarehouseError("У этого места нельзя менять сортность")
    if unit.pallet_id and unit.pallet.kind == "inbound":
        raise WarehouseError("Снимите место с входной паллеты, затем меняйте сортность")
    if allocation_of(unit) is not None:
        raise WarehouseError("Снимите назначение на заказ, затем меняйте сортность")
    grade = get_row(Grade, grade_id, "сортность")
    unit.grade = grade
    unit.save(update_fields=["grade"])

        raise WarehouseError("Отгруженное место нельзя отправить в брак")
    if unit.status == "rejected":
        return
    if unit.pallet_id and unit.pallet.kind == "inbound":
        raise WarehouseError("Снимите место с входной паллеты, затем отметьте брак")
    if allocation_of(unit) is not None:
        raise WarehouseError("Снимите назначение на заказ, затем отметьте брак")
    unit.status = "rejected"

def stock_place_rows() -> list[dict]:
    units = HandlingUnit.objects.filter(status__in=("received", "on_pallet", "in_stock", "allocated")).select_related(
        "lot__product", "lot__caliber", "grade"
    )
    buckets: dict[tuple, dict] = {}
    for unit in units:
        lot = unit.lot
        key = (lot.product.name, lot.caliber.name, unit.grade.name)
        row = buckets.setdefault(
            key,
            {
                "product": key[0],
                "caliber": key[1],
                "grade": key[2],
                "free_kg": Decimal("0.000"),
                "allocated_kg": Decimal("0.000"),
                "places": 0,
            },
        )
        kg = as_kg(unit.net_kg)
        row["places"] += 1
        if unit.status == "allocated":
            row["allocated_kg"] += kg
        else:
            row["free_kg"] += kg
    return [buckets[key] for key in sorted(buckets)]


def find_by_barcode(code: str):
    raw = (code or "").strip()
    pieces = [raw]
    first = raw.splitlines()[0].strip() if raw else ""
    if first and first != raw:
        pieces.append(first)
    for piece in pieces:
        normalized = piece.upper()
        if not normalized:
            continue
        unit = HandlingUnit.objects.filter(barcode=normalized).first()
        if unit:
            return ("unit", unit)
        pallet = Pallet.objects.filter(barcode=normalized).first() or Pallet.objects.filter(number=piece).first()
        if pallet:
            return ("pallet", pallet)
    return (None, None)

def _inventory_lines(document):
    rows = list(
        InventoryLine.objects.filter(document=document)
        .select_related(
            "pallet",
            "lot__product",
            "lot__caliber",
            "lot__grade",
            "lot__supplier",
            "supplier",
            "product",
            "caliber",
            "grade",
            "cell__zone",
        )
        .order_by("id")
    )

    def label(row):
        product = row.lot.product if row.lot_id else row.product
        number = row.pallet.number if row.pallet_id else ""
        return ((product.name if product else "").casefold(), number)

    rows.sort(key=lambda row: (row.fact_kg is not None, *label(row)))
    return rows

class OpenAdmin(ModelAdmin):
    def _permit(self, request, action):
        user = request.user
        if not user.is_authenticated or not user.is_active:
            return False
        if user.is_superuser or getattr(user, "role", "") == "admin":
            return True
        if not getattr(user, "access_managed", False):
            return True
        opts = self.opts
        return user.has_perm(f"{opts.app_label}.{action}_{opts.model_name}")

    def has_module_permission(self, request):
        return any(self._permit(request, action) for action, _label in ACTIONS)

    def has_view_permission(self, request, obj=None):
        return self._permit(request, "view") or self._permit(request, "change")

    def has_add_permission(self, request):
        return self._permit(request, "add")

    def has_change_permission(self, request, obj=None):
        return self._permit(request, "change")

    def has_delete_permission(self, request, obj=None):
        return self._permit(request, "delete")
