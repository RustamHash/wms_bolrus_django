<script>
(function () {
  const nav = document.getElementById("catalog-nav");
  const box = document.getElementById("catalog-by-groups");
  if (!nav || !box) return;
  let stored = null;
  try { stored = localStorage.getItem("catalog-by-groups"); } catch (error) { stored = null; }
  const grouped = stored === null ? true : stored === "1";
  box.checked = grouped;
  nav.classList.toggle("by-groups", grouped);
  box.addEventListener("change", () => {
    nav.classList.toggle("by-groups", box.checked);
    nav.querySelectorAll(".catalog-group.open").forEach((group) => {
      group.classList.remove("open");
      const toggle = group.querySelector(".catalog-toggle");
      if (toggle) toggle.setAttribute("aria-expanded", "false");
    });
    try { localStorage.setItem("catalog-by-groups", box.checked ? "1" : "0"); } catch (error) {}
  });
  nav.querySelectorAll(".catalog-toggle").forEach((button) => {
    button.addEventListener("click", () => {
      if (!nav.classList.contains("by-groups")) return;
      const group = button.closest(".catalog-group");
      const willOpen = !group.classList.contains("open");
      nav.querySelectorAll(".catalog-group.open").forEach((item) => {
        item.classList.remove("open");
        const toggle = item.querySelector(".catalog-toggle");
        if (toggle) toggle.setAttribute("aria-expanded", "false");
      });
      if (willOpen) {
        group.classList.add("open");
        button.setAttribute("aria-expanded", "true");
      }
    });
  });
})();
</script>

  document.querySelectorAll("[data-order-tab]").forEach((button) => {
    button.addEventListener("click", () => {
      const name = button.dataset.orderTab;
      document.querySelectorAll("[data-order-tab]").forEach((item) => item.classList.toggle("on", item === button));
      document.querySelectorAll("[data-order-pane]").forEach((pane) => {
        pane.hidden = pane.dataset.orderPane !== name;
      });
    });
  });
  const table = document.getElementById("lines");

  function renumber() {
    let index = 1;
    table.querySelectorAll("tr").forEach((row) => {
      const cell = row.querySelector(".n");
      if (!cell) return;
      cell.textContent = index;
      index += 1;
    });
  }

  function bindRow(row) {
    const remove = row.querySelector("[data-remove]");
    if (remove) {
      remove.addEventListener("click", () => {
        const rows = table.querySelectorAll("tr .n");
        if (rows.length <= 1) {
          row.querySelectorAll("input").forEach((input) => {
            if (input.name !== "keep") input.value = "";
          });
          return;
        }
        row.remove();
        renumber();
      });
    }
  }

  function consigneeGrade() {
    const party = document.querySelector("[name=consignee_id]");
    if (!party) return "";
    const option = party.options[party.selectedIndex];
    return option && option.dataset.grade ? option.dataset.grade : "";
  }

  function fillFreshGrade(select) {
    if (!select || select.dataset.touched === "1") return;
    const grade = consigneeGrade();
    if (!grade) return;
    const has = Array.from(select.options).some((item) => item.value === grade);
    if (has) select.value = grade;
  }

  function watchGrade(select) {
    if (!select || select.dataset.watched === "1") return;
    select.dataset.watched = "1";
    select.addEventListener("change", () => {
      select.dataset.touched = "1";
    });
  }

  document.querySelectorAll("#lines tr").forEach(bindRow);
  document.querySelectorAll("[name=grade][data-fresh]").forEach(watchGrade);
  renumber();
  const party = document.querySelector("[name=consignee_id]");
  if (party) {
    party.addEventListener("change", () => {
      document.querySelectorAll("[name=grade][data-fresh]").forEach(fillFreshGrade);
    });
  }
  const add = document.getElementById("add-line");
  if (add) {
    add.addEventListener("click", () => {
      const row = document.getElementById("customer-line").content.firstElementChild.cloneNode(true);
      table.appendChild(row);
      bindRow(row);
      const grade = row.querySelector("[name=grade]");
      watchGrade(grade);
      fillFreshGrade(grade);
      renumber();
    });
  }

<script>
  const table = document.getElementById("lines");
  const customerCatalog = JSON.parse(document.getElementById("customer-line-data").textContent || "[]");

  function escapeHtml(value) {
    return String(value).replace(/[&<>"]/g, (char) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", "\"": "&quot;" }[char]));
  }

  function fillCustomer(select) {
    const row = select.closest("tr");
    const product = row ? row.querySelector("[data-line-product]") : null;
    const productId = product ? product.value : "";
    const selected = select.value || select.dataset.selected || "";
    const options = ['<option value="">Свободный склад</option>'];
    customerCatalog.forEach((item) => {
      const sameProduct = !productId || !item.product_id || String(item.product_id) === String(productId);
      if (!sameProduct && String(item.id) !== String(selected)) return;
      const mark = String(item.id) === String(selected) ? " selected" : "";
      options.push(`<option value="${item.id}"${mark}>${escapeHtml(item.label)}</option>`);
    });
    select.innerHTML = options.join("");
  }

  function renumber() {
    let index = 1;
    table.querySelectorAll("tbody tr, tr").forEach((row) => {
      const cell = row.querySelector(".n");
      if (!cell) return;
      cell.textContent = index;
      index += 1;
    });
  }

  function bindRow(row) {
    const customer = row.querySelector("[data-customer-line]");
    const product = row.querySelector("[data-line-product]");
    if (customer && !customer.dataset.bound) {
      customer.dataset.bound = "1";
      customer.addEventListener("focus", () => fillCustomer(customer));
    }
    if (product && !product.dataset.bound) {
      product.dataset.bound = "1";
      product.addEventListener("change", () => {
        const select = row.querySelector("[data-customer-line]");
        if (select) fillCustomer(select);
      });
    }
    const remove = row.querySelector("[data-remove]");
    if (!remove) return;
    remove.addEventListener("click", () => {
      const rows = table.querySelectorAll("tr .n");
      if (rows.length <= 1) {
        row.querySelectorAll("input").forEach((input) => { input.value = ""; });
        return;
      }
      row.remove();
      renumber();
    });
  }

  document.querySelectorAll("#lines tr").forEach(bindRow);
  renumber();
  const add = document.getElementById("add-line");
  if (add) {
    add.addEventListener("click", () => {
      const rows = [...table.querySelectorAll("tr")].filter((row) => row.querySelector(".n"));
      const copy = rows[rows.length - 1].cloneNode(true);
      copy.querySelectorAll("input").forEach((input) => {
        if (input.name === "unit") input.value = "кг";
        else if (input.name === "for_task") input.value = "1";
        else input.value = "";
      });
      const product = copy.querySelector("[data-line-product]");
      if (product) product.required = true;
      const customer = copy.querySelector("[name=customer_line]");
      if (customer) {
        customer.dataset.selected = "";
        customer.dataset.bound = "";
        customer.innerHTML = '<option value="">Свободный склад</option>';
      }
      const copiedProduct = copy.querySelector("[data-line-product]");
      if (copiedProduct) copiedProduct.dataset.bound = "";
      const hint = copy.querySelector(".reserve-hint");
      if (hint) hint.remove();
      table.appendChild(copy);
      bindRow(copy);
      renumber();
    });
  }
</script>
