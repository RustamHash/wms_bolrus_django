### 1. Архитектура сущностей и связей (ORM Django)
Учет остатков строится на неизменяемом Регистре движений (бухгалтерский метод). Таблица Stock выполняет роль денормализованного «снимка» текущего состояния для быстрой работы алгоритмов (например, подбора по FEFO). Любое изменение остатков обязано проводиться через запись транзакции в InventoryTransaction.

#### Набор ключевых полей для разработки моделей (apps/inbound/models.py):

class InboundOrder(models.Model):
    # status: PLAN (План), IN_PROGRESS (В работе), COMPLETED (Завершен), CANCELLED (Отменено)
    # receiver: ForeignKey на User (Назначенный приемщик)
    # truck_temperature: Decimal (Температура в кузове)
    # driver_info: CharField (ФИО, Госномер, данные авто)
    
    # Пакет документов в формате PDF. Может быть null, если документов нет.
    document_package_pdf = models.FileField("Скан пакета документов (PDF)", upload_to='inbound/documents/%Y/%m/%d/', null=True, blank=True)
    document_comment = models.TextField("Комментарий к документам", blank=True, help_text="Обязателен, если PDF отсутствует")
    created_at = models.DateTimeField(auto_now_add=True)

    def clean(self):
        from django.core.exceptions import ValidationError
        if not self.document_package_pdf and not self.document_comment:
            raise ValidationError({'document_comment': 'Вы должны загрузить PDF-скан документов или указать причину их отсутствия.'})

class InboundItem(models.Model):
    """ Строка плана приемки """
    order = models.ForeignKey(InboundOrder, on_delete=models.CASCADE, related_name='items')
    product = models.ForeignKey(Product, on_delete=models.PROTECT, verbose_name="Товар")
    grade = models.CharField("Сорт из плана (Напр: 1 сорт, Экстра. Для вала - Вал/Несортированный)", max_length=50)
    planned_quantity = models.DecimalField("Плановый вес, кг", max_digits=10, decimal_places=2)

    @property
    def actual_quantity(self):
        """ Факт — сумма транзакций INBOUND по связанным партиям этой строки """
        from apps.inventory.models import InventoryTransaction
        batch_ids = self.batches.values_list('id', flat=True)
        result = InventoryTransaction.objects.filter(
            transaction_type='INBOUND',
            pallet__batch_id__in=batch_ids
        ).aggregate(total=models.Sum('quantity'))['total']
        return result or 0.00

        ('DEFECT', 'Фото брака / дефекта товара'),
    ]
    order = models.ForeignKey(InboundOrder, on_delete=models.CASCADE, related_name='photos')
    photo_type = models.CharField(max_length=15, choices=PHOTO_TYPES)
    image = models.ImageField(upload_to='inbound/photos/', null=True, blank=True)
    comment = models.TextField("Комментарий к фото", blank=True)

    def clean(self):
        from django.core.exceptions import ValidationError
        if not self.image and not self.comment:
            raise ValidationError({'comment': 'Обязательно добавьте фото или напишите комментарий с причиной его отсутствия.'})
