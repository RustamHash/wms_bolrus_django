from pathlib import Path

from django.db import transaction

from apps.catalog.families import _base_name, critical_family, plain_title, with_pack
from apps.catalog.models import ExternalProduct, Product, ProductAlias, ProductGroup
from apps.catalog.parties import ensure_alias, upsert_external, warehouse_product
from apps.common.xlsx import read_rows


def stock_title(name: str) -> str:
    lowered = name.lower()
    if "морож" not in lowered:
        return critical_family(name)
    family = _base_name(name)
    if "морож" not in family.lower():
        family = f"{family} замороженная"
    return with_pack(name, plain_title(family))


def _catalog(source: Path | bytes) -> dict[str, dict[str, str]]:
    rows = read_rows(source, 1)
    header = rows.get(1) or {}
    columns = {name.strip(): column for column, name in header.items()}
    found = {}
    for number, cells in rows.items():
        if number == 1:
            continue
        code = (cells.get(columns.get("Код", ""), "") or "").strip()
        if not code:
            continue
        found[code] = {
            "name": (cells.get(columns.get("Наименование", ""), "") or "").strip(),
            "group": (cells.get(columns.get("В группе", ""), "") or "").strip(),
        }
    return found


def _used(source: Path | bytes) -> list[tuple[str, str]]:
    rows = []
    for cells in read_rows(source, 2).values():
        raw = (cells.get("A") or "").strip()
        code, _, name = raw.partition(",")
        code = code.strip()
        name = name.strip()
        if not code.startswith("НФ-") or not name:
            continue
        rows.append((code, name))
    return rows


def _group(name: str, groups: dict[str, ProductGroup]) -> ProductGroup | None:
    if not name:
        return None
    found = groups.get(name)
    if found is None:
        found = ProductGroup.objects.create(name=name)
        groups[name] = found
    return found


def _product(title: str, group: ProductGroup | None) -> Product:
    found = Product.objects.filter(name=title).first()
    if found is not None and group is not None and found.group_id not in (None, group.id):
        titled = f"{title} ({group.name})"[:120]
        found = Product.objects.filter(name=titled).first()
        title = titled
    if found is None:
        found = warehouse_product(title)
    if group is not None and found.group_id is None:
        found.group = group
        found.save(update_fields=["group"])
    return found


def _fill_known_groups(catalog: dict[str, dict[str, str]], used_codes: set[str], groups: dict[str, ProductGroup]) -> None:
    named: dict[int, set[str]] = {}
    aliases = ProductAlias.objects.filter(external_product__code__in=used_codes).select_related("external_product")
    for alias in aliases:
        group_name = (catalog.get(alias.external_product.code) or {}).get("group") or ""
        if group_name:
            named.setdefault(alias.product_id, set()).add(group_name)
    for product in Product.objects.filter(pk__in=named, group__isnull=True):
        names = named[product.id]
        if len(names) != 1:
            continue
        product.group = _group(next(iter(names)), groups)
        product.save(update_fields=["group"])


@transaction.atomic
def load_used_nomenclature(source: Path | bytes) -> dict[str, int]:
    catalog = _catalog(source)
    used = _used(source)
    groups = {item.name: item for item in ProductGroup.objects.all()}
    _fill_known_groups(catalog, {code for code, _name in used}, groups)
    created_external = 0
    created_product = 0
    created_alias = 0
    grouped = 0
    missing = 0
    products_before = set(Product.objects.values_list("id", flat=True))
    for code, name in used:
        card = catalog.get(code)
        if card is None:
            missing += 1
            continue
        existed = ExternalProduct.objects.filter(code=code).exists()
        external = upsert_external(code, card["name"] or name)
        if not existed:
            created_external += 1
        group = _group(card["group"], groups)
        if group is not None and external.group_id != group.id:
            external.group = group
            external.save(update_fields=["group"])
            grouped += 1
        if ProductAlias.objects.filter(external_product=external).exists():
            continue
        product = _product(stock_title(card["name"] or name), group)
        ensure_alias(external, product)
        created_alias += 1
        if product.id not in products_before:
            created_product += 1
            products_before.add(product.id)
    return {
        "used": len(used),
        "external": created_external,
        "products": created_product,
        "aliases": created_alias,
        "groups": grouped,
        "missing": missing,
    }

def assign_route_zones(apps, schema_editor):
    DeliveryZone = apps.get_model("catalog", "DeliveryZone")
    Consignee = apps.get_model("catalog", "Consignee")
    for zone_name, names in ROUTE_ZONES.items():
        zone, _created = DeliveryZone.objects.get_or_create(name=zone_name)
        for index, name in enumerate(names, start=1):
            Consignee.objects.filter(name=name).update(zone_id=zone.id, delivery_order=index)

def _clean(value: str, limit: int) -> str:
    return " ".join((value or "").split())[:limit]


def upsert_supplier(name: str, code: str = "") -> Supplier:
    name = _clean(name, 200)
    code = _clean(code, 32)
    if not name:
        raise WarehouseError("Укажите поставщика")
    found = Supplier.objects.filter(code=code).first() if code else None
    if found is None:
        found = Supplier.objects.filter(name=name).first()
    if found is None:
        return Supplier.objects.create(name=name, code=code or None)
    changed = []
    if found.name != name:
        found.name = name
        changed.append("name")
    if code and found.code != code:
        found.code = code
        changed.append("code")
    if changed:
        found.save(update_fields=changed)
    return found


def upsert_consignee(name: str, code: str = "", address: str = "") -> Consignee:

    if not name:

        if named is not None and (not code or not named.code or named.code == code):

    if found is None:

    return found


def load_consignee_addresses(source) -> dict:
    from apps.common.xlsx import read_rows

    rows = read_rows(source)
    header = rows.get(1) or {}
    columns = {value.strip(): key for key, value in header.items()}

    def cell(row: dict, title: str) -> str:
        key = columns.get(title)
        if not key:
            return ""
        return (row.get(key) or "").strip()

    def moment(row: dict) -> float:
        raw = cell(row, "Дата").replace(",", ".")
        try:
            return float(raw)
        except ValueError:
            return 0.0

    parties = {}
    buyers = {}
    for number, row in rows.items():
        if number == 1:
            continue
        buyer_name = cell(row, "Покупатель / Поставщик")
        buyer_code = cell(row, "Код Покупателя")
        if buyer_name or buyer_code:
            buyers[buyer_code or buyer_name] = (buyer_name, buyer_code)
        address = _clean(cell(row, "Адрес доставки"), 1000)
        if not address:
            continue
        code = cell(row, "Код Грузополучателя")
        name = cell(row, "Грузополучатель")
        if not code and not name:
            code = buyer_code
            name = buyer_name
        if not code and not name:
            continue
        record = (name, code, address, moment(row), number)
        key = code or name
        previous = parties.get(key)
        if previous is None or (record[3], record[4]) >= (previous[3], previous[4]):
            parties[key] = record

    created = 0
    updated = 0
    unchanged = 0
    for name, code, address, _moment, _number in parties.values():
        existed = Consignee.objects.filter(code=code).first() if code else None
        if existed is None:
            existed = Consignee.objects.filter(name=_clean(name, 200)).first()
        is_new = existed is None
        consignee = upsert_consignee(name, code)
        if is_new:
            created += 1
        if consignee.address == address:
            unchanged += 1
            continue
        consignee.address = address
        consignee.save(update_fields=["address"])
        updated += 1
    return {
        "consignees": len(parties),
        "created": created,
        "updated": updated,
        "unchanged": unchanged,
    }


def _next_product_code() -> str:
    numbers = []
    for code, _name in CRITICAL_PRODUCTS:
        if code.startswith("Т-") and code[2:].isdigit():
            numbers.append(int(code[2:]))
    for code in Product.objects.values_list("code", flat=True):
        if code.startswith("Т-") and code[2:].isdigit():
            numbers.append(int(code[2:]))
    return f"Т-{max(numbers, default=0) + 1:03d}"


def warehouse_product(family: str) -> Product:
    family = _clean(family, 120)
    found = Product.objects.filter(name=family).first()
    if found is not None:
        return found
    code = FAMILY_CODE.get(family) or _next_product_code()
    taken = Product.objects.filter(code=code).first()
    if taken is not None:
        return taken
    return Product.objects.create(name=family, code=code)


def upsert_external(code: str, name: str) -> ExternalProduct:

    found = ExternalProduct.objects.filter(code=code).first()

def _picked_grades(form) -> list[Grade]:
    raw_ids = []
    for raw in form.getlist("allowed_grade"):
        try:
            raw_ids.append(int(raw))
        except (TypeError, ValueError) as exc:
            raise WarehouseError("Выберите сортность") from exc
    grades = list(Grade.objects.filter(pk__in=set(raw_ids)).order_by("sort_order", "name"))
    if len(grades) != len(set(raw_ids)):
        raise WarehouseError("Выберите сортность")
    return grades


def assign_consignee_grades(consignee, grade_ids: list[int]) -> str:
    wanted = list(dict.fromkeys(grade_ids))
    grades = list(Grade.objects.filter(pk__in=wanted).order_by("sort_order", "name"))
    if len(grades) != len(wanted):
        raise WarehouseError("Выберите сортность")
    consignee.allowed_grades.set(grades)
    if not grades:
        return "любой сорт"
    return ", ".join(item.name for item in grades)


def assign_consignee_zone(consignee, name: str) -> str:
    name = " ".join((name or "").split())
    if len(name) > 120:
        raise WarehouseError("Название зоны длиннее 120 символов")
    if not name:
        if consignee.zone_id:
            consignee.zone = None
            consignee.save(update_fields=["zone"])
        return ""
    zone = DeliveryZone.objects.filter(name__iexact=name).first()
    if zone is None:
        try:
            zone = DeliveryZone.objects.create(name=name)
        except IntegrityError:
            zone = DeliveryZone.objects.get(name=name)
    if consignee.zone_id != zone.id:
        consignee.zone = zone
        consignee.save(update_fields=["zone"])
    return zone.name


def ensure_category_grade(name: str) -> Grade:
    grade = Grade.objects.filter(name=name).first()

    last = Grade.objects.order_by("-sort_order").values_list("sort_order", flat=True).first() or 0
    return Grade.objects.create(name=name, sort_order=last + 10)

def _code(form, label: str, limit: int = 32) -> str:
    value = " ".join(text(form, "code").split())
    if not value:
        raise WarehouseError(f"Укажите {label}")
    if len(value) > limit:
        raise WarehouseError(f"{label} длиннее {limit} символов")
    return value


def _save_warehouse_zone(item, form) -> None:
    name = " ".join(text(form, "name").split())
    code = _code(form, "код зоны")
    if not name:
        raise WarehouseError("Укажите название зоны")
    if len(name) > 120:
        raise WarehouseError("Название зоны длиннее 120 символов")
    barcode_value = " ".join(text(form, "barcode").split()) or code
    if len(barcode_value) > 64:
        raise WarehouseError("QR зоны длиннее 64 символов")
    others = WarehouseZone.objects.exclude(pk=item.pk or 0)
    if others.filter(code__iexact=code).exists():
        raise WarehouseError("Такой код зоны уже есть")
    if others.filter(name__iexact=name).exists():
        raise WarehouseError("Такая зона уже есть")
    if others.filter(barcode__iexact=barcode_value).exists():
        raise WarehouseError("Такой QR зоны уже есть")
    item.code = code
    item.name = name
    item.barcode = barcode_value
    item.save()


def _save_warehouse_cell(item, form) -> None:
    zone = WarehouseZone.objects.filter(pk=optional_int(form, "zone") or 0).first()
    if zone is None:

        raise WarehouseError("Название ячейки длиннее 120 символов")
    barcode_value = " ".join(text(form, "barcode").split()) or cell_address(zone.code, code)

    if WarehouseCell.objects.exclude(pk=item.pk or 0).filter(barcode__iexact=barcode_value).exists():
        raise WarehouseError("Такой QR ячейки уже есть")

    item.code = code

    item.barcode = barcode_value
    item.save()


def apply_master(kind: str, item, form) -> None:
    if kind == "warehouse-zones":
        _save_warehouse_zone(item, form)
        return
    if kind == "warehouse-cells":
        _save_warehouse_cell(item, form)
        return
    if kind == "printers":
        from apps.printing.services import apply_printer

        apply_printer(item, form)
        return
    if kind == "aliases":
        external = ExternalProduct.objects.filter(pk=optional_int(form, "external_product_id") or 0).first()
        product = Product.objects.filter(pk=optional_int(form, "product_id") or 0).first()
        if external is None or product is None:
            raise WarehouseError("Выберите номенклатуру 1С и номенклатуру склада")
        taken = ProductAlias.objects.filter(external_product=external).exclude(pk=item.pk or 0).first()
        if taken is not None:
            raise WarehouseError("Соответствие для этого кода уже есть")
        item.external_product = external
        item.product = product
        item.save()
        return
    name = text(form, "name")
    if not name:
        raise WarehouseError("Укажите название")
    item.name = name
    if kind in ("calibers", "grades", "categories"):
        item.sort_order = optional_int(form, "sort_order") or 0
    if kind == "categories":
        taken = ConsigneeCategory.objects.filter(name=item.name).exclude(pk=item.pk or 0).exists()
        if taken:
            raise WarehouseError("Такая категория уже есть")
        item.grade = ensure_category_grade(item.name)
    if kind == "products":
        item.code = text(form, "code")
        item.barcode = text(form, "barcode")
        if not item.code:
            raise WarehouseError("Укажите код")
        if not item.barcode:
            raise WarehouseError("Укажите штрихкод")
        if Product.objects.filter(code=item.code).exclude(pk=item.pk or 0).exists():
            raise WarehouseError("Такой код уже есть")
        if Product.objects.filter(barcode=item.barcode).exclude(pk=item.pk or 0).exists():
            raise WarehouseError("Такой штрихкод уже есть")
        group_id = optional_int(form, "group_id")
        if group_id is not None and not ProductGroup.objects.filter(pk=group_id).exists():
            raise WarehouseError("Выберите группу")
        item.group_id = group_id
    if kind == "externals":
        item.code = text(form, "code")
        if not item.code:
            raise WarehouseError("Укажите код")
        if ExternalProduct.objects.filter(code=item.code).exclude(pk=item.pk or 0).exists():
            raise WarehouseError("Такой код уже есть")
        group_id = optional_int(form, "group_id")
        if group_id is not None and not ProductGroup.objects.filter(pk=group_id).exists():
            raise WarehouseError("Выберите группу")
        item.group_id = group_id
    if kind == "suppliers":
        item.code = text(form, "code") or None
        if item.code and Supplier.objects.filter(code=item.code).exclude(pk=item.pk or 0).exists():
            raise WarehouseError("Такой код уже есть")
    if kind == "consignees":
        item.code = text(form, "code") or None
        if item.code and Consignee.objects.filter(code=item.code).exclude(pk=item.pk or 0).exists():
            raise WarehouseError("Такой код уже есть")
        item.address = text(form, "address")
        item.shipping_terms = text(form, "shipping_terms")
        zone_id = optional_int(form, "zone_id")
        if zone_id and not DeliveryZone.objects.filter(pk=zone_id).exists():
            raise WarehouseError("Выберите зону доставки")
        item.zone_id = zone_id
        delivery_order = optional_int(form, "delivery_order")
        if delivery_order is None:
            item.delivery_order = 0
        elif delivery_order < 0:
            raise WarehouseError("Очередность не может быть отрицательной")
        else:
            item.delivery_order = delivery_order
        category_id = optional_int(form, "category_id")
        if category_id and not ConsigneeCategory.objects.filter(pk=category_id).exists():
            raise WarehouseError("Выберите категорию")
        item.category_id = category_id
    if kind == "sellers":
        item.code = text(form, "code")
        if not item.code:
            raise WarehouseError("Укажите код продавца")
        item.requisites = text(form, "requisites")
    if kind == "packagings":
        raw = text(form, "tare_kg").replace(",", ".")
        if not raw:
            item.tare_kg = Decimal("0.000")
        else:
            try:
                tare = Decimal(raw).quantize(Decimal("0.001"))
            except InvalidOperation as exc:
                raise WarehouseError("Некорректный вес тары") from exc
            if tare < 0:
                raise WarehouseError("Вес тары не может быть отрицательным")
            item.tare_kg = tare
    item.save()
    if kind == "categories":
        from apps.catalog.grade_policy import fill_consignee_orders

        item.allowed_grades.set(_picked_grades(form))
        for consignee in Consignee.objects.filter(category=item).select_related("category").prefetch_related(
            "allowed_grades", "category__allowed_grades"
        ):
            fill_consignee_orders(consignee)
    if kind == "consignees":
        item.allowed_grades.set(_picked_grades(form))
        from apps.catalog.grade_policy import fill_consignee_orders

        fill_consignee_orders(item)
