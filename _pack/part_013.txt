class OnecOrders:
    """Прослойка: заказы поставщикам и покупателям из УНФ через OData."""

    def __init__(self, client: ODataClient | None = None):
        self.client = client or ODataClient.from_settings()

    def list_purchase_orders(
        self,
        *,
        top: int = 50,
        skip: int = 0,
        posted: bool | None = True,
        number: str | None = None,
        date_from: date | None = None,
        date_to: date | None = None,
        with_lines: bool = True,
        resolve: bool = True,
    ) -> list[OrderDTO]:
        rows = self.client.get(
            ENTITY_PURCHASE,
            filter=_filter(
                posted=posted,
                number=number,
                date_from=date_from,
                date_to=date_to,
                date_field="ДатаПоступления",
            ),
            select=PURCHASE_SELECT,
            orderby="ДатаПоступления desc",
            top=top,
            skip=skip,
        )
        states = None if resolve else catalogs.states_map(self.client, kind="purchase")
        return [
            self._map_purchase(row, with_lines=with_lines, resolve=resolve, states=states)
            for row in rows
        ]

    def get_purchase_order(self, ref_key: str, *, with_lines: bool = True) -> OrderDTO:
        try:
            row = self.client.get_one(ENTITY_PURCHASE, ref_key, select=PURCHASE_SELECT)
        except OnecNotFound as exc:
            raise OnecNotFound(f"Заказ поставщику {ref_key} не найден") from exc
        return self._map_purchase(row, with_lines=with_lines, resolve=True)

    def expand_purchase(self, stub: OrderDTO) -> OrderDTO:
        """Строки и справочники по уже загруженной шапке (без повторного GET документа)."""
        if not stub.raw:
            return self.get_purchase_order(stub.ref_key, with_lines=True)
        return self._map_purchase(stub.raw, with_lines=True, resolve=True)

    def find_purchase_by_number(self, number: str, *, with_lines: bool = True, resolve: bool = True) -> OrderDTO | None:
        rows = self.list_purchase_orders(
            top=1, posted=None, number=number, with_lines=with_lines, resolve=resolve
        )
        return rows[0] if rows else None

    def _map_purchase(
        self,
        row: dict[str, Any],
        *,
        with_lines: bool,
        resolve: bool = True,
        states: dict[str, str] | None = None,
    ) -> OrderDTO:
        party_key = catalogs.ref_guid(row, "Контрагент_Key", "Контрагент")
        org_key = catalogs.ref_guid(row, "Организация_Key", "Организация")
        state = catalogs.order_state_label(row, kind="purchase", client=self.client, states=states)
        if resolve:
            party = catalogs.party(self.client, party_key)
            organization = catalogs.organization(self.client, org_key)
        else:
            party = PartyRef(ref_key=party_key)
            organization = PartyRef(ref_key=org_key)
        dto = OrderDTO(
            kind="purchase",
            ref_key=str(row.get("Ref_Key") or ""),
            number=str(row.get("Number") or ""),
            date=_parse_datetime(row.get("Date")),
            posted=bool(row.get("Posted")),
            party=party,
            organization=organization,
            state=state,
            comment=str(row.get("Комментарий") or ""),
            receive_date=_parse_date(row.get("ДатаПоступления")),
            amount=_decimal(row.get("СуммаДокумента")),
            raw=row,
        )
        if with_lines and dto.ref_key:
            dto.lines = self._purchase_lines(dto.ref_key)
        return dto

    def _map_customer(
        self,
        row: dict[str, Any],
        *,
        with_lines: bool,
        resolve: bool = True,
        states: dict[str, str] | None = None,
    ) -> OrderDTO:
        party_key = catalogs.ref_guid(row, "Контрагент_Key", "Контрагент")
        org_key = catalogs.ref_guid(row, "Организация_Key", "Организация")
        consignee_key = catalogs.ref_guid(row, "Грузополучатель_Key", "Грузополучатель")
        state = catalogs.order_state_label(row, kind="customer", client=self.client, states=states)
        if resolve:
            party = catalogs.party(self.client, party_key)
            organization = catalogs.organization(self.client, org_key)
            consignee = catalogs.party(self.client, consignee_key) if consignee_key else PartyRef()
        else:
            party = PartyRef(ref_key=party_key)
            organization = PartyRef(ref_key=org_key)
            consignee = PartyRef(ref_key=consignee_key) if consignee_key else PartyRef()
        dto = OrderDTO(
            kind="customer",
            ref_key=str(row.get("Ref_Key") or ""),
            number=str(row.get("Number") or ""),
            date=_parse_datetime(row.get("Date")),
            posted=bool(row.get("Posted")),
            party=party,
            organization=organization,
            consignee=consignee,
            delivery_address=" ".join(str(row.get("АдресДоставки") or "").split()),
            state=state,
            comment=str(row.get("Комментарий") or ""),
            ship_date=_parse_date(row.get("ДатаОтгрузки")),
            amount=_decimal(row.get("СуммаДокумента")),
            raw=row,
        )
        if with_lines and dto.ref_key:
            dto.lines = self._customer_lines(dto.ref_key)
        return dto

    def _purchase_lines(self, ref_key: str) -> list[OrderLineDTO]:
        rows = self.client.get(
            ENTITY_PURCHASE_LINES,
            filter=f"Ref_Key eq guid'{ref_key}'",
            orderby="LineNumber",
        )
        return [self._map_line(row, product_fields=("Номенклатура_Key", "Номенклатура")) for row in rows]

    def _customer_lines(self, ref_key: str) -> list[OrderLineDTO]:
        rows = self.client.get(
            ENTITY_CUSTOMER_LINES,
            filter=f"Ref_Key eq guid'{ref_key}'",
            orderby="LineNumber",
        )
        return [self._map_line(row, product_fields=("Номенклатура", "Номенклатура_Key")) for row in rows]

    def _map_line(self, row: dict[str, Any], *, product_fields: tuple[str, ...]) -> OrderLineDTO:
        product_key = catalogs.ref_guid(row, *product_fields)
        unit_key = catalogs.ref_guid(row, "ЕдиницаИзмерения", "ЕдиницаИзмерения_Key")
        return OrderLineDTO(
            line_number=int(row.get("LineNumber") or 0),
            product=catalogs.product(self.client, product_key),
            qty=_decimal(row.get("Количество")) or Decimal("0"),
            unit=catalogs.unit_name(self.client, unit_key),
            content=str(row.get("Содержание") or ""),
            price=_decimal(row.get("Цена")),
            amount=_decimal(row.get("Сумма") or row.get("Всего")),
            weight=_decimal(row.get("Вес")),
            raw=row,
        )


def _filter(
    *,
    posted: bool | None,
    number: str | None,
    date_from: date | None,
    date_to: date | None,
    date_field: str = "Date",
) -> str | None:
    parts: list[str] = ["DeletionMark eq false"]
    if posted is True:
        parts.append("Posted eq true")
    elif posted is False:
        parts.append("Posted eq false")
    if number:
        safe = number.replace("'", "''")
        parts.append(f"Number eq '{safe}'")
    if date_from:
        parts.append(f"{date_field} ge datetime'{date_from.isoformat()}T00:00:00'")
    if date_to:
        nxt = date_to + timedelta(days=1)
        parts.append(f"{date_field} lt datetime'{nxt.isoformat()}T00:00:00'")
    return " and ".join(parts) if parts else None


def _parse_datetime(value: Any) -> datetime | None:
    if not value:
        return None
    if isinstance(value, datetime):
        return value
    text = str(value).replace("Z", "+00:00")
    try:
        return datetime.fromisoformat(text)
    except ValueError:
        return None


def _parse_date(value: Any) -> date | None:
    dt = _parse_datetime(value)
    return dt.date() if dt else None


def _decimal(value: Any) -> Decimal | None:
    if value is None or value == "":
        return None
    try:
        return Decimal(str(value))
    except (InvalidOperation, ValueError):
        return None

def _emit(report: ImportReport, message: str, on_log: LogFn | None = None) -> str:
    report.log.append(message)
    if on_log:
        on_log(message)
    return message

def _filter_new_headers(
    headers: list[OrderDTO],
    *,
    existing: set[str],
    report: ImportReport,
) -> list[OrderDTO]:
    """Оставляет только заказы, которых ещё нет в WMS; дубли пишет в report."""
    fresh = []
    for doc in headers:
        external = (doc.number or "").strip()
        if external and external in existing:
            report.duplicates.append(external)
            continue
        fresh.append(doc)
    return fresh


def _existing_purchase_numbers(numbers: list[str]) -> set[str]:
    cleaned = [item for item in numbers if item]
    if not cleaned:

    return set(

    )

        mode = "только новые (уже есть в WMS — пропуск)"
    yield _emit(report, f"Режим: {mode}", on_log)
    yield _emit(report, "Подключение к 1С…", on_log)
    try:
        api = api or OnecOrders()
        if by_number:
            found = api.find_purchase_by_number(number, with_lines=False, resolve=False)
            headers = [found] if found else []
        else:

            )

            )

    yield _emit(report, f"К импорту по статусу: {len(candidates)}", on_log)
    if skipped:
        yield _emit(report, f"Пропущено по статусу: {skipped}", on_log)
    if not allow_update and candidates:
        existing = _existing_purchase_numbers([(doc.number or "").strip() for doc in candidates])
        before = len(candidates)
        candidates = _filter_new_headers(candidates, existing=existing, report=report)
        skipped_existing = before - len(candidates)
        if skipped_existing:
            yield _emit(report, f"Уже есть в WMS, пропущено: {skipped_existing}", on_log)
        yield _emit(report, f"К загрузке строк (только новые): {len(candidates)}", on_log)
    externals, aliases = _catalogs()
    total = len(candidates)
    for index, header in enumerate(candidates, start=1):
        label = header.number or header.ref_key or "?"

            aliases=aliases,
            report=report,
            any_status=by_number,
        )
        if message:
            yield _emit(report, f"{message} — {index} из {total}", on_log)
    yield _emit(report, "Импорт завершён", on_log)

def delivery_period(date_from: date | None = None, date_to: date | None = None) -> tuple[date, date]:
    """Период поставки/отгрузки; по умолчанию — завтра."""
    tomorrow = date.today() + timedelta(days=1)
    start = date_from or date_to or tomorrow
    end = date_to or date_from or tomorrow
    if end < start:
        start, end = end, start
    return start, end

def _norm(value: str) -> str:
    value = (value or "").lower().replace("ё", "е")
    return re.sub(r"[^a-zа-я0-9]+", " ", value).strip()


def _match_seller(text: str) -> Seller | None:
    hay = _norm(text)
    if not hay:
        return None
    sellers = list(Seller.objects.all())
    for seller in sellers:
        name = _norm(seller.name)
        if len(name) >= 4 and name in hay:
            return seller
        if hay and name and (hay in name or name in hay):
            return seller
    for needle, key in SELLER_ALIASES:
        if needle not in hay:
            continue
        for seller in sellers:
            if key in _norm(seller.name):
                return seller
    return None


def _is_kg(unit: str) -> bool:
    value = (unit or "").strip().lower().replace(".", "")

def _line_dicts(
    doc: OrderDTO,
    *,
    externals: dict,
    aliases: dict,
    report: ImportReport,
    upsert_missing: bool,
) -> list[dict]:
    lines = []
    for item in doc.lines:
        name = item.product.name or item.content
        code = (item.product.code or "").strip()
        feedback = is_feedback_name(name)
        unit = (item.unit or "").strip() or "кг"
        if not _is_kg(unit) and not feedback:
            report.skipped_units.append({"order": doc.number, "name": name or "—", "unit": unit or "—"})
            continue
        qty = (item.qty or Decimal("0")).quantize(Decimal("0.001"))
        if qty <= 0:
            continue
        if not code:
            report.missing_codes.append({"order": doc.number, "code": "", "name": name or "нет кода"})
            continue
        card = externals.get(code)
        if card is None and (upsert_missing or feedback):
            card = upsert_external(code, name)
            externals[code] = card
        if card is None:
            report.missing_codes.append({"order": doc.number, "code": code, "name": name})
            continue
        ours = None if feedback else aliases.get(code)
        lines.append(
            {
                "product_id": ours.id if ours is not None else None,
                "external_product_id": card.id,
                "qty_kg": qty,
                "external_code": code,
                "external_name": name,
                "unit": "кг" if _is_kg(unit) else unit,
                "for_task": not feedback,
            }
        )
    return lines

        release_unpicked_plan(task)

    task = line.task

    left = as_kg(alloc.qty_kg) - fact

    if qty <= 0:

    if qty > left:

    if qty > line_left:

    if source is None:
        raise WarehouseError("Паллета не принята")

        pallet=alloc.pallet,
        source=source,
        user=user,

    )
    _mark_allocated(alloc.pallet)
    refresh_pick_status(task)
    return movement

                left = as_kg(kept.qty_kg) - _picked_on(kept.task_line_id, pallet_id)
                if left > 0:
                    have = as_kg(have - left)
            rooms[pallet_id] = have if have > 0 else Decimal("0.000")
        return rooms[pallet_id]

    created = 0
    planned_kg = Decimal("0.000")
    short = Decimal("0.000")
    for line in lines:
        fact = _picked_kg(line.id)
        unpicked_kept = Decimal("0.000")
        for alloc in PickAllocation.objects.filter(task_line=line):
            got = _picked_on(line.id, alloc.pallet_id)
            left = as_kg(alloc.qty_kg) - got
            if left > 0:
                unpicked_kept += left
        need = as_kg(line.qty_kg) - fact - unpicked_kept
        if need < 0:
            need = Decimal("0.000")
        ordered = sorted(
            candidates,
            key=lambda movement: (
                0 if movement.pallet.reserved_line_id == line.order_line_id else 1,
                movement.lot.received_on or date.min,
                movement.id,
            ),
        )
        for movement in ordered:
            if need <= 0:
                break
            pallet = movement.pallet
            if not _pallet_allowed(pallet, line, task) or not _fits_task(movement, line, task):
                continue
            have = room(pallet.id)
            if have <= 0:
                continue
            take = have if have < need else need
            existing = PickAllocation.objects.filter(task_line=line, pallet=pallet).first()
            if existing is None:
                PickAllocation.objects.create(task_line=line, pallet=pallet, qty_kg=take)
            else:
                existing.qty_kg = as_kg(existing.qty_kg) + take
                existing.save(update_fields=["qty_kg"])
            rooms[pallet.id] = as_kg(have - take)
            need = as_kg(need - take)
            planned_kg += take
            created += 1
        if need > 0:
            short += need
    refresh_pick_status(task)
    if created == 0 and planned_kg == 0:
        covered = all(_covered(line.id, _picked_kg(line.id)) >= as_kg(line.qty_kg) for line in lines)
        if covered:
            return "уже в плане"
        raise WarehouseError("Нет паллет под задание")
    text = f"{created} назн., {format_kg(planned_kg)} кг"
    if short > 0:
        text += f", не хватило {format_kg(short)} кг"
    return text


def pick_ready_task(task, user) -> str:
    """Проводит оставшийся план по назначениям."""
    _open_pick(task)
    ready = [item for item in allocation_rows(task) if item["left"] > 0]
    if not ready:
        raise WarehouseError("Нет назначений к отбору")
    picked = 0
    kg = Decimal("0.000")
    failed = []
    for item in ready:
        alloc = item["allocation"]
        try:
            movement = pick_allocation(allocation_id=alloc.id, user=user)
            picked += 1
            kg += as_kg(movement.qty_kg)
        except WarehouseError as exc:
            number = alloc.pallet.number or alloc.pallet_id
            failed.append(f"{number}: {exc}")
    if picked == 0:
        raise WarehouseError("; ".join(failed) or "Ничего не отобрано")
    text = f"{picked} назн., {format_kg(kg)} кг"
    if failed:
        text += ", не отобрано: " + "; ".join(failed)
    return text

@roles_required("warehouse", tsd=("pick",))
def task_pick(request, task_id):
    form = request.POST
    back = text(form, "next") or f"/tasks/{task_id}"

    def action():
        task = Task.objects.filter(pk=task_id).first()
        if task is None:
            raise WarehouseError("Задание не найдено")
        require_assigned(task, request.user)
        raw = text(form, "qty_kg")
        qty = parse_kg(raw, "Кг") if raw else None
        allocation_id = optional_int(form, "allocation_id")
        if not allocation_id:
            raise WarehouseError("Нет назначения отбора")
        return pick_allocation(allocation_id=allocation_id, user=request.user, qty_kg=qty)

    result, response = guarded(request, back, action)
    if response:
        return response
    return done(request, back, f"Отобрано {format_kg(result.qty_kg)} кг")

def _parse_date(raw: str) -> date | None:
    value = (raw or "").strip()
    if not value:
        return None
    for fmt in ("%d.%m.%Y", "%d.%m.%y"):
        try:
            return datetime.strptime(value, fmt).date()
        except ValueError:
            continue
    if re.fullmatch(r"\d+(\.\d+)?", value):
        return date(1899, 12, 30) + timedelta(days=int(float(value)))
    return None

def _seller(text: str) -> Seller | None:
    hay = _norm(text)
    sellers = list(Seller.objects.all())
    for seller in sellers:
        name = _norm(seller.name)
        if len(name) >= 4 and name in hay:
            return seller
        for token in re.findall(r"\d{10,12}", seller.requisites or ""):
            if token in text:
                return seller
    for needle, key in SELLER_ALIASES:
        if needle not in hay:
            continue
        for seller in sellers:
            if key in _norm(seller.name):
                return seller
    return None

def _supplier_files(key: int) -> list:
    token = f"№ НФ-{key} от"
    if not ORDERS_DIR.is_dir():
        return []
    return sorted(path for path in ORDERS_DIR.glob("Заказ поставщику*.xlsx") if token in path.name)


def _row(order: PurchaseOrder, external: str, party: str, changes: dict | None = None) -> dict:
    return {
        "id": order.id,
        "number": order.number,
        "external": external,
        "party": party,
        "lines": order.lines.count(),
        "summary": revision_summary(changes or {}),
        "changes": changes or {},
    }

    lines = []

        feedback = is_feedback_name(item["name"])
        if item["unit"] != "кг" and not feedback:
            report.skipped_units.append({"order": external, "name": item["name"], "unit": item["unit"] or "—"})
            continue

            qty = Decimal(item["qty"]).quantize(Decimal("0.001"))

            )
            continue
