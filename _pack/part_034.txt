  function readList(id) {

  function escapeHtml(text) {
    return String(text ?? "").replace(/[&<>"']/g, (char) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[char]));
  }
  const lists = {
    grouping: readList("group-state"),
    fields: readList("column-state"),
    filter: readList("pick-state"),
    sorting: readList("sort-state"),
    groups: readList("category-state"),
  };
  const catalogs = {
    grouping: readList("group-fields"),
    fields: readList("column-fields"),
    filter: readList("pick-fields"),
    sorting: readList("column-fields"),

  const bodies = {
    grouping: "grouping-body",
    fields: "fields-body",
    filter: "filter-body",
    sorting: "sorting-body",
    groups: "groups-body",

  const emptyText = {
    grouping: ["3", "Нет полей группировки"],
    fields: ["3", "Нет полей"],
    filter: ["5", "Нет условий отбора"],
    sorting: ["4", "Нет полей сортировки"],
    groups: ["3", "Нет групп"],

  let dragKind = null;
  let dragIndex = null;
  let suppressClick = false;

  function options(items, selected, valueKey) {
    return items.map((item) => {
      const value = item[valueKey || "key"];
      return `<option value="${escapeHtml(value)}"${value === selected ? " selected" : ""}>${escapeHtml(item.label)}</option>`;
    }).join("");
  }
  function paint(kind) {
    const body = document.getElementById(bodies[kind]);
    const rows = lists[kind];
    if (!rows.length) {
      const [span, text] = emptyText[kind];
      body.innerHTML = `<tr class="setup-empty"><td colspan="${span}">${text}</td></tr>`;
      return;
    }
    body.innerHTML = "";
    rows.forEach((item, index) => {
      const row = document.createElement("tr");
      row.dataset.index = String(index);
      const box = `<td class="grip" draggable="true" title="Перетащить">↕</td><td class="mark"><input type="checkbox" data-index="${index}"${item.on ? " checked" : ""}></td>`;
      if (kind === "filter") {
        const needsValue = item.op !== "is_empty" && item.op !== "is_not_empty";
        const inList = item.op === "in_list" || item.op === "not_in_list";
        const value = !needsValue
          ? "<span class=\"muted\">—</span>"
          : inList
            ? `<textarea data-value rows="3" placeholder="Значение (каждое с новой строки)">${escapeHtml(item.value || "")}</textarea>`
            : `<input type="text" data-value value="${escapeHtml(item.value || "")}" placeholder="Значение">`;
        row.innerHTML = `${box}<td><select data-field>${options(catalogs.filter, item.field)}</select></td><td><select data-op>${options(ops, item.op)}</select></td><td>${value}</td>`;
      } else if (kind === "sorting") {
        row.innerHTML = `${box}<td>${escapeHtml(item.label)}</td><td><select data-dir><option value="asc"${item.dir !== "desc" ? " selected" : ""}>По возрастанию</option><option value="desc"${item.dir === "desc" ? " selected" : ""}>По убыванию</option></select></td>`;
      } else if (kind === "groups") {
        row.innerHTML = `${box}<td>${escapeHtml(item.name)}</td>`;
      } else {
        row.innerHTML = `${box}<td>${escapeHtml(item.label)}</td>`;
      }
      const checkbox = row.querySelector('input[type="checkbox"]');
      checkbox.addEventListener("click", (event) => event.stopPropagation());
      checkbox.addEventListener("change", () => { lists[kind][index].on = checkbox.checked; });
      const field = row.querySelector("[data-field]");
      if (field) field.addEventListener("change", () => { lists[kind][index].field = field.value; });
      const op = row.querySelector("[data-op]");
      if (op) op.addEventListener("change", () => { lists[kind][index].op = op.value; paint(kind); });
      const value = row.querySelector("[data-value]");
      if (value) value.addEventListener("input", () => { lists[kind][index].value = value.value; });
      const dir = row.querySelector("[data-dir]");
      if (dir) dir.addEventListener("change", () => { lists[kind][index].dir = dir.value; });
      row.addEventListener("click", (event) => {
        if (suppressClick || event.target.closest("input, select, textarea, button, .grip")) return;
        checkbox.checked = !checkbox.checked;
        lists[kind][index].on = checkbox.checked;
      });
      const grip = row.querySelector(".grip");
      grip.addEventListener("dragstart", (event) => {
        dragKind = kind;
        dragIndex = index;
        event.dataTransfer.effectAllowed = "move";
        event.dataTransfer.setData("text/plain", kind + ":" + index);
        row.classList.add("dragging");
      });
      grip.addEventListener("dragend", () => {
        suppressClick = true;
        setTimeout(() => { suppressClick = false; }, 0);
        dragKind = null;
        dragIndex = null;
        body.querySelectorAll("tr").forEach((item) => item.classList.remove("dragging", "drop-before", "drop-after"));
      });
      row.addEventListener("dragover", (event) => {
        if (dragKind !== kind) return;
        event.preventDefault();
        event.dataTransfer.dropEffect = "move";
        const box = row.getBoundingClientRect();
        const after = event.clientY > box.top + box.height / 2;
        row.classList.toggle("drop-after", after);
        row.classList.toggle("drop-before", !after);
      });
      row.addEventListener("drop", (event) => {
        if (dragKind !== kind || dragIndex === null) return;
        event.preventDefault();
        const after = row.classList.contains("drop-after");
        let to = index + (after ? 1 : 0);
        const from = dragIndex;
        if (from < to) to -= 1;
        dragKind = null;
        dragIndex = null;
        if (from === to) return;
        const [moved] = lists[kind].splice(from, 1);
        lists[kind].splice(to, 0, moved);
        paint(kind);
      });
      body.appendChild(row);
    });
  }

  function chooseField(title, catalog, taken, onPick) {
    const modal = document.createElement("div");
    modal.className = "setup-modal";
    modal.innerHTML = `<div class="card"><h3>${escapeHtml(title)}</h3><div class="setup-pick"><table class="grouping-table" data-setup><thead><tr><th>Поле</th></tr></thead><tbody>${catalog.map((field, index) => `<tr data-index="${index}"><td>${escapeHtml(field.label)}</td></tr>`).join("")}</tbody></table></div><div class="actions"><button type="button" class="slim alt" data-close>Отмена</button></div></div>`;
    document.body.appendChild(modal);
    modal.querySelectorAll("tbody tr").forEach((row) => {
      row.addEventListener("mouseenter", () => { row.style.background = "#e7f0ea"; });
      row.addEventListener("mouseleave", () => { row.style.background = ""; });
      row.addEventListener("click", () => {
        const field = catalog[Number(row.dataset.index)];
        if (!taken || !taken.has(field.key)) onPick(field);
        modal.remove();
      });
    });
    modal.querySelector("[data-close]").addEventListener("click", () => modal.remove());
    modal.addEventListener("click", (event) => { if (event.target === modal) modal.remove(); });

  const addTitle = { grouping: "Выберите поле для группировки", filter: "Выберите поле для отбора", fields: "Выберите поле", sorting: "Выберите поле для сортировки" };

        else lists[kind].push({ key: field.key, label: field.label, on: true });

      });
    });
  });
  function marked(kind) {
    return [...document.querySelectorAll("#" + bodies[kind] + ' input[type="checkbox"]:checked')].map((box) => Number(box.dataset.index));
  }

        if (next < 0 || next >= rows.length || indexes.includes(next)) return;

    });
  });

<script>
(function () {
  const table = document.getElementById("route-stops");
  if (!table || !table.dataset.reorder) return;
  const tbody = table.querySelector("tbody");
  const tokenInput = document.querySelector("[name=csrfmiddlewaretoken]");
  let dragged = null;
  let moved = false;

  function saveOrder() {
    const orderIds = Array.from(tbody.querySelectorAll("tr[data-order-id]")).map((row) => Number(row.dataset.orderId));
    fetch(table.dataset.reorder, {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "X-CSRFToken": tokenInput ? tokenInput.value : "",
      },
      body: JSON.stringify({ order_ids: orderIds }),
    })
      .then((response) => response.json().then((data) => ({ ok: response.ok, data })))
      .then(({ ok, data }) => {
        if (!ok || !data.ok) {
          window.alert((data && data.error) || "Не удалось сохранить порядок");
          window.location.reload();
          return;
        }
        tbody.querySelectorAll("tr[data-order-id] .pos").forEach((cell, index) => {
          cell.textContent = String(index + 1);
        });
      })
      .catch(() => {
        window.alert("Не удалось сохранить порядок");
        window.location.reload();
      });
  }

  tbody.addEventListener("dragstart", (event) => {
    const row = event.target.closest("tr[data-order-id]");
    if (!row || event.target.closest("a, button, input, select, textarea")) {
      event.preventDefault();
      return;
    }
    dragged = row;
    moved = false;
    row.classList.add("dragging");
    event.dataTransfer.effectAllowed = "move";
    event.dataTransfer.setData("text/plain", row.dataset.orderId);
  });
  tbody.addEventListener("dragend", () => {
    if (dragged) dragged.classList.remove("dragging");
    dragged = null;
    if (moved) saveOrder();
    moved = false;
  });
  tbody.addEventListener("dragover", (event) => {
    if (!dragged) return;
    event.preventDefault();
    const row = event.target.closest("tr[data-order-id]");
    if (!row || row === dragged) return;
    const rows = Array.from(tbody.querySelectorAll("tr[data-order-id]"));
    const from = rows.indexOf(dragged);
    const to = rows.indexOf(row);
    if (from < 0 || to < 0 || from === to) return;
    moved = true;
    if (from < to) tbody.insertBefore(dragged, row.nextSibling);
    else tbody.insertBefore(dragged, row);
  });
  tbody.addEventListener("drop", (event) => {
    event.preventDefault();
  });
})();
</script>
