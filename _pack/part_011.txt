from datetime import date
from decimal import Decimal

from django.db.models import Count, Prefetch, Sum

from apps.logistics.models import Route, RouteOrder

def _routes_for_delivery_day(day: date):
    """Все маршруты с датой доставки (рейса) = day."""
    return (
        Route.objects.filter(route_date=day)
        .exclude(status=Route.Status.CANCELLED)
        .select_related("driver")
    )


def schedule_context(day: date) -> dict:
    """График по дате доставки: список, кто во сколько грузится."""
    routes = list(
        _routes_for_delivery_day(day).annotate(
            order_count=Count("route_orders", distinct=True),
            weight=Sum("route_orders__order__lines__qty_kg"),
        )
    )
    items = []
    for route in routes:
        items.append(
            {
                "id": route.id,
                "name": route.name or f"Маршрут {route.id}",
                "driver": route.driver.full_name if route.driver_id else "Без водителя",
                "when": _load_stamp(route.planned_load_at),
                "timed": bool(route.planned_load_at),
                "order_count": route.order_count or 0,
                "weight": route.weight or Decimal("0"),
                "sort": (_sort_key_plan(route.planned_load_at), route.id),
            }
        )
    items.sort(key=lambda item: item["sort"])
    for item in items:
        del item["sort"]
    return {"day": day, "items": items}


def _tt_key(order) -> str:
    if not order.consignee_id:
        return f"order:{order.id}"
    address = (order.consignee.address or "").strip()
    if address:
        return address.casefold()
    return f"consignee:{order.consignee_id}"


def _clock(value) -> str:
    if not value:
        return ""
    local = load_local(value)
    return f"{local.hour:02d}:{local.minute:02d}"


def _load_stamp(value) -> str:
    """Дата и время плана погрузки, всегда вместе."""
    if not value:
        return ""
    local = load_local(value)
    return f"{local:%d.%m.%Y} {local.hour:02d}:{local.minute:02d}"


def _fact_label(value, delivery_day: date) -> str:
    if not value:
        return ""
    local = load_local(value)
    clock = f"{local.hour:02d}:{local.minute:02d}"
    if local.date() != delivery_day:
        return f"{local:%d.%m} {clock}"
    return clock


def _sort_key_plan(value) -> tuple:
    """Полная дата+время плана, чтобы вечерняя погрузка шла раньше утренней."""
    if not value:
        return (1, date.max, 0, 0)
    local = load_local(value)
    return (0, local.date(), local.hour, local.minute)


def schedule_print_tree(day: date) -> dict:
    """
    Все маршруты с датой доставки = day, по строке на маршрут.
    Сортировка по плану погрузки. План — дата и время, как в графике на экране.
    Строки: план | факт | водитель | маршрут | комментарий | вес | ТТ.
    """
    routes = list(
        _routes_for_delivery_day(day)
        .prefetch_related(
            Prefetch(
                "route_orders",
                queryset=RouteOrder.objects.select_related(
                    "order__consignee",
                    "order__consignee",
                ).order_by("position", "id"),
            )
        )
        .order_by("planned_load_at", "id")
    )
    all_orders = [stop.order for route in routes for stop in route.route_orders.all()]
    weights = order_weights(all_orders)

    prepared = []
    for route in routes:
        driver_name = route.driver.full_name if route.driver_id else "Без водителя"
        route_name = route.name or f"Маршрут {route.id}"
        route_weight = Decimal("0")
        tt: set[str] = set()
        for stop in route.route_orders.all():
            order = stop.order
            route_weight += weights.get(order.id) or Decimal("0")
            tt.add(_tt_key(order))
        prepared.append(
            {
                "sort": (
                    _sort_key_plan(route.planned_load_at),
                    driver_name.casefold(),
                    route_name.casefold(),
                    route.id,
                ),
                "plan": _load_stamp(route.planned_load_at),
                "fact": _fact_label(route.actual_load_at, day),
                "label": driver_name,
                "route": route_name,
                "notes": (route.notes or "").strip(),
                "weight": route_weight,
                "tt": tt,
            }
        )
    prepared.sort(key=lambda item: item["sort"])

    table_rows = []
    grand_weight = Decimal("0")
    grand_tt: set[str] = set()
    for item in prepared:
        table_rows.append(
            {
                "kind": "driver",
                "plan": item["plan"],
                "fact": item["fact"],
                "label": item["label"],
                "route": item["route"],
                "notes": item["notes"],
                "weight": item["weight"],
                "tt_count": len(item["tt"]),
            }
        )
        grand_weight += item["weight"]
        grand_tt |= item["tt"]

    return {
        "day": day,
        "rows": table_rows,
        "total_weight": grand_weight,
        "total_tt": len(grand_tt),
        "route_count": len(routes),
    }

def _stamp(obj, field: str, label: str) -> None:
    if getattr(obj, field) is not None:
        raise WarehouseError(f"{label} уже зафиксировано")
    setattr(obj, field, timezone.now())
    obj.save(update_fields=[field])


def parse_shift_date(raw: str) -> date:
    value = (raw or "").strip()
    if not value:
        return timezone.localdate()
    try:
        return date.fromisoformat(value)
    except ValueError as exc:
        raise WarehouseError("Некорректная дата") from exc


def parse_shift_period(raw_from: str, raw_to: str, raw_date: str = "") -> tuple[date, date, str]:
    today = timezone.localdate()
    start_raw = (raw_from or "").strip()
    end_raw = (raw_to or "").strip()
    single = (raw_date or "").strip()
    if not start_raw and not end_raw and single:
        start_raw = end_raw = single
    if not start_raw and not end_raw:
        return today, today, "today"
    start = parse_shift_date(start_raw or end_raw)
    end = parse_shift_date(end_raw or start_raw)
    if end < start:
        raise WarehouseError("Дата окончания раньше начала")
    preset = ""
    if start == end:
        if start == today:
            preset = "today"
        elif start == today - timedelta(days=1):
            preset = "yesterday"
        elif start == today + timedelta(days=1):
            preset = "tomorrow"
    return start, end, preset


def task_status_label(tasks, work: str) -> str:
    if not tasks:
        return "Нет задания"
    statuses = {task.status for task in tasks}
    if work in statuses:
        return TASK_STATUS[work]
    if "open" in statuses:
        return TASK_STATUS["open"]
    if statuses <= {"closed", "picked"}:
        if statuses == {"picked"}:
            return TASK_STATUS["picked"]
        return TASK_STATUS["closed"]
    status = next(iter(statuses))
    return TASK_STATUS.get(status, status)


def shift_orders(start: date, end: date) -> list[dict]:
    orders = list(
        Order.objects.filter(ship_date__gte=start, ship_date__lte=end)
        .select_related("consignee", "consignee__zone")
        .prefetch_related(
            Prefetch(
                "route_orders",
                queryset=RouteOrder.objects.filter(route__status__in=Route.ASSIGNED_STATUSES).select_related("route"),
            ),
            Prefetch("tasks", queryset=Task.objects.filter(kind="pick")),
        )
        .order_by("ship_date", "id")
    )
    weights = order_weights(orders)
    rows = []
    for order in orders:
        names = [link.route.name or f"Маршрут {link.route_id}" for link in order.route_orders.all()]
        rows.append(
            {
                "id": order.id,
                "date": order.ship_date,
                "number": order.number or order.external_number or f"#{order.pk}",
                "consignee": order.consignee.name if order.consignee_id else "",
                "zone": order.consignee.zone.name if order.consignee_id and order.consignee.zone_id else "",
                "weight": weights.get(order.id, Decimal("0")),
                "plans": [link.route.planned_load_at for link in order.route_orders.all() if link.route.planned_load_at],
                "route": ", ".join(names),
                "pick": task_status_label(list(order.tasks.all()), "created"),
            }
        )
    return rows

def shift_routes(start: date, end: date) -> list[Route]:
    routes = list(
        Route.objects.filter(route_date__gte=start, route_date__lte=end)
        .exclude(status=Route.Status.CANCELLED)
        .select_related("driver")
        .annotate(

def dock_route(route: Route) -> Route:
    if route.status not in Route.LIVE_STATUSES:
        raise WarehouseError("Поставить к отгрузке можно черновик или запланированный маршрут")
    route.status = Route.Status.DOCKED
    if route.docked_at is None:
        route.docked_at = timezone.now()
    route.save()
    return route


def ship_route(route: Route) -> Route:
    if route.status != Route.Status.DOCKED:
        raise WarehouseError("Отгрузить можно только маршрут, поставленный к отгрузке")
    route.status = Route.Status.COMPLETED
    if route.actual_load_at is None:
        route.actual_load_at = timezone.now()
    route.save()
    return route

def revert_route(route: Route) -> Route:
    if route.status == Route.Status.DOCKED:
        route.status = Route.Status.PLANNED
        route.docked_at = None
        route.save()
        return route
    if route.status == Route.Status.COMPLETED:
        route.status = Route.Status.DOCKED
        route.actual_load_at = None
        route.save()
        return route
    raise WarehouseError("Этот маршрут нельзя вернуть")

def _posted_load_times(request, ship_date):
    """Время загрузки по водителю. Пустое поле — маршрут без плана."""
    load_day = parse_load_date(text(request.POST, "planned_load_date")) or ship_date
    times = {}
    for key, raw in request.POST.items():
        if not key.startswith("load_time_"):
            continue
        value = (raw or "").strip()
        if not value:
            continue
        try:
            driver_id = int(key.removeprefix("load_time_"))
        except ValueError as exc:
            raise WarehouseError("Некорректное время загрузки") from exc
        times[driver_id] = parse_load_datetime(load_day, value)
    return times


@roles_required("office")
def routing_create(request):
    def action():
        ship_date = _ship_date(request)
        load_times = _posted_load_times(request, ship_date)
        rows, assignments, _total = routing_rows(ship_date, request.session)
        created = create_routes_from_zone_assignments(ship_date=ship_date, zone_rows=rows)
        warnings = []
        for route in created:
            load_at = load_times.get(route.driver_id or 0)
            if load_at is None:
                continue
            warning = set_planned_load_at(route, load_at)
            if warning:
                warnings.append(warning)
        order_ids, zone_ids = consumed_for_create(rows)
        present = {row["zone_id"] for row in rows}
        stale_zones = [zone_id for zone_id in assignments if zone_id not in present]
        clear_session_assignments(request.session, ship_date, list(zone_ids) + stale_zones)
        clear_session_order_assignments(request.session, ship_date, order_ids)
        return ship_date, created, warnings

    result, response = guarded(request, "/logistics/routing", action)
    if response:
        return response
    ship_date, created, warnings = result
    messages.success(request, f"Создано маршрутов: {len(created)}")
    for warning in warnings:
        messages.warning(request, warning)
    return redirect(f"/logistics/routing?date={ship_date.isoformat()}")

def is_guid(value: str) -> bool:
    return bool(value) and value != EMPTY_GUID and bool(_GUID_RE.match(value))


def ref_guid(row: dict[str, Any], *names: str) -> str:
    for name in names:
        value = row.get(name)
        if isinstance(value, str) and value and value != EMPTY_GUID:
            return value
        if isinstance(value, dict):
            nested = value.get("Ref_Key") or value.get("Key")
            if isinstance(nested, str) and nested and nested != EMPTY_GUID:

    return ""

def states_map(client: ODataClient, *, kind: str) -> dict[str, str]:
    """Все состояния заказов одним запросом: Ref_Key → наименование."""
    entity = STATE_PURCHASE if kind == "purchase" else STATE_CUSTOMER
    cache_key = f"states_map:{entity}"
    cached = client._cache.get(cache_key)
    if isinstance(cached, dict) and cached.get("__map__"):
        return cached["__map__"]
    rows = client.get(entity, select="Ref_Key,Description")
    mapping = {
        str(row.get("Ref_Key") or ""): str(row.get("Description") or "")
        for row in rows
        if row.get("Ref_Key")
    }
    client._cache[cache_key] = {"__map__": mapping}
    return mapping

) -> str:

        if states is not None:

    value = row.get("СостояниеЗаказа")

    if client is not None and lookup:

    """Низкоуровневый клиент стандартного интерфейса OData 1С."""

    def __init__(
        self,
        base_url: str,
        username: str,
        password: str,
        *,
        timeout: float = 60.0,
    ):
        self.base_url = base_url.rstrip("/")
        self.username = username
        self.password = password
        self.timeout = timeout
        token = base64.b64encode(f"{username}:{password}".encode("utf-8")).decode("ascii")
        self._auth = f"Basic {token}"
        self._cache: dict[str, dict[str, Any]] = {}

    @classmethod
    def from_settings(cls) -> ODataClient:
        base = getattr(settings, "ONEC_ODATA_URL", "") or ""
        user = getattr(settings, "ONEC_ODATA_USER", "") or ""
        password = getattr(settings, "ONEC_ODATA_PASSWORD", "") or ""
        timeout = float(getattr(settings, "ONEC_ODATA_TIMEOUT", 60) or 60)
        if not base or not user or not password:
            raise OnecConfigError(
                "Задайте ONEC_ODATA_URL, ONEC_ODATA_USER и ONEC_ODATA_PASSWORD в окружении"
            )
        return cls(base, user, password, timeout=timeout)

    def clear_cache(self) -> None:
        self._cache.clear()

    def get(
        self,
        entity: str,
        *,
        filter: str | None = None,
        select: str | None = None,
        orderby: str | None = None,
        top: int | None = None,
        skip: int | None = None,
        format: str = "json",
    ) -> list[dict[str, Any]]:
        params: dict[str, str] = {"$format": format}
        if filter:
            params["$filter"] = filter
        if select:
            params["$select"] = select
        if orderby:
            params["$orderby"] = orderby
        if top is not None:
            params["$top"] = str(top)
        if skip is not None:
            params["$skip"] = str(skip)
        payload = self._request("GET", entity, params=params)
        if isinstance(payload, dict) and "value" in payload:
            return list(payload["value"])
        raise OnecRequestError(f"Ожидался список в {entity}", body=str(payload)[:500])

    def get_one(self, entity: str, guid: str, *, select: str | None = None) -> dict[str, Any]:
        if not guid or guid == EMPTY_GUID:
            raise OnecNotFound(f"Пустой ключ для {entity}")
        path = f"{entity}(guid'{guid}')"
        params: dict[str, str] = {"$format": "json"}
        if select:
            params["$select"] = select
        payload = self._request("GET", path, params=params)
        if not isinstance(payload, dict):
            raise OnecRequestError(f"Ожидался объект {entity}", body=str(payload)[:500])
        return payload

    def get_cached(self, entity: str, guid: str, *, select: str | None = None) -> dict[str, Any] | None:
        if not guid or guid == EMPTY_GUID:
            return None
        key = f"{entity}:{guid}:{select or ''}"
        if key not in self._cache:
            try:
                self._cache[key] = self.get_one(entity, guid, select=select)
            except (OnecNotFound, OnecRequestError):
                self._cache[key] = {}
        row = self._cache[key]
        return row or None

    def _request(self, method: str, path: str, *, params: dict[str, str] | None = None) -> Any:
        query = urllib.parse.urlencode(params or {}, quote_via=urllib.parse.quote)
        encoded_path = urllib.parse.quote(path.lstrip("/"), safe="/()'=,_-$:")
        url = f"{self.base_url}/{encoded_path}"
        if query:
            url = f"{url}?{query}"
        request = urllib.request.Request(
            url,
            method=method,
            headers={
                "Authorization": self._auth,
                "Accept": "application/json",
            },
        )
        try:
            with urllib.request.urlopen(request, timeout=self.timeout) as response:
                raw = response.read().decode("utf-8")
        except urllib.error.HTTPError as exc:
            body = exc.read().decode("utf-8", errors="replace")
            if exc.code == 401:
                raise OnecAuthError(
                    "OData: неверный логин или пароль (сквозная авторизация Scloud)"
                ) from exc
            if exc.code == 404:
                raise OnecNotFound(f"OData: не найдено {path}") from exc
            message = _odata_message(body) or f"OData HTTP {exc.code}"
            raise OnecRequestError(message, status=exc.code, body=body[:1000]) from exc
        except urllib.error.URLError as exc:
            raise OnecRequestError(f"OData недоступен: {exc.reason}") from exc
        if not raw:
            return {}
        return json.loads(raw)


def _odata_message(body: str) -> str:
    try:
        data = json.loads(body)
        err = data.get("odata.error") or data.get("error") or {}
        msg = err.get("message")
        if isinstance(msg, dict):
            return str(msg.get("value") or "")
        if isinstance(msg, str):
            return msg
    except (json.JSONDecodeError, AttributeError, TypeError):
        return ""
    return ""

        parser.add_argument("--date-from", default="", help="YYYY-MM-DD; по умолчанию завтра")

        user = self._user(options["user"])
        update = not options["no_update"]
        if options["tomorrow"]:
            date_from, date_to = delivery_period()
        else:
            date_from, date_to = delivery_period(
                _parse_date(options["date_from"]),
                _parse_date(options["date_to"]),
            )

        top = options["top"]

                report = sync_purchase_orders(
                    user, update=update, top=top, date_from=date_from, date_to=date_to
                )
                self._print("Заказы поставщикам (дата поступления)", report)
            if kind in ("customer", "both"):
                report = sync_customer_orders(
                    user, update=update, top=top, date_from=date_from, date_to=date_to
                )
                self._print("Заказы покупателей (дата отгрузки)", report)
        except OnecError as exc:
            raise CommandError(str(exc)) from exc

    def _user(self, username: str):
        User = get_user_model()
        if username:
            user = User.objects.filter(login=username).first()
            if user is None:
                raise CommandError(f"Пользователь WMS «{username}» не найден")
            return user
        user = User.objects.filter(role="admin").order_by("id").first()
        if user is None:
            user = User.objects.order_by("id").first()
        if user is None:
            raise CommandError("В WMS нет пользователей")
        return user

    def _print(self, title: str, report):
        self.stdout.write(self.style.MIGRATE_HEADING(title))
        self.stdout.write(
            f"  создано={len(report.created)} обновлено={len(report.updated)} "
            f"без изменений={len(report.unchanged)} пропущено статусов={report.other_status}"
        )
        if report.duplicates:
            self.stdout.write(f"  уже были (без update): {len(report.duplicates)}")
        if report.closed:
            self.stdout.write(f"  не обновлены (закрыты/приняты): {len(report.closed)}")
        if report.blocked:
            self.stdout.write(f"  блокировки: {len(report.blocked)}")
        if report.empty:
            self.stdout.write(f"  пустые: {', '.join(report.empty[:10])}")
        if report.missing_files:
            for item in report.missing_files[:10]:
                self.stdout.write(self.style.WARNING(f"  {item}"))
        if report.missing_codes:
            self.stdout.write(f"  нет кодов: {len(report.missing_codes)}")
        if report.skipped_units:
            self.stdout.write(f"  не кг: {len(report.skipped_units)}")


def _parse_date(value: str) -> date | None:
    text = (value or "").strip()

    return datetime.strptime(text, "%Y-%m-%d").date()

def nomenclature_rows(client: ODataClient | None = None) -> list[dict]:

    rows = _pages(client, "Catalog_Номенклатура", NOMENCLATURE_SELECT)
    units = {

        for row in _pages(client, "Catalog_КлассификаторЕдиницИзмерения", UNIT_SELECT)
    }

    report = []

        path = _group_path(row, by_key)
        unit_key = str(row.get("ЕдиницаИзмерения_Key") or "")
        report.append(
            {
                "code": str(row.get("Code") or ""),
                "name": str(row.get("Description") or ""),
                "article": str(row.get("Артикул") or ""),
                "group": path[-1] if path else "",
                "path": " / ".join(path),
                "unit": units.get(unit_key, ""),
                "folder": bool(row.get("IsFolder")),
                "deleted": bool(row.get("DeletionMark")),
            }
        )
    report.sort(key=lambda item: (item["path"], item["folder"] is False, item["code"], item["name"]))
    return report


def _pages(client: ODataClient, entity: str, select: str) -> list[dict]:
    found: list[dict] = []
    seen: set[str] = set()
    skip = 0
    while skip < 200_000:
        try:
            page = client.get(entity, select=select, orderby="Code", top=PAGE, skip=skip)
        except OnecRequestError:
            if skip:
                break
            page = client.get(entity, select=select, top=PAGE, skip=skip)
        if not page:
            break
        first = str(page[0].get("Ref_Key") or "")
        if first and first in seen:
            break
        found.extend(page)
        for row in page:
            key = str(row.get("Ref_Key") or "")
            if key:
                seen.add(key)
        if len(page) < PAGE:
            break
        skip += PAGE
    return found
