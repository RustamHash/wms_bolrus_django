#!/usr/bin/env bash
# Первый запуск на чистом/частично подготовленном Ubuntu.
# Запуск: sudo bash /tmp/wms-bootstrap/deploy/bootstrap-server.sh
set -euo pipefail

APP_DIR="${WMS_APP_DIR:-/opt/wms}"
SRC_DIR="${WMS_SRC_DIR:-/tmp/wms-bootstrap}"
DB_NAME="${WMS_DB_NAME:-wms_django}"
DB_USER="${WMS_DB_USER:-wms}"

if [[ "$(id -u)" -ne 0 ]]; then
  echo "Запустите от root: sudo bash $0" >&2
  exit 1
fi

if [[ ! -f "$SRC_DIR/manage.py" ]]; then
  echo "Нет исходников в $SRC_DIR. Сначала скопируйте репозиторий туда." >&2
  exit 1
fi

echo "==> Пакеты"
export DEBIAN_FRONTEND=noninteractive
apt-get update -y
apt-get install -y \
  git curl ca-certificates rsync ufw \
  python3 python3-venv python3-pip \
  postgresql postgresql-contrib \
  docker.io docker-compose-v2 \
  libreoffice-writer \
  fonts-liberation fonts-crosextra-carlito fonts-crosextra-caladea fonts-dejavu-core \
  cups-client

systemctl enable --now postgresql docker
install -m 644 "$SRC_DIR/deploy/99-wms-fonts.conf" /etc/fonts/conf.d/99-wms-fonts.conf
fc-cache -f || true

echo "==> Firewall"
ufw allow OpenSSH
ufw allow 80/tcp
ufw allow 443/tcp
ufw --force enable || true

echo "==> Пользователь wms"
id -u wms >/dev/null 2>&1 || useradd -r -m -d "$APP_DIR" -s /bin/bash wms
mkdir -p "$APP_DIR"
chown wms:wms "$APP_DIR"

echo "==> PostgreSQL"
if ! sudo -u postgres psql -tAc "SELECT 1 FROM pg_roles WHERE rolname='${DB_USER}'" | grep -q 1; then
  DB_PASS="$(openssl rand -base64 24 | tr -d '/+=' | head -c 24)"
  sudo -u postgres psql -v ON_ERROR_STOP=1 \
    -c "CREATE USER ${DB_USER} WITH PASSWORD '${DB_PASS}';"
  echo "Создан пользователь БД ${DB_USER}, пароль сохранён в ${APP_DIR}/.env"
else
  DB_PASS=""
  echo "Пользователь БД ${DB_USER} уже есть"
fi

if ! sudo -u postgres psql -tAc "SELECT 1 FROM pg_database WHERE datname='${DB_NAME}'" | grep -q 1; then
  if [[ -z "$DB_PASS" ]]; then
    echo "Базы ${DB_NAME} нет, а пароль пользователя неизвестен." >&2
    echo "Задайте пароль вручную и перезапустите, либо удалите роль ${DB_USER}." >&2
    exit 1
  fi
  sudo -u postgres psql -v ON_ERROR_STOP=1 -c "CREATE DATABASE ${DB_NAME} OWNER ${DB_USER};"
  sudo -u postgres psql -d "$DB_NAME" -c "GRANT ALL ON SCHEMA public TO ${DB_USER};" || true
fi

echo "==> Код → ${APP_DIR}"
rsync -a --delete \
  --exclude '.git/' \
  --exclude '.venv/' \
  --exclude '.env' \
  --exclude 'pgdata/' \
  --exclude 'staticfiles/' \
  --exclude '__pycache__/' \
  --exclude '*.pyc' \
  "$SRC_DIR"/ "$APP_DIR"/
chown -R wms:wms "$APP_DIR"

echo "==> .env"
if [[ ! -f "$APP_DIR/.env" ]]; then
  SECRET="$(openssl rand -base64 48 | tr -d '/+=' | head -c 64)"
  if [[ -z "$DB_PASS" ]]; then
    echo "Нет ${APP_DIR}/.env и неизвестен пароль БД — создайте .env вручную." >&2
    exit 1
  fi
  cat >"$APP_DIR/.env" <<EOF
DATABASE_URL=postgresql://${DB_USER}:${DB_PASS}@127.0.0.1:5432/${DB_NAME}
SECRET_KEY=${SECRET}
DEBUG=0
CSRF_TRUSTED_ORIGINS=https://wms.bolrus.ru
EOF
  chown wms:wms "$APP_DIR/.env"
  chmod 600 "$APP_DIR/.env"
  echo "Создан ${APP_DIR}/.env"
else
  echo "Оставляем существующий ${APP_DIR}/.env"
  # Допишем CSRF/DEBUG, если пусто
  grep -q '^DEBUG=' "$APP_DIR/.env" || echo 'DEBUG=0' >>"$APP_DIR/.env"
  grep -q '^CSRF_TRUSTED_ORIGINS=' "$APP_DIR/.env" || \
    echo 'CSRF_TRUSTED_ORIGINS=https://wms.bolrus.ru' >>"$APP_DIR/.env"
fi

echo "==> venv + migrate"
sudo -u wms python3 -m venv "$APP_DIR/.venv"
sudo -u wms "$APP_DIR/.venv/bin/pip" install -U pip
sudo -u wms "$APP_DIR/.venv/bin/pip" install -r "$APP_DIR/requirements-server.txt"
sudo -u wms "$APP_DIR/.venv/bin/python" "$APP_DIR/manage.py" migrate --noinput
sudo -u wms "$APP_DIR/.venv/bin/python" "$APP_DIR/manage.py" seed
sudo -u wms "$APP_DIR/.venv/bin/python" "$APP_DIR/manage.py" collectstatic --noinput

echo "==> systemd wms"
cp "$APP_DIR/deploy/wms.service" /etc/systemd/system/wms.service
systemctl daemon-reload
systemctl enable --now wms
systemctl is-active --quiet wms
curl -sI http://127.0.0.1:8050/ | head -n 5 || true

echo "==> Traefik"
install -d -m 755 "$APP_DIR/deploy/traefik"
touch "$APP_DIR/deploy/traefik/acme.json"
chmod 600 "$APP_DIR/deploy/traefik/acme.json"
chown root:root "$APP_DIR/deploy/traefik/acme.json"
cd "$APP_DIR/deploy/traefik"
docker compose up -d
docker compose ps

echo
echo "Готово."
echo "  Локально:  curl -I http://127.0.0.1:8050/"
echo "  HTTPS:     curl -I https://wms.bolrus.ru   (нужны DNS + проброс 80/443)"
echo "  Логин:     admin / PIN 0000"
echo "  .env:      ${APP_DIR}/.env"

#!/usr/bin/env bash
# Перенос GitHub Actions runner на пользователя wms (приложение + деплой = один uid).
# Не трогает Docker, PostgreSQL, Traefik, ControlGate и прочие службы.
# Запуск: sudo bash deploy/migrate-runner-to-wms.sh
set -euo pipefail

if [[ "$(id -u)" -ne 0 ]]; then
  echo "Нужен root: sudo bash $0" >&2
  exit 1
fi

OLD="${OLD_RUNNER_DIR:-/home/bolrus/actions-runner}"
NEW="${NEW_RUNNER_DIR:-/home/wms/actions-runner}"
APP_DIR="${WMS_APP_DIR:-/opt/wms}"
UNIT="actions.runner.RustamHash-wms_bolrus_django.wms.service"

echo "==> Stop/uninstall runner (bolrus)"
if [[ -x "$OLD/svc.sh" ]]; then
  (cd "$OLD" && ./svc.sh stop) || true
  (cd "$OLD" && ./svc.sh uninstall) || true
fi
systemctl stop "$UNIT" 2>/dev/null || true
systemctl disable "$UNIT" 2>/dev/null || true
rm -f /etc/systemd/system/"$UNIT"
systemctl daemon-reload

if [[ ! -f "$OLD/.runner" ]]; then
  echo "Нет $OLD/.runner — runner не найден у bolrus." >&2
  exit 1
fi

if [[ -e "$NEW" ]]; then
  echo "Уже есть $NEW — удалите вручную, если это старый мусор, и перезапустите." >&2
  exit 1
fi

echo "==> Copy runner → $NEW (outside $APP_DIR, чтобы rsync --delete не стёр)"
install -d -o wms -g wms /home/wms
rsync -a "$OLD"/ "$NEW"/
chown -R wms:wms /home/wms
chown -R wms:wms "$NEW"

echo "==> App tree owner wms"
chown -R wms:wms "$APP_DIR"
if [[ -f "$APP_DIR/deploy/traefik/acme.json" ]]; then
  chown root:root "$APP_DIR/deploy/traefik/acme.json"
  chmod 600 "$APP_DIR/deploy/traefik/acme.json"
fi

echo "==> Install/start runner service as wms"
cd "$NEW"
./svc.sh install wms
./svc.sh start
sleep 2
./svc.sh status || systemctl status "$UNIT" --no-pager -l

echo "==> sudoers for deploy restart"
cat >/etc/sudoers.d/wms-deploy <<'EOF'
wms ALL=(root) NOPASSWD: /bin/systemctl restart wms, /bin/systemctl is-active wms
EOF
chmod 440 /etc/sudoers.d/wms-deploy
visudo -cf /etc/sudoers.d/wms-deploy

echo "==> Checks"
systemctl is-active "$UNIT"
sudo -u wms -n systemctl is-active wms
sudo -u wms test -w "$APP_DIR" && echo "wms can write $APP_DIR"

echo
echo "Готово. Runner должен быть Idle на GitHub."
echo "Старый каталог можно удалить позже: rm -rf $OLD"
echo "Re-run failed Deploy в GitHub Actions."

# Деплой WMS BolRus (сервер с нуля)

Стек: Ubuntu → PostgreSQL → Django (gunicorn/systemd на `127.0.0.1:8050`) → Traefik (Docker, `network_mode: host`, HTTPS).  
Домен: `wms.bolrus.ru`. LAN-сервер: `192.168.1.109`.  
Traefik слушает 80/443 на хосте и проксирует на `http://127.0.0.1:8050` (gunicorn снаружи не торчит).

После установки вход: пользователь `admin` / PIN `0000` (из `seed`).

---

## 0. До сервера

1. DNS: A-запись `wms.bolrus.ru` → **публичный** IP сервера (нужно для Let's Encrypt).
2. На роутере проброс **80** и **443** на `192.168.1.109` (если сервер за NAT).
3. Код репозитория доступен (git clone или rsync с вашей машины).

---

## 1. Базовые пакеты

Под root / sudo на Ubuntu:

```bash
sudo apt update
sudo apt install -y \
  git curl ca-certificates \
  python3 python3-venv python3-pip \
  postgresql postgresql-contrib \
  docker.io docker-compose-v2 \
  rsync ufw \
  libreoffice-writer \
  fonts-liberation fonts-crosextra-carlito fonts-crosextra-caladea fonts-dejavu-core \
  cups-client

sudo systemctl enable --now postgresql docker
```

Firewall:

```bash
sudo ufw allow OpenSSH
sudo ufw allow 80/tcp
sudo ufw allow 443/tcp
sudo ufw --force enable
sudo ufw status
```

---

## 2. PostgreSQL

Придумайте пароль БД (ниже `CHANGE_ME_DB_PASSWORD`):

```bash
sudo -u postgres psql <<'SQL'
CREATE USER wms WITH PASSWORD 'CHANGE_ME_DB_PASSWORD';
CREATE DATABASE wms_django OWNER wms;
GRANT ALL PRIVILEGES ON DATABASE wms_django TO wms;
SQL
```

На новых PostgreSQL (15+) дополнительно:

```bash
sudo -u postgres psql -d wms_django -c "GRANT ALL ON SCHEMA public TO wms;"
```

Проверка:

```bash
psql "postgresql://wms:CHANGE_ME_DB_PASSWORD@127.0.0.1:5432/wms_django" -c 'SELECT 1'
```

## 3. Пользователь и код

```bash

```

**A) git clone** (подставьте URL репозитория):


**B) с вашего ПК (PowerShell / scp):**

```bash

```

## 7. Traefik (HTTPS, host network)

`network_mode: host` — порты 80/443 на хосте, бэкенд `http://127.0.0.1:8050`.  
Email ACME: `traefik.yml` → `certificatesResolvers.le.acme.email`.

```bash
cd /opt/wms/deploy/traefik
sudo touch acme.json
sudo chmod 600 acme.json
sudo docker compose down
sudo docker compose up -d
sudo docker compose logs --tail=40
curl -skI -H 'Host: wms.bolrus.ru' https://127.0.0.1/ | head -5
```

Ожидается `HTTP/2 302` или `200` (не 502).  
ACME: DNS A + проброс 80/443. CI не перезапускает Traefik — конфиг меняется редко, вручную как выше.

---

## 8. ПК в локальной сети (Windows)

Пока DNS с ПК резолвит публичный IP, внутри офиса удобнее hosts → `192.168.1.109`:

1. Скопировать на ПК `deploy/windows/`
2. ПКМ `set-wms-hosts.cmd` → **Запуск от имени администратора**
3. Открыть `https://wms.bolrus.ru`

---

## 9. Автодеплой GitHub Actions

Один пользователь **`wms`**: gunicorn (`/opt/wms`) + self-hosted runner (`/home/wms/actions-runner`).  
Runner **не** кладём внутрь `/opt/wms` — иначе `rsync --delete` сотрёт его.

Workflow: [`.github/workflows/deploy.yml`](../.github/workflows/deploy.yml) → [`update.sh`](update.sh).

Passwordless sudo только для `wms`:

```
wms ALL=(root) NOPASSWD: /bin/systemctl restart wms, /bin/systemctl is-active wms
```

Если runner уже стоял от другого пользователя — один раз:  
`sudo bash deploy/migrate-runner-to-wms.sh` (см. скрипт в репозитории).

Файл `/opt/wms/.env` деплоем **не** перезаписывается.

После смены unit-файла:

```bash
sudo cp /opt/wms/deploy/wms.service /etc/systemd/system/wms.service
sudo systemctl daemon-reload
sudo systemctl restart wms
```

## Быстрая диагностика

| Симптом | Куда смотреть |
|--------|----------------|
| Нет ответа на 8050 | `systemctl status wms`, `journalctl -u wms -n 80` |
| Traefik не стартует | `acme.json` должен быть **файлом** chmod 600, не каталогом |
| Нет сертификата | DNS A, порт 80 с интернета, `docker compose logs` |
| CSRF / cookie | `DEBUG=0`, `CSRF_TRUSTED_ORIGINS=https://wms.bolrus.ru` |
| LAN не открывается | hosts на ПК, ufw 80/443, Traefik слушает 0.0.0.0 |

Локальная разработка на Windows: `DEBUG=1`, `py manage.py runserver 0.0.0.0:8050` — без Traefik.

#Requires -RunAsAdministrator
param(
    [string]$ServerIP = "192.168.1.109",
    [string]$Hostname = "wms.bolrus.ru"
)

$ErrorActionPreference = "Stop"
$hostsPath = Join-Path $env:SystemRoot "System32\drivers\etc\hosts"
$marker = "# WMS BolRus"
$entry = "$ServerIP`t$Hostname`t$marker"

if (-not (Test-Path $hostsPath)) {
    throw "hosts file not found: $hostsPath"
}

$content = Get-Content -Path $hostsPath -ErrorAction Stop
$filtered = @(
    $content | Where-Object {
        $_ -notmatch '(?i)^\s*\d+\.\d+\.\d+\.\d+\s+wms\.bolrus\.ru(\s|$)' -and
        $_ -notmatch '(?i)WMS BolRus'
    }
)
$filtered += $entry

# ASCII avoids PowerShell encoding/parser issues on Windows
Set-Content -Path $hostsPath -Value $filtered -Encoding Ascii

ipconfig /flushdns | Out-Null

Write-Host "OK: $Hostname -> $ServerIP"
Write-Host "Open: https://$Hostname/"

# Как работать в WMS

Складская программа для приёмки, отбора и отгрузки овощей и фруктов. Офис работает в браузере компьютера, приёмщики и кладовщики — на ТСД. Это одни и те же данные.

Килограммы везде нетто: вес товара без деревяшки и без тары упаковки.

## Как открыть

На этом компьютере:

- офис: http://127.0.0.1:8001/login
- ТСД: http://127.0.0.1:8001/tsd/login

С телефона или ТСД страница откроется, только если сервер запущен для всей сети, а не только для этого компьютера. Адрес тогда такой: `http://адрес-компьютера:8001/tsd/login`.

Первый вход после пустой базы:

| Кто | Логин | Пароль | PIN на ТСД |
| --- | --- | --- | --- |
| Администратор | admin | admin | 0000 |
| Приёмка | receiver | receiver | 1111 |
| Склад | warehouse | warehouse | 2222 |

Пароли и PIN лучше сменить в справочнике «Пользователи».

## Кто что может

- **Приёмщик** принимает товар.
- **Кладовщик** отбирает и отгружает.
- **Офис** ведёт справочники, заказы, задания и может делать складскую работу.
- **Администратор** делает всё то же и открывает раздел «Сервис»: ТСД и админку.

В системе должен остаться хотя бы один активный пользователь офиса или администратор. Последнего такого пользователя программа не даст отключить и не даст сменить ему роль.

## Офис

Меню сверху: Заказы, Задания, Отчеты, Накладные, Справочники. У администратора ещё «Сервис».

Главная страница показывает открытые заказы, число заданий на приёмку, отбор и отгрузку и килограммы на остатке.

### Справочники

Их заполняют до первых заказов:

- группы и товары;
- калибры и сортность;
- тара (вес одной упаковки);
- поставщики, клиенты, продавцы;
- принтеры;
- пользователи.

Продавец печатается на стикере отгрузки и не делит складской остаток. Два продавца пользуются одним остатком.

Тара упаковки задаётся в килограммах. Ноль значит, что вес упаковки при приёмке не вычитается. Вес деревянного поддона к таре упаковки не относится: европоддон 25 кг, американец 30 кг, программа подставляет его сама.

### Заказ покупателя

Заказы → Покупателям → новый заказ.

Указываются покупатель, продавец, дата отгрузки, условия отгрузки и строки: товар, калибр, сортность, килограммы. Номер заказа начинается с **ЗК**.

Калибр и сортность строки можно поменять, пока по ней нет прихода и отбора. Открытое задание получит те же значения. Если по строке уже есть движение, меняется только количество.

Из заказа создаётся задание на отбор или отгрузку. Номер задания начинается с **ЗД**.

### Заказ поставщику

Заказы → Поставщикам.

Указываются поставщик, дата документа, дата поступления, статус и строки. Номер начинается с **ЗП**. Статусы: черновик, к поступлению, закрыт.

Из заказа создаётся задание на приёмку. В задании можно указать ТТН.

### Задания

Список делится на все, приёмку, отбор и «на отгрузку».

В приёмке принимают паллету: строка задания, калибр, сортность, вид поддона, упаковка, число упаковок, грязный вес. Чистый вес считается сам: грязный минус вес поддона и минус тара одной упаковки, умноженная на число упаковок. Поле можно поправить руками.

Принятую паллету можно исправить или удалить, пока по ней нет отбора и отгрузки.

На отборе кладовщик забирает подходящую принятую паллету в заказ. На отгрузке отмечает, что условия проверены, и паллета уходит в накладную.

### Отчёты

- **Остатки** — свободно и отобрано. Килограммы здесь не правят: их меняют только приёмка, отбор и отгрузка.
- **Потребность** — открытые строки заказов покупателей: заказано, отобрано, на остатке, в пути и не хватает. Список можно сгруппировать по товару, клиенту или дате.
- **Паллеты** — список и карточка паллеты.
- **Накладные** — документы отгрузки, номер начинается с **ОТ**.

## ТСД

Вход по PIN или по QR-бейджу. Бейдж печатается из карточки пользователя.

Меню:

1. Приёмка
2. Отбор
3. На отгрузку
4. Поиск по штрихкоду

Цифры 1–9 на экране заданий открывают задание, 0 возвращает в меню.

Штрихкод места начинается с **HU**, штрихкод паллеты — с **PL**.

## Печать

Термопринтер TSC печатает этикетку по сети, порт 9100. Паспорт паллеты и накладная — PDF на сетевой принтер Windows.

Принтер задаётся в справочнике. Пока его нет, печать сообщит, что термопринтер TSC или сетевой принтер не задан. PDF можно открыть и без принтера.

## Номера документов

| Документ | Начало номера |
| --- | --- |
| Заказ покупателя | ЗК |
| Заказ поставщику | ЗП |
| Задание | ЗД |
| Приход | ПР |
| Паллета | ПЛ |
| Накладная | ОТ |

# Регламент приёмки

Приёмка начинается, когда машина стоит на рампе, и заканчивается, когда по каждой паллете есть паспорт, а общий чистый вес сходится с заказом и с документами водителя.

Вес везде чистый: грязный минус деревянный поддон и минус тара упаковки.

## 1. Узнаём, кто привёз

У водителя берём поставщика, товар и товарно-транспортную накладную.

Дальше не идём, пока не ясно, чей это груз. Без поставщика заказ не ищем и машину не разгружаем.

## 2. Берём заказ поставщику из 1С

По поставщику находим заказ и печатаем его либо открываем на экране. В WMS это заказ поставщику, из него уже должно быть задание на приёмку. Номер ТТН вписываем в задание.

Если заказа нет, товар не выгружаем. Звоним в офис: заказ заводят, либо машину отправляют.

## 3. Читаем комментарий заказа до выгрузки

В комментарии смотрим, как принимать эту поставку:

- кому товар: в общий остаток или в резерв под клиента;
- какой калибр и какая сортность;
- какая упаковка и какой поддон;
- что обязательно снять на фото или видео;
- нужен ли анализ, температура или другой документ.

Комментарий задаёт условия этой машины. Если его не прочитать, паллету легко принять не тем калибром или отдать в общий остаток чужой заказ.

## 4. Проверяем качество до полной выгрузки

Сначала смотрим образец, не всю машину.

- Снимаем фото и видео качества так, как написано в комментарии: общий вид, дефект, этикетка, вскрытие.
- Сверяем сортность и калибр с заказом.
- Смотрим анализ и сопроводительные документы, если они требуются.

Товар явно не тот, гнилой или без нужного анализа — выгрузку останавливаем и зовём офис. Частичную приёмку делаем только после их решения.

## 5. Выгружаем, перевешиваем, печатаем паспорт

Каждую паллету принимаем отдельно:

- строка заказа;
- фактические калибр и сортность;
- вид поддона и упаковка, число мест;
- грязный вес с весов, чистый вес проверяем;
- клиент, если паллета ждёт его заказ.

На каждую паллету печатаем паспорт и клеим его сразу. Паллета без паспорта на складе не стоит.

## 6. Подтверждаем приёмку

Сводим чистый вес отдельно по товару, калибру и сортности. Этот итог сверяем с заказом и с документами водителя.

Расхождение проговариваем с офисом до закрытия приёмки. После подтверждения задание закрывают, и принятый вес становится остатком.
