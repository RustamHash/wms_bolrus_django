def wms(request):
    storage = messages.get_messages(request)
    flashes = []
    for message in storage:
        level = "error" if message.level >= messages.ERROR else "ok"
        text = str(message)
        if flashes and flashes[-1]["m"] == text and flashes[-1]["l"] == level:
            continue
        flashes.append({"m": text, "l": level})
    notices, unread = notice_state(request.user)
    return {
        "flashes": flashes,
        "notices": notices,
        "notice_unread": unread,
        "ROLE_LABELS": ROLE_LABELS,

def accept(get=None, post=None):
    def view(request, *args, **kwargs):
        handler = post if request.method == "POST" else get
        if handler is None:
            return HttpResponseNotAllowed(["POST"] if post else ["GET"])
        return handler(request, *args, **kwargs)

    return view


def roles_required(*roles, tsd=()):
    def decorator(view):
        @wraps(view)
        def wrapped(request, *args, **kwargs):
            user = request.user
            path = request.path
            on_tsd = path.startswith("/tsd")
            login_url = "/tsd/login" if on_tsd else "/login"
            if not user.is_authenticated or not user.is_active:
                return redirect(login_url)
            if not on_tsd and request.method != "POST" and not user.opens_web():
                if user.opens_tsd() and not (path.startswith("/shift") and user.can_shift()):
                    return redirect("/tsd")
            needed = set(tsd)
            if needed and needed & user.granted_access():
                return view(request, *args, **kwargs)
            if "shift" in roles and user.can_shift():
                return view(request, *args, **kwargs)
            if on_tsd and path.rstrip("/") == "/tsd" and user.can_shift():
                return view(request, *args, **kwargs)
            if user.role != "admin":
                allowed = set(roles)
                if "office" not in allowed:
                    allowed.add("office")
                if user.role not in allowed:
                    messages.error(request, "Недостаточно прав")
                    return redirect("/tsd" if on_tsd or user.opens_tsd() else "/")
            return view(request, *args, **kwargs)

        return wrapped

    return decorator


def guarded(request, back, fn):
    try:
        with transaction.atomic():
            return fn(), None
    except WarehouseError as exc:
        messages.error(request, str(exc))
        return None, redirect(back)
    except (IntegrityError, ProtectedError):
        messages.error(request, "Запись уже есть или используется")
        return None, redirect(back)


def done(request, url, message):
    messages.success(request, message)
    return redirect(url)


def back_url(request, fallback: str) -> str:
    return request.META.get("HTTP_REFERER") or fallback

from apps.common.models import Notice, NoticeCursor


def notify(user, lines: list[str]) -> None:
    if not lines:
        return
    actor = (getattr(user, "name", "") or getattr(user, "login", "") or "").strip() or "Пользователь"
    Notice.objects.bulk_create(
        [Notice(user=user if getattr(user, "pk", None) else None, actor=actor, text=f"{actor} — {line}") for line in lines]
    )


def notice_state(user) -> tuple[list[Notice], int]:
    if not getattr(user, "is_authenticated", False):
        return [], 0
    seen = NoticeCursor.objects.filter(user=user).values_list("seen_id", flat=True).first() or 0
    latest = list(Notice.objects.order_by("-id")[:80])
    for item in latest:
        item.unread = item.id > seen
    unread = Notice.objects.filter(id__gt=seen).count()
    return latest, unread


def mark_notices_seen(user) -> None:
    latest = Notice.objects.order_by("-id").values_list("id", flat=True).first() or 0
    NoticeCursor.objects.update_or_create(user=user, defaults={"seen_id": latest})

"""Пресеты периода как в журнале заказов: сегодня, неделя, месяц, своя дата."""
from datetime import date, timedelta

from django.utils import timezone

from apps.common.forms import query_date


def period_date(period_type: str, today: date | None = None) -> date | None:
    today = today or timezone.localdate()
    if period_type == "today":
        return today
    if period_type == "yesterday":
        return today - timedelta(days=1)
    if period_type == "tomorrow":
        return today + timedelta(days=1)
    if period_type == "start_of_week":
        return today - timedelta(days=today.weekday())
    if period_type == "start_of_last_week":
        return today - timedelta(days=today.weekday() + 7)
    if period_type == "end_of_week":
        return today + timedelta(days=6 - today.weekday())
    if period_type == "end_of_last_week":
        return today + timedelta(days=6 - today.weekday() - 7)
    if period_type == "start_of_month":
        return today.replace(day=1)
    if period_type == "start_of_last_month":
        first = today.replace(day=1)
        return (first - timedelta(days=1)).replace(day=1)
    if period_type == "end_of_month":
        if today.month == 12:
            return date(today.year, 12, 31)
        return date(today.year, today.month + 1, 1) - timedelta(days=1)
    if period_type == "end_of_last_month":
        return today.replace(day=1) - timedelta(days=1)
    return None


def _bound(kind: str, raw: str) -> date | None:
    if kind == "custom":
        return query_date(raw)
    if kind:
        return period_date(kind)
    return None


def range_from_query(params, prefix: str, *, default_from: str = "") -> dict:
    """Границы «от/до» по пресету. Пустой «от» при заданном «до» считается сегодняшним днём."""
    from_type = (params.get(f"{prefix}_from_type") or "").strip()
    to_type = (params.get(f"{prefix}_to_type") or "").strip()
    from_value = (params.get(f"{prefix}_from") or "").strip()
    to_value = (params.get(f"{prefix}_to") or "").strip()
    if not any((from_type, to_type, from_value, to_value)):
        if not default_from:
            return {
                "from_type": "",
                "from_value": "",
                "to_type": "",
                "to_value": "",
                "from_date": None,
                "to_date": None,
            }
        from_type = default_from
    from_date = _bound(from_type, from_value)
    to_date = _bound(to_type, to_value)
    if from_date is None and (to_type or to_value):
        from_date = period_date("today")
    return {
        "from_type": from_type,
        "from_value": from_value if from_type == "custom" else "",
        "to_type": to_type,
        "to_value": to_value if to_type == "custom" else "",
        "from_date": from_date,
        "to_date": to_date,
    }

from dataclasses import dataclass, field
from decimal import Decimal

from apps.common.kg import as_kg, format_kg


def line_caliber(line) -> str:
    if line.caliber_mode == "exact" and line.caliber_id:
        return line.caliber.name
    if line.caliber_mode == "range" and line.caliber_from_id and line.caliber_to_id:
        return f"{line.caliber_from.name}–{line.caliber_to.name}"
    return ""


def describe_line(line) -> str:
    parts = [line.product_title()]
    caliber = line_caliber(line)
    if caliber:
        parts.append(caliber)
    grades = sorted(line.grades.all(), key=lambda item: (item.sort_order, item.name))
    if grades:
        parts.append("сортность: " + ", ".join(item.name for item in grades))
    return ", ".join(parts)


def unit_lines(unit) -> list[str]:
    lot = unit.lot
    return [
        lot.product.name,
        f"{lot.caliber.name} · {unit.grade.name}",
        f"{format_kg(unit.net_kg)} кг · {unit.packaging.name}",
        f"{lot.supplier.name} · {lot.received_on.strftime('%d.%m.%Y')}",
    ]


@dataclass
class UnitGroup:
    product: str
    caliber: str
    grade: str
    count: int = 0
    kg: Decimal = field(default_factory=lambda: Decimal("0.000"))
    units: list = field(default_factory=list)


def group_units(units) -> list[UnitGroup]:
    order: list[tuple] = []
    buckets: dict[tuple, UnitGroup] = {}
    for unit in units:
        if unit.status in ("rejected", "transformed"):
            continue
        key = (unit.lot.product.name, unit.lot.caliber.name, unit.grade.name)
        if key not in buckets:
            buckets[key] = UnitGroup(*key)
            order.append(key)
        group = buckets[key]
        group.units.append(unit)
        group.count += 1
        group.kg += as_kg(unit.net_kg)
    return [buckets[key] for key in order]

            root = ET.fromstring(book.read("xl/sharedStrings.xml"))
            for item in root.findall("m:si", NS):
                shared.append("".join(node.text or "" for node in item.findall(".//m:t", NS)))
        sheet_name = f"xl/worksheets/sheet{sheet}.xml"
        if sheet_name not in book.namelist():
            raise ValueError(f"В книге нет листа {sheet}")
        root = ET.fromstring(book.read(sheet_name))
    rows: dict[int, dict[str, str]] = {}
    for cell in root.findall(".//m:c", NS):
        ref = cell.get("r") or ""
        kind = cell.get("t")

        inline = cell.find("m:is", NS)
        text = ""
        if kind == "s" and value is not None and value.text:
            text = shared[int(value.text)]

_STYLES = {
    "text": 0,
    "header": 1,
    "num": 2,
    "date": 3,

    "group_num": 5,
    "total": 6,

def _col_name(index: int) -> str:
    name = ""
    while index:
        index, rem = divmod(index - 1, 26)
        name = chr(65 + rem) + name
    return name


def _excel_day(value: date) -> int:
    return (value - date(1899, 12, 30)).days


def _split_cell(cell):
    style = None
    if isinstance(cell, tuple):
        cell, style = cell
    if isinstance(cell, datetime):
        cell = cell.date()
    if style is None:
        if isinstance(cell, date):
            style = "date"
        elif isinstance(cell, (Decimal, int, float)) and not isinstance(cell, bool):
            style = "num"
        else:
            style = "text"
    return cell, _STYLES.get(style, 0)


def _cell_xml(ref: str, cell) -> str:
    value, style = _split_cell(cell)
    if value is None or value == "":
        if not style:
            return ""
        return f'<c r="{ref}" s="{style}"/>'
    if isinstance(value, date):
        return f'<c r="{ref}" s="{style}"><v>{_excel_day(value)}</v></c>'
    if isinstance(value, Decimal):
        return f'<c r="{ref}" s="{style}"><v>{format(value, "f")}</v></c>'
    if isinstance(value, bool):
        value = "1" if value else "0"
    if isinstance(value, (int, float)):
        return f'<c r="{ref}" s="{style}"><v>{value}</v></c>'
    text = escape(str(value))
    return f'<c r="{ref}" t="inlineStr" s="{style}"><is><t xml:space="preserve">{text}</t></is></c>'


def _sheet_xml(rows: list[list]) -> str:
    width = max((len(row) for row in rows), default=1)
    lengths = [10] * width
    body = []
    for row_index, row in enumerate(rows, start=1):
        cells = []
        for col_index, cell in enumerate(row, start=1):
            value, _style = _split_cell(cell)
            shown = "" if value is None else str(value)
            lengths[col_index - 1] = min(48, max(lengths[col_index - 1], len(shown) + 2))
            xml = _cell_xml(f"{_col_name(col_index)}{row_index}", cell)
            if xml:
                cells.append(xml)
        body.append(f'<row r="{row_index}">{"".join(cells)}</row>')
    cols = "".join(
        f'<col min="{index}" max="{index}" width="{width}" customWidth="1"/>'
        for index, width in enumerate(lengths, start=1)
    )
    return (
        '<?xml version="1.0" encoding="UTF-8" standalone="yes"?>'
        '<worksheet xmlns="http://schemas.openxmlformats.org/spreadsheetml/2006/main">'
        f"<cols>{cols}</cols><sheetData>{''.join(body)}</sheetData></worksheet>"
    )

def parse_fact_kg(raw: str) -> Decimal:
    value_text = (raw or "").strip().replace(",", ".")
    if not value_text:
        raise WarehouseError("Укажите фактический вес. Ноль — паллеты нет")
    try:
        value = Decimal(value_text)
    except InvalidOperation as exc:
        raise WarehouseError("Некорректный вес") from exc
    value = value.quantize(Decimal("0.001"))
    if value < 0:
        raise WarehouseError("Вес не может быть отрицательным")
    return value

def start_document(document: InventoryDocument) -> int:
    document = InventoryDocument.objects.select_for_update().get(pk=document.pk)

        raise WarehouseError("Начать можно только черновик")
    product_ids = None

        product_ids = list(document.products.values_list("pk", flat=True))

            raise WarehouseError("Отметьте номенклатуру")
    lines = []
    for movement in stock_inbounds(product_ids):
        lines.append(

            )
        )
    if lines:
        InventoryLine.objects.bulk_create(lines)
    document.status = InventoryDocument.Status.COUNTING
    document.save(update_fields=["status"])
    return len(lines)

def _post_found(line: InventoryLine, user) -> None:

    pallet.barcode = barcode("PL", pallet.id)
    pallet.save(update_fields=["number", "barcode"])

def _post_line(line: InventoryLine, user) -> None:
    if line.pallet_id is None:
        _post_found(line, user)
        return
    movement = (
        Movement.objects.select_for_update()
        .filter(pallet_id=line.pallet_id, kind=Movement.Kind.IN)
        .select_related("lot", "pallet")
        .first()
    )
    label = line.pallet.number or f"#{line.pallet_id}"
    if movement is None:
        raise WarehouseError(f"Нет прихода по паллете {label}")
    if has_left(movement):
        raise WarehouseError(f"По паллете {label} уже есть отбор, отгрузка или переработка")
    fact = as_kg(line.fact_kg)
    if fact == 0:
        remove_in(movement)
        return
    book = as_kg(line.book_kg)
    supplier_changed = line.supplier_id != line.lot.supplier_id
    if fact == book and not supplier_changed:
        return
    lot = movement.lot
    qty = fact if fact != book else as_kg(movement.qty_kg)
    if supplier_changed:
        lot = _lot_for_supplier(movement.lot, line.supplier_id)
    if lot.id == movement.lot_id and as_kg(movement.qty_kg) == qty:
        return
    update_in(movement, lot=lot, qty_kg=qty)


@transaction.atomic
def post_document(document: InventoryDocument, user) -> None:
    document = InventoryDocument.objects.select_for_update().get(pk=document.pk)
    if document.status != InventoryDocument.Status.COUNTING:
        raise WarehouseError("Провести можно только документ в подсчёте")
    lines = list(document.lines.select_related("lot", "pallet", "supplier", "product", "caliber", "grade", "cell__zone"))
    pending = sum(1 for line in lines if line.fact_kg is None)
    if pending:
        raise WarehouseError(f"Не посчитано паллет: {pending}")
    for line in lines:
        _post_line(line, user)
    document.status = InventoryDocument.Status.POSTED
    document.posted_by = user
    document.posted_at = timezone.now()
    document.save(update_fields=["status", "posted_by", "posted_at"])

@roles_required("office")
def inventory_create(request):
    def action():
        scope = text(request.POST, "scope") or InventoryDocument.Scope.ALL
        raw_ids = request.POST.getlist("product_id")
        product_ids = []
        for raw in raw_ids:
            if not raw:
                continue
            try:
                product_ids.append(int(raw))
            except ValueError as exc:
                raise WarehouseError("Некорректная номенклатура") from exc
        if scope == InventoryDocument.Scope.PRODUCTS:
            known = set(Product.objects.filter(pk__in=product_ids).values_list("pk", flat=True))
            if len(known) != len(set(product_ids)):
                raise WarehouseError("Номенклатура не найдена")
        else:
            product_ids = []
        return create_document(
            doc_date=parse_date(text(request.POST, "doc_date"), "Дата"),
            scope=scope,
            product_ids=product_ids,
            notes=text(request.POST, "notes"),
            user=request.user,
        )

    result, response = guarded(request, "/inventory/new", action)
    if response:
        return response
    return redirect(f"/inventory/{result.id}")

def move_named_pallets(apps, schema_editor):
    Movement = apps.get_model("inventory", "Movement")
    Balance = apps.get_model("inventory", "Balance")
    Pallet = apps.get_model("storage", "Pallet")
    arrivals = Movement.objects.filter(kind="in").select_related("pallet", "task_line")
    for movement in arrivals:
        pallet = movement.pallet
        if pallet.order_id is None:
            continue
        if Movement.objects.filter(pallet_id=pallet.id, kind__in=("pick", "out")).exists():
            continue
        if movement.task_line_id and movement.task_line.order_line_id and pallet.reserved_line_id is None:
            Pallet.objects.filter(pk=pallet.id).update(reserved_line_id=movement.task_line.order_line_id)
        balance = Balance.objects.filter(lot_id=movement.lot_id).first()
        if balance is None:
            continue
        qty = movement.qty_kg if balance.free_kg >= movement.qty_kg else balance.free_kg
        if qty <= 0:
            continue
        balance.free_kg = balance.free_kg - qty
        balance.reserved_kg = (balance.reserved_kg or Decimal("0")) + qty
        balance.save(update_fields=["free_kg", "reserved_kg"])
