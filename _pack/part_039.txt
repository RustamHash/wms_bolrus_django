class Batch(models.Model):
    """ Электронный паспорт партии. Ключевой объект Traceability. """
    product = models.ForeignKey(Product, on_delete=models.PROTECT)
    vendor = models.ForeignKey(Vendor, on_delete=models.PROTECT)
    inbound_item = models.ForeignKey(InboundItem, on_delete=models.PROTECT, related_name='batches', null=True, blank=True)
    batch_number = models.CharField("Номер партии", max_length=50, unique=True)
    production_date = models.DateField("Дата сбора")
    expiration_date = models.DateField("Срок годности (для FEFO)")
    reserved_for_order = models.ForeignKey('outbound.CustomerOrder', on_delete=models.SET_NULL, null=True, blank=True)
    reserved_for_customer = models.ForeignKey('outbound.Customer', on_delete=models.SET_NULL, null=True, blank=True)

class SortingWorkOrder(models.Model):
    """ Документ: Наряд-заказ на конвейерную переборку """
    STATUS_CHOICES = [('IN_PROGRESS', 'В процессе'), ('COMPLETED', 'Завершен')]
    status = models.CharField(max_length=15, choices=STATUS_CHOICES, default='IN_PROGRESS')
    batch = models.ForeignKey(Batch, on_delete=models.PROTECT, verbose_name="Перебираемая партия")
    
    total_input_weight = models.DecimalField("Всего вошло грязного веса, кг", max_digits=10, decimal_places=2, default=0.00)
    total_output_weight = models.DecimalField("Всего выпущено чистого веса, кг", max_digits=10, decimal_places=2, default=0.00)
    waste_weight = models.DecimalField("Взвешенные отходы (гниль), кг", max_digits=10, decimal_places=2, default=0.00)
    unaccounted_loss = models.DecimalField("Неучтенные потери / усушка, кг", max_digits=8, decimal_places=2, default=0.00)
    
    is_force_closed = models.BooleanField("Закрыто принудительно начальником", default=False)
    force_close_reason = models.TextField("Причина принудительного закрытия", blank=True)
    force_closed_by = models.ForeignKey(settings.AUTH_USER_MODEL, on_delete=models.SET_NULL, null=True, blank=True)
    created_at = models.DateTimeField(auto_now_add=True)

    def calculate_balance_delta(self):
        return self.total_input_weight - (self.total_output_weight + self.waste_weight)

class Pallet(models.Model):
    """ Грузовая единица (Контейнер LPN) """
    PALLET_STATUS = [
        ('NEW', 'Принят, ожидает размещения'),
        ('NEED_SORT', 'Требуется переборка (Заблокирован для отгрузок)'),
        ('READY', 'Кондиция (Доступен для FEFO)'),
        ('REJECT', 'Брак'),
    ]
    pallet_number = models.CharField("Штрихкод LPN", max_length=50, unique=True)
    batch = models.ForeignKey(Batch, on_delete=models.CASCADE, related_name='pallets')
    status = models.CharField(max_length=10, choices=PALLET_STATUS, default='NEW')
    grade = models.CharField("Реальный сорт товара (определяется при выпуске)", max_length=50, blank=True)
    
    # Математика тары фруктов/овощей
    gross_weight = models.DecimalField("Вес Брутто, кг", max_digits=8, decimal_places=2)
    pallet_tare = models.DecimalField("Вес поддона, кг", max_digits=5, decimal_places=2, default=20.0)
    package_type = models.ForeignKey(PackageType, on_delete=models.PROTECT, null=True, blank=True)
    package_count = models.PositiveIntegerField("Количество коробок/сеток", default=0)
    net_weight = models.DecimalField("Вес Нетто, кг", max_digits=8, decimal_places=2)

    consumed_in_sorting = models.ForeignKey(SortingWorkOrder, on_delete=models.SET_NULL, null=True, blank=True, related_name='input_pallets')
    created_in_sorting = models.ForeignKey(SortingWorkOrder, on_delete=models.SET_NULL, null=True, blank=True, related_name='output_pallets')

    def save(self, *args, **kwargs):
        pkg_tare = (self.package_type.weight * self.package_count) if self.package_type and self.package_count else 0
        self.net_weight = self.gross_weight - self.pallet_tare - pkg_tare
        super().save(*args, **kwargs)

class InventoryTransaction(models.Model):
    """ Регистр движений (Лог транзакций) """
    TRANSACTION_TYPES = [
        ('INBOUND', 'Приемка на рампу'),
        ('PUTAWAY', 'Размещение в ячейку'),
        ('SORT_IN', 'Списание в переборку'),
        ('SORT_OUT', 'Выпуск из переборки'),
        ('WASTE', 'Списание брака/гнили/усушки'),
    ]
    transaction_type = models.CharField(max_length=10, choices=TRANSACTION_TYPES)
    pallet = models.ForeignKey(Pallet, on_delete=models.PROTECT)
    from_cell = models.ForeignKey('Cell', on_delete=models.PROTECT, related_name='from_trans', null=True, blank=True)
    to_cell = models.ForeignKey('Cell', on_delete=models.PROTECT, related_name='to_trans', null=True, blank=True)
    quantity = models.DecimalField(max_digits=10, decimal_places=2)

    user = models.ForeignKey(settings.AUTH_USER_MODEL, on_delete=models.PROTECT)
    created_at = models.DateTimeField(auto_now_add=True)


### 2. Ключевые бизнес-правила и логика закрытия переборки

#### А. Принудительное закрытие наряда (force_close_order)
Обычное закрытие наряда через ТСД требует условия равенства баланса нулю. Если веса не бьются (из-за усушки овощей или скрытых потерь), бэкенд предоставляет специальный метод для Начальника смены (apps/inventory/services.py):

def force_close_sorting_order(work_order_id, manager_user, reason_text):
    """ Принудительное схлопывание баланса конвейера переборки """
    from django.db import transaction
    from django.core.exceptions import ValidationError
    
    with transaction.atomic():
        order = SortingWorkOrder.objects.select_for_update().get(id=work_order_id)
        if order.status == 'COMPLETED':
            raise ValidationError("Наряд уже закрыт.")
            
        delta = order.calculate_balance_delta() # Находим разницу веса
        
        if delta != 0:
            order.unaccounted_loss = delta
            # Генерируем балансировочную транзакцию списания усушки/потерь
            last_pallet = order.input_pallets.first() or order.output_pallets.first()
            
            InventoryTransaction.objects.create(
                transaction_type='WASTE',
                pallet=last_pallet,
                quantity=delta,
                reference_document_id=order.id,
                user=manager_user
            )
            
        order.status = 'COMPLETED'
        order.is_force_closed = True
        order.force_close_reason = reason_text
        order.force_closed_by = manager_user
        order.save()

## ЧАСТЬ 2: Технологический регламент для персонала (Инструкция склада)

### ГЛАВА 1: Приемка автотранспорта на рампе

1. Запуск процесса: Откройте интерфейс WMS на ТСД (или ПК). Найдите плановое Поступление, назначенное вам. Переведите его в статус «В работе».
2. Входной фото/инспекционный контроль: Пройдите шаги контроля на ТСД. Если физическая возможность загрузить фото отсутствует (внутренний рейс, форс-мажор), вы обязаны заполнить текстовое поле причины!
   * Шаг 1 (Госномер): Сделайте фото номера авто. Если ТС без номеров — оставьте коммент причины.
   * Шаг 2 (Кузов): Откройте ворота кузова, сделайте фото первого товарного ряда.
   * Шаг 3 (Температура фуры): Внесите в систему показатели температуры воздуха внутри рефрижератора.
   * Шаг 4 (Пакет документов): Заберите у водителя накладные и сертификаты.
     * Если документы есть: Отсканируйте всю пачку документов с помощью складского сканера или приложения ТСД в один многостраничный файл формата PDF и загрузите его в систему.
     * Если документов нет (внутренний рейс/дошлют в офис): Нажмите «Пропустить загрузку PDF» и обязательно заполните поле комментария (например: «Документы отсутствуют, рейс внутренний, сканы пришлют напрямую в офис»). Без комментария система заблокирует приемку.

### ГЛАВА 2: Выгрузка и Оценка качества
1. Выгрузка: Направьте рабочих на разгрузку поддонов в зону приемки (STAGE-RECEIPT).
2. Температурный тест плода: Возьмите игольчатый термометр-щуп. Измерьте температуру внутри плода из глубины поддона, зафиксируйте значение в ТСД.
3. Органолептическая проверка: Осмотрите ящики/сетки. 
   * Если товар чистый и калиброванный — принимайте стандартно.
   * Если товар идет навалом, вперемешку по калибрам, содержит землю, сор или следы порчи — сделайте фото дефекта, внесите примерный процент брака на глаз (например, 5%) и примите решение о приеме через переборку.

### ГЛАВА 3: Попаллетное взвешивание валового товара под переборку
1. Закатите грязный паллет на весы.
2. На экране ТСД выберите плановую строку «Товар несортированный (Вал)».
3. Учет тары: Укажите тип поддона (деревянный евро = 20 кг) и тип упаковки (сетка/биг-бэг). Введите количество упаковок на поддоне. Система автоматически вычислит чистый вес «грязного» нетто.
4. Печать этикетки LPN: Подтвердите взвешивание. Система присвоит паллету статус «Требуется переборка» (NEED_SORT) и заблокирует его для коммерческих отгрузок. 
5. Наклейте распечатанную этикетку со штрихкодом LPN на торец паллета, отсканируйте её для подтверждения.
6. Направьте паллет силами погрузчика в буферную зону ожидания сортировки (ZONE-BUFFER-SORTING).
7. Повторите процедуру для всей машины. По окончании разгрузки подпишите водителю документы и переведите Поступление на ПК в статус «Завершено».

### ГЛАВА 4: Техпроцесс конвейерной переборки (Зона сортировки)
Операция выполняется в плановом режиме силами сортировочной бригады (возможно, на следующий день).

1. Старт наряда: Старший смены на ТСД открывает «Наряд на переборку», привязанный к исходной партии заехавшего товара.
2. Подача сырья (Вход): Погрузчик привозит грязные паллеты из буфера. Рабочий у ленты конвейера обязан отсканировать штрихкод LPN грязного паллета перед тем, как высыпать его на ленту. Вес списывается в «котел» наряда.
3. Переборка: Сортировщики очищают овощи/фрукты от грязи, калибруют и укладывают в чистую складскую тару.
4. Выпуск готовой продукции (Выход): Как только на выходе из конвейера формируется строго однородный, полный паллет, весовщик закатывает его на весы.
   * Выберите на ТСД его реальный получившийся сорт (например, 1 сорт или 2 сорт).
   * Укажите чистый вес тары. Нажмите «Выпустить паллет».
   * Система распечатает этикетку со статусом «Кондиция» (READY). Наклейте её. Такой паллет погрузчик может сразу увозить на стеллажи постоянного хранения для отгрузки клиентам. Паллет сохраняет сквозную связь с исходным поставщиком.
5. Учет гнили и отходов: Гниль, землю и сор рабочие сбрасывают в технический бак. По мере заполнения бака завезите его на весы, нажмите на ТСД кнопку «Зафиксировать брак». WMS автоматически вычтет вес бака и мгновенно спишет чистый вес гнили в утиль для учета в претензиях.
6. Закрытие наряда (Валидация баланса): Когда партия полностью переработана, взвесьте финальный остаток (если поддон остался неполным) и нажмите «Закрыть наряд».
   * Сценарий А (Баланс сошелся): Входной вес равен сумме выходов и отходов. Наряд успешно закрывается.
   * Сценарий Б (Ошибка веса): ТСД выдает ошибку расхождения веса (из-за усушки, испарения влаги или погрешности). Перепроверьте, все ли баки с отходами были взвешены. Если ошибок персонала нет — вызовите Начальника смены.
7. Принудительное закрытие (Только для Начальника смены):
   * Начальник смены оценивает величину дельты. Так как процент отбраковки и усушки уникален для каждой партии и поставщика овощей, начальник имеет право списать разницу.
   * Начальник заходит под своей учетной записью в WMS, нажимает кнопку «Закрыть принудительно» и вносит обязательный комментарий причины (например: «Естественная усушка влаги и осыпание песка/земли при калибровке партии»).
   * Система автоматически проводит балансировочное списание остатков под ноль и успешно переводит наряд в статус «Завершен».
